export const INITIAL_FEEDBACK_OPERATION_LIMITS = Object.freeze({
  // Two independently bounded 4MiB scoring objects plus bounded sidecars/metadata.
  envelopeBytes: 12 * 1024 * 1024,
  visits: 500000,
  depth: 32,
});
export class InitialFeedbackContractError extends Error {
  constructor(

      | "not_found"
      | "stale"

function bounded(raw: unknown) {
  const stack = [{ value: raw, depth: 0 }];
  let visits = 0,
    bytes = 0;
  while (stack.length) {
    const { value, depth } = stack.pop()!;
    if (
      ++visits > INITIAL_FEEDBACK_OPERATION_LIMITS.visits ||
      depth > INITIAL_FEEDBACK_OPERATION_LIMITS.depth
    )
      throw bad();
    if (typeof value === "string") {
      if (
        !utf8(value) ||
        value.length > INITIAL_FEEDBACK_OPERATION_LIMITS.envelopeBytes
      )
        throw bad();
      bytes += Buffer.byteLength(JSON.stringify(value), "utf8");
    } else if (
      value === null ||
      typeof value === "boolean" ||
      typeof value === "number"
    ) {
      if (typeof value === "number" && !Number.isFinite(value)) throw bad();
      bytes += 24;
    } else if (Array.isArray(value)) {
      if (value.length > INITIAL_FEEDBACK_OPERATION_LIMITS.visits) throw bad();
      bytes += value.length + 2;
      for (const v of value) stack.push({ value: v, depth: depth + 1 });
    } else if (typeof value === "object" && value !== null) {
      const proto = Object.getPrototypeOf(value);
      if (proto !== Object.prototype && proto !== null) throw bad();
      const entries = Object.entries(Object.getOwnPropertyDescriptors(value));
      if (
        Reflect.ownKeys(value).length !== entries.length ||
        entries.length > INITIAL_FEEDBACK_OPERATION_LIMITS.visits
      )
        throw bad();
      for (const [key, d] of entries) {
        if (!d.enumerable || d.get || d.set || !utf8(key)) throw bad();
        bytes += Buffer.byteLength(JSON.stringify(key), "utf8") + 2;
        stack.push({ value: d.value, depth: depth + 1 });
      }
    } else if (value !== undefined) throw bad();
    if (bytes > INITIAL_FEEDBACK_OPERATION_LIMITS.envelopeBytes) throw bad();
  }
}
function wire<T extends z.ZodType>(schema: T) {
  return z.preprocess((raw, ctx) => {
    try {
      bounded(raw);
      return raw;
    } catch {
      ctx.addIssue({ code: "custom", message: "Wire resource or text limit." });
      return z.NEVER;
    }
  }, schema);
}
function decoded<T>(schema: z.ZodType<T>, raw: unknown): T {
  try {
    const v = schema.safeParse(raw);
    if (!v.success) throw bad();
    return v.data;
  } catch {
    throw bad();
  }
}

export function initialFeedbackPartialFeedback(
  d: z.infer<typeof LegacyScoringResultSchema>,
) {
  const unavailable = {
    score: 0,
    tips: Array.from({ length: 3 }, () => ({
      type: "improve" as const,
      tip: "Qualitative feedback is currently unavailable.",
      explanation:
        "Your saved deterministic checks remain available. Writing feedback was not completed.",
    })),
  };
  return {
    scoreSemantics: BASELINE_SEMANTICS,
    overallScore: d.score,
    ATS: {
      score: d.score,
      tips: d.ruleTrace
        .slice(0, 4)
        .map((r) => ({
          type: r.passed ? ("good" as const) : ("improve" as const),
          tip: r.detail,
        })),
    },
    toneAndStyle: unavailable,
    content: unavailable,
    structure: unavailable,
    skills: unavailable,
  };
}

                x.index === part.originalIndex &&

              part.title.textSha256 !== obs?.title.textSha256 ||
              part.explanation.textSha256 !== obs?.explanation.textSha256,

        }
        for (const part of parts.writer.bullets) {
          const row = p.prose.writer.bullets[part.retainedIndex],

              (x) => x.index === part.originalIndex && x.retained,

            ctx,

          p.compatibility.writer !== null ||

        issue(
          ctx,
          r.attemptId !== null ||
            p.recordedObservation !== null ||
            p.generationFailure !== null,
        );
      else issue(ctx, r.attemptId === null);
    }
  });

    isDeepStrictEqual(op.binding, p.binding) &&

      ? op.attempts.every((a) => a.claimedAt === null)
      : op.attempts.some((a) => a.attemptId === p.receipt.attemptId))
  );
}

const budgetReceipt = z

    schemaVersion: z.literal("ai-budget/1.0.0"),
    userId: id,
    organizationId: id,
    operationId: id,
    attemptId: id,
    attemptOrdinal: z.number().int().min(1).max(3),
    attemptLimit: z.number().int().min(1).max(3),
    surface: z.enum(["qualitative", "rewrite", "build"]),
    payloadSha256: sha,
    model: technicalVersion,
    promptPolicyVersion: technicalVersion,
    policyVersion: technicalVersion,
    poolPolicyVersion: technicalVersion,
    bindingSha256: sha,
    period,
    state: z.enum([
      "reserved",
      "dispatched",
      "held_unknown",
      "settled",
      "released",
    ]),
    reservedInputTokens: tokens,
    reservedOutputTokens: tokens,
    reservedCostMicrousd: decimal,
    inputMicrousdPerMillion: rate,
    outputMicrousdPerMillion: rate,
    actualInputTokens: tokens.nullable(),
    actualOutputTokens: tokens.nullable(),
    actualCostMicrousd: decimal.nullable(),
    replayed: z.boolean(),
    dispatchAllowed: z.boolean(),
    poolBlocked: z.boolean(),
  })
  .superRefine((r, ctx) => {
    if (
      ![
        r.reservedInputTokens,
        r.reservedOutputTokens,
        r.reservedCostMicrousd,
        r.inputMicrousdPerMillion,
        r.outputMicrousdPerMillion,
      ].every((x) => decimalPattern.test(x))
    )
      return;
    issue(
      ctx,
      r.attemptOrdinal > r.attemptLimit ||
        BigInt(r.reservedInputTokens) === 0n ||
        BigInt(r.reservedOutputTokens) === 0n,
    );
    if (
      BigInt(r.reservedInputTokens) > BigInt(Number.MAX_SAFE_INTEGER) ||
      BigInt(r.reservedOutputTokens) > BigInt(Number.MAX_SAFE_INTEGER)
    )
      return;
    const b = {
      userId: r.userId,
      organizationId: r.organizationId,
      operationId: r.operationId,
      attemptId: r.attemptId,
      attemptOrdinal: r.attemptOrdinal,
      surface: r.surface,
      payloadSha256: r.payloadSha256,
      model: r.model,
      promptPolicyVersion: r.promptPolicyVersion,
      policyVersion: r.policyVersion,
      reservedInputTokens: Number(r.reservedInputTokens),
      reservedOutputTokens: Number(r.reservedOutputTokens),
    };
    try {
      issue(ctx, hashAiBudgetBinding(b) !== r.bindingSha256);
    } catch {
      issue(ctx, true);
    }
    const cost = (i: string, o: string) =>
      (
        (BigInt(i) * BigInt(r.inputMicrousdPerMillion) +
          BigInt(o) * BigInt(r.outputMicrousdPerMillion) +
          999999n) /
        1000000n
      ).toString();
    issue(
      ctx,
      cost(r.reservedInputTokens, r.reservedOutputTokens) !==
        r.reservedCostMicrousd,
    );
    if (r.state === "settled")
      issue(
        ctx,
        r.actualInputTokens === null ||
          r.actualOutputTokens === null ||
          r.actualCostMicrousd === null,
      );
    else
      issue(
        ctx,
        r.actualInputTokens !== null ||
          r.actualOutputTokens !== null ||
          r.actualCostMicrousd !== null,
      );
    if (
      r.actualInputTokens !== null &&
      r.actualOutputTokens !== null &&
      decimalPattern.test(r.actualInputTokens) &&
      decimalPattern.test(r.actualOutputTokens)
    )
      issue(
        ctx,
        cost(r.actualInputTokens, r.actualOutputTokens) !==
          r.actualCostMicrousd,
      );
    issue(
      ctx,
      r.dispatchAllowed &&
        (r.state !== "dispatched" || r.replayed || r.poolBlocked),
    );
  });
const budgetEnvelope = z.discriminatedUnion("status", [
  z.strictObject({
    schemaVersion: z.literal("ai-budget/1.0.0"),
    status: z.literal("ok"),
    receipt: budgetReceipt,
  }),
  z.strictObject({
    schemaVersion: z.literal("ai-budget/1.0.0"),
    status: z.literal("denied"),
    code: z.enum([
      "AI_BUDGET_INVALID",
      "AI_BUDGET_UNAVAILABLE",
      "AI_BUDGET_DISABLED",
      "AI_BUDGET_DENIED",
      "AI_BUDGET_EXHAUSTED",
      "AI_BUDGET_CONFLICT",
      "AI_BUDGET_STATE",
      "AI_BUDGET_BLOCKED",
    ]),
  }),
]);
const claim = z
  .strictObject({
    schemaVersion: z.literal("initial-feedback-claim/1.0.0"),
    operation,
    attempt,
    dispatchAllowed: z.boolean(),
    replayed: z.boolean(),
    budget: budgetEnvelope.nullable(),
  })
  .superRefine((v, ctx) => {
    issue(
      ctx,
      !v.operation.attempts.some((a) => isDeepStrictEqual(a, v.attempt)),
    );
    const p = v.operation.binding.policy;
    if (p.budgetPolicyVersion === null) issue(ctx, v.budget !== null);
    else if (v.dispatchAllowed)
      issue(ctx, v.budget === null || v.budget.status !== "ok");
    if (v.dispatchAllowed)
      issue(
        ctx,
        v.replayed ||
          v.operation.state !== "open" ||
          v.operation.publicationId !== null ||
          v.attempt.state !== "dispatched",
      );
    if (v.budget?.status === "denied") issue(ctx, v.dispatchAllowed);
    if (v.budget?.status === "ok") {
      const r = v.budget.receipt;
      issue(
        ctx,
        r.userId !== v.operation.ownerId ||
          r.organizationId !== v.operation.organizationId ||
          r.operationId !== v.operation.operationId ||
          r.attemptId !== v.attempt.budgetAttemptId ||
          r.surface !== "qualitative" ||
          r.payloadSha256 !== v.operation.bindingSha256 ||
          r.model !== p.model ||
          r.promptPolicyVersion !== p.promptVersion ||
          r.policyVersion !== p.budgetPolicyVersion ||
          r.dispatchAllowed !== v.dispatchAllowed,
      );
    }
  });

export function validateInitialFeedbackReadiness(
  raw: unknown,
  expected: InitialFeedbackProfile,
  capabilities: ReadonlyArray<(typeof INITIAL_FEEDBACK_CAPABILITIES)[number]>,
) {
  const pin = decoded(InitialFeedbackProfileSchema, expected),
    value = decoded(InitialFeedbackReadinessSchema, raw);
  if (
    !value.schemaReady ||
    Object.keys(pin).some(
      (k) => pin[k as keyof typeof pin] !== value[k as keyof typeof pin],
    ) ||
    !z
      .array(z.enum(INITIAL_FEEDBACK_CAPABILITIES))
      .max(7)
      .safeParse(capabilities).success ||
    capabilities.some((k) => !value[k])
  )
    throw bad();
  return value;
}
const usageReply = z
  .strictObject({
    schemaVersion: z.literal("initial-feedback-usage/1.0.0"),
    profileVersion: z.literal("initial-feedback-readiness-profile/1.0.0"),
    usageSha256: sha,
    ownerId: id,
    organizationId: id,
    analysisId: id,
    periodStart: period,
    periodEnd: period,
    legacyUnlinkedTokens: decimal,
    canonicalTokens: decimal,
    initialFeedbackTokens: decimal,
    totalTokens: decimal,
    unknownClaims: decimal,
  })
  .superRefine((v, ctx) => {
    if (
      [
        v.legacyUnlinkedTokens,
        v.canonicalTokens,
        v.initialFeedbackTokens,
        v.totalTokens,
      ].every((x) => decimalPattern.test(x))
    )
      issue(
        ctx,
        BigInt(v.legacyUnlinkedTokens) +
          BigInt(v.canonicalTokens) +
          BigInt(v.initialFeedbackTokens) !==
          BigInt(v.totalTokens),
      );
    const y = Number(v.periodStart.slice(0, 4)),
      m = Number(v.periodStart.slice(5, 7));
    issue(
      ctx,
      v.periodEnd !==
        `${String(y + (m === 12 ? 1 : 0)).padStart(4, "0")}-${String(m === 12 ? 1 : m + 1).padStart(2, "0")}-01`,
    );
  });
export const InitialFeedbackUsageReplySchema = wire(usageReply);
export type InitialFeedbackUsageReply = z.infer<
  typeof InitialFeedbackUsageReplySchema
>;
export function validateInitialFeedbackUsage(
  raw: unknown,
  expected: {
    ownerId: string;
    organizationId: string;
    analysisId: string;
    usageSha256: string;
  },
  at: number,
) {
  const value = decoded(InitialFeedbackUsageReplySchema, raw),
    b = decoded(
      z.strictObject({
        ownerId: id,
        organizationId: id,
        analysisId: id,
        usageSha256: sha,
      }),
      expected,
    ),
    date = new Date(at);
  if (
    !Number.isFinite(at) ||
    !Number.isFinite(date.getTime()) ||
    value.ownerId !== b.ownerId ||
    value.organizationId !== b.organizationId ||
    value.analysisId !== b.analysisId ||
    value.usageSha256 !== b.usageSha256 ||
    value.periodStart !==
      `${String(date.getUTCFullYear()).padStart(4, "0")}-${String(date.getUTCMonth() + 1).padStart(2, "0")}-01` ||
    value.unknownClaims !== "0"
  )
    throw bad();
  return value;
}
