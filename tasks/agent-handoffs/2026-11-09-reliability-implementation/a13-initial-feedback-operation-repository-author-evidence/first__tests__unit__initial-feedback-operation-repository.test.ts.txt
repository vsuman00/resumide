function claiming(f: ReturnType<typeof fixture>): InitialFeedbackClaim {
  const attempt = {
    ...f.prepared.operation.attempts[0],
    state: "dispatched",
    claimedAt: "2026-11-15T00:00:00.123456Z",
  };
  return InitialFeedbackClaimSchema.parse({
    schemaVersion: "initial-feedback-claim/1.0.0",
    operation: { ...f.prepared.operation, attempts: [attempt] },
    attempt,
    dispatchAllowed: true,
    replayed: false,
    budget: null,
  });
}
function settlement(
  c: InitialFeedbackClaim,
  usage: typeof known | null = known,
  errorCode: "timeout" | null = null,
) {
  return {
    schemaVersion: "initial-feedback-settlement/1.0.0",
    operationId: c.operation.operationId,
    attemptId: c.attempt.attemptId,
    bindingSha256: c.operation.bindingSha256,
    state:
      usage === null ? "held_unknown" : errorCode === null ? "known" : "failed",
    usage,
    errorCode,
    replayed: false,
  };
}
function harness(
  f = fixture(),
  options: { timeoutMs?: number; now?: () => number } = {},
) {
  const data = {
    metadata: ready(),
    prepare: structuredClone(f.prepared),
    claim: claiming(f),
    settle: settlement(claiming(f)),
  };
  const rpc = vi.fn<InitialFeedbackOperationClient["rpc"]>(
    async (name: string) => {
      const value =
        name === "read_initial_feedback_operation_readiness"
          ? data.metadata
          : name === "prepare_initial_feedback"
            ? data.prepare
            : name === "claim_initial_feedback_dispatch"
              ? data.claim
              : name === "settle_initial_feedback_attempt"
                ? data.settle
                : null;
      if (value === null) throw Error("Unexpected RPC");
      return { data: structuredClone(value), error: null };
    },
  );
  const repo = createInitialFeedbackOperationRepository(
    { rpc },
    f.scope,
    options,
  );
  return { f, data, rpc, repo };
}
async function winner(h: ReturnType<typeof harness>) {
  const p = await h.repo.prepare();
  const c = await h.repo.claim(p);
  expect(c.status).toBe("won");
  if (c.status !== "won") throw Error("Winning fixture unavailable");
  return c;
}

      reservedInputTokens: 20,
      reservedOutputTokens: 1200,
    };
    const budget: InitialFeedbackBudgetClaim = {
      p_user_id: b.userId,
      p_organization_id: b.organizationId,
      p_request: {
        ...b,
        reservedInputTokens: "20",
        reservedOutputTokens: "1200",
      },
    };
    const a = {
      ...f.prepared.operation.attempts[0],
      state: "dispatched",
      claimedAt: "2026-11-15T00:00:00Z",
      budgetAttemptId: b.attemptId,
    };
    const receipt = {
      schemaVersion: "ai-budget/1.0.0",
      ...b,
      reservedInputTokens: "20",
      reservedOutputTokens: "1200",
      bindingSha256: hashAiBudgetBinding(b),
      poolPolicyVersion: "unit-pool/1",
      attemptLimit: 3,
      period: "2026-11-01",
      state: "dispatched",
      inputMicrousdPerMillion: "1",
      outputMicrousdPerMillion: "1",
      reservedCostMicrousd: calculateAiBudgetCost({
        inputTokens: 20,
        outputTokens: 1200,
        inputMicrousdPerMillion: "1",
        outputMicrousdPerMillion: "1",
      }),
      actualInputTokens: null,
      actualOutputTokens: null,
      actualCostMicrousd: null,
      replayed: false,
      dispatchAllowed: true,
      poolBlocked: false,
    };
    const claim = InitialFeedbackClaimSchema.parse({
      schemaVersion: "initial-feedback-claim/1.0.0",
      operation: { ...f.prepared.operation, attempts: [a] },
      attempt: a,
      dispatchAllowed: true,
      replayed: false,
      budget: { schemaVersion: "ai-budget/1.0.0", status: "ok", receipt },
    });
    h.rpc.mockImplementation(async (name) => ({
      data:
        name === "read_initial_feedback_operation_readiness"
          ? ready()
          : name === "prepare_initial_feedback"
            ? f.prepared
            : claim,
      error: null,
    }));
    const repo = createInitialFeedbackOperationRepository(
      { rpc: h.rpc },
      f.scope,
    );
    const prepared = await repo.prepare();
    const won = await repo.claim(prepared, budget);
    expect(won.status).toBe("won");
    expect(won.claim.budget).toEqual(claim.budget);
    expect(h.rpc.mock.calls.at(-1)?.[1].p_budget).toEqual(budget);
    expect(f.scope.lease.attempt).toBe(2);
    expect(b.attemptOrdinal).toBe(1);
    expect(
      h.rpc.mock.calls.every(([name]) => !name.startsWith("ai_budget_")),
    ).toBe(true);
  });

    const h = harness();
    const won = await winner(h);
    const before = h.rpc.mock.calls.length;
    await expect(
      h.repo.settle(won.dispatch, {
        usage: { inputTokens: "01", outputTokens: "7" },
        errorCode: null,
      }),
    ).rejects.toMatchObject({ code: "invalid" });
    await expect(
      h.repo.claim(structuredClone(h.f.prepared)),
    ).rejects.toMatchObject({ code: "invalid" });
    expect(h.rpc).toHaveBeenCalledTimes(before);
  });
