function args(origin = "https://example.test", form = body()) {

const evidenceContext = {
  id: attemptId,
  ownerId,
  analysisId,
  sourceId: "10000000-0000-4000-8000-000000000005",
  inputSha256: "a".repeat(64),
  createdAt: "2026-11-13T00:00:00.000Z",
  retentionPolicyId: "synthetic-local-30d/1.0.0",
};
function reviewModel(
  pageTexts: string[],
  ocrFailure: DocumentEvidence["ocrFailure"] = null,
  ocrSource: unknown = null,
) {
  const evidence = reconcileDocumentEvidence(
    evidenceContext,
    buildNativeEvidence(pageTexts),
    ocrSource,
    ocrFailure,
  );
  return {
    analysisId,
    attemptId,
    attemptNumber: 1,
    evidence,
    corrections: [],
    revision: 0,
    scoringEligible: false,
    nativePageDiagnostics: deriveNativePageDiagnostics(evidence.nativeSource),
    targets: deriveEvidenceReviewTargets(evidence, {
      ownerId,
      analysisId,
      sourceId: evidenceContext.sourceId,
      inputSha256: evidenceContext.inputSha256,
      attemptId,
    }),
  };
}
async function renderReview(model: ReturnType<typeof reviewModel>) {
  mocks.load.mockResolvedValue(model);
  const handler = createStaticHandler([
    {
      id: "evidence",
      path: "/analysis/:id/evidence",
      Component: AnalysisEvidence,
      loader,
      action,
    },
  ]);
  const context = await handler.query(
    new Request(`https://example.test/analysis/${analysisId}/evidence`),
  );
  if (context instanceof Response)
    throw Error("Expected a rendered evidence response.");
  const router = createStaticRouter(handler.dataRoutes, context);
  const html = renderToString(
    createElement(StaticRouterProvider, { router, context, hydrate: false }),
  ).replaceAll("<!-- -->", "");
  return { html, context };
}

function recordedOcrWithText(nativeText: string, text: string) {
  const source = savedOcrSource();
  return {
    ...source,
    request: { ...source.request, nativePageTexts: [nativeText] },
    status: "review_required",
    pages: source.pages.map((page) => ({
      ...page,
      text,
      words: [...text.matchAll(/\S+/gu)].map((word) => ({
        text: word[0],
        start: Array.from(text.slice(0, word.index)).length,
        end:
          Array.from(text.slice(0, word.index)).length +
          Array.from(word[0]).length,
        box: { x: 0.1, y: 0.1, width: 0.1, height: 0.1 },
        confidence: {
          kind: "uncalibrated",
          scale: "tesseract_0_100",
          value: 80,
        },
      })),
    })),
  };
}
function entryReviewHtml(html: string) {
  return html.slice(
    html.indexOf(
      '<section aria-labelledby="entry-association-review-heading">',
    ),
    html.indexOf('<section aria-labelledby="original-evidence-heading">'),
  );
}

describe("owner-authenticated evidence route", () => {
  it("propagates private SSR headers and session refresh cookies on loader/action/error paths", () => {
    const output = new Headers(
      headers({
        parentHeaders: new Headers({
          "Cache-Control": "public, max-age=3600",
          "Set-Cookie": "root=session; HttpOnly",
          "Content-Security-Policy": "default-src 'self'",
        }),
        loaderHeaders: new Headers({
          "Set-Cookie": "loader=session; HttpOnly",
        }),
        actionHeaders: new Headers({
          "Set-Cookie": "action=session; HttpOnly",
        }),
        errorHeaders: new Headers({ "Set-Cookie": "error=session; HttpOnly" }),
      }),
    );
    expect(output.get("Cache-Control")).toBe("private, no-store");
    expect(output.get("X-Robots-Tag")).toBe("noindex, nofollow");
    expect(output.get("Content-Security-Policy")).toBe("default-src 'self'");
    expect(output.getSetCookie()).toEqual([
      "root=session; HttpOnly",
      "loader=session; HttpOnly",
      "action=session; HttpOnly",
      "error=session; HttpOnly",
    ]);
  });
  beforeEach(() => {
    vi.clearAllMocks();
    mocks.user.mockResolvedValue({ id: ownerId });
    mocks.correct.mockResolvedValue({
      status: "applied",
      scoringEligible: false,
    });
  });
  it("requires authentication before creating privileged clients", async () => {
    mocks.user.mockResolvedValue(null);
    const response = await action(args());
    expect(response.init?.status).toBe(401);
    expect(mocks.admin).not.toHaveBeenCalled();
    await expect(
      loader({
        request: new Request("https://example.test/evidence"),
        params: { id: analysisId },
      }),
    ).rejects.toMatchObject({ status: 302 });
  });
  it("rejects cross-origin browser writes before authentication or RPC", async () => {
    const response = await action(args("https://attacker.test"));
    expect(response.init?.status).toBe(403);
    expect(mocks.user).not.toHaveBeenCalled();
    expect(mocks.correct).not.toHaveBeenCalled();
  });
  it("does not accept identity fields or malformed duplicate submissions", async () => {
    const malicious = body();
    malicious.set("userId", ownerId);
    expect(
      (await action(args("https://example.test", malicious))).init?.status,
    ).toBe(422);
    const duplicated = body();
    duplicated.append("expectedRevision", "1");
    expect(
      (await action(args("https://example.test", duplicated))).init?.status,

    expect(mocks.correct).not.toHaveBeenCalled();

  it("uses authenticated identity, hides caching and preserves no-score feedback", async () => {
    const response = await action(args());
    expect(mocks.correct).toHaveBeenCalledWith(
      undefined,

        operation: "correct",

        candidateId: null,

      expect.any(String),

    expect(new Headers(response.init?.headers).get("Cache-Control")).toBe(
      "private, no-store",
    );
  });
  it.each([404, 409, 422] as const)(
    "returns safe correction status %i",
    async (status) => {
      mocks.correct.mockRejectedValue(
        new EvidenceReviewError(status, "Safe review error."),

      const response = await action(args());
      expect(response.init?.status).toBe(status);
      expect(response.data.error).toBe("Safe review error.");
    },

  it("does not leak provider/database errors", async () => {
    mocks.correct.mockRejectedValue(
      new Error("private raw text and database diagnostic"),

    const response = await action(args());
    expect(response.init?.status).toBe(503);
    expect(JSON.stringify(response.data)).not.toContain("private raw text");

  it("renders bounded native-page reasons separately from immutable escaped source", async () => {
    const repeated =
      "Employer: Fictional A\nEmployer: Fictional B\nNATIVE_SENTINEL <script>synthetic_marker</script>\uFFFD\u0001\uD800";
    const model = reviewModel([repeated, repeated, "", "tiny"]);
    const before = structuredClone(model.evidence);
    const { html, context } = await renderReview(model);
    expect(html).toContain("Page extraction review");
    const diagnostics = html.slice(
      html.indexOf('aria-labelledby="page-extraction-review-heading"'),
      html.indexOf('aria-labelledby="original-evidence-heading"'),
    );
    for (const label of [
      "No native text was extracted.",
      "Limited native text was extracted.",
      "Extracted native text contains a replacement character.",
      "Extracted text contains an invalid character sequence. Compare this page with your PDF.",
      "Extracted native text contains an unexpected control character.",
      "Identical native text was extracted from another page.",
      "Matching extracted text on pages: 2.",
      "Native text detected.",
      "Little or no native text detected.",
      "No OCR evidence recorded for this page.",
      "The saved extraction recorded conflicting candidate values. Review their source and entry relationships.",
      "OCR evidence is missing for pages with little or no native text.",
    ])
      expect(diagnostics).toContain(label);
    const detectionDetails = [
      ...diagnostics.matchAll(/<details>([\s\S]*?)<\/details>/gu),
    ].map((match) => match[1]);
    expect(detectionDetails).toHaveLength(4);
    for (const details of detectionDetails) {
      expect(details).toContain("<summary>Text detection details</summary>");
      expect(details).toContain(
        "Native text is treated as available at 8 or more non-whitespace UTF-16 units. This count does not assess reliability or accuracy.",
      );
    }
    expect(detectionDetails[0]).toContain(
      `Native text length: ${model.nativePageDiagnostics.pages[0].nonWhitespaceUtf16Units} non-whitespace UTF-16 units.`,
    );
    const mainDiagnostics = diagnostics.replace(
      /<details>[\s\S]*?<\/details>/gu,
      "",
    );
    expect(mainDiagnostics).not.toContain("UTF-16");
    expect(mainDiagnostics).not.toContain("threshold");
    expect(diagnostics).not.toContain("NATIVE_SENTINEL");
    expect(diagnostics).not.toContain("synthetic_marker");
    expect(html).toContain("&lt;script&gt;synthetic_marker&lt;/script&gt;");
    expect(html).not.toContain("<script>synthetic_marker</script>");
    expect(html).toContain('name="attemptId" value="' + attemptId + '"');
    expect(html).toContain('name="expectedRevision" value="0"');
    expect(html).toContain("Save employer correction");
    expect(context.loaderData.evidence.scoringEligible).toBe(false);
    expect(model.evidence).toEqual(before);
  });

    const model = reviewModel([""], null, savedOcrSource());
    const { html } = await renderReview(model);
    expect(html).toContain("OCR evidence recorded for this page.");

    expect(html).toContain("Little or no native text detected.");

    expect(html).toContain("No extracted text was recorded.");
    expect(model.evidence.ocrSource?.status).toBe("insufficient");
    expect(model.evidence.scoringEligible).toBe(false);
  });
  it("shows possible employment entries through the real owned evidence renderer", async () => {
    const model = reviewModel([
      "Experience\nEmployer: Fictional Alpha\nRole: Engineer\nDates: 2020-2024\nEmployer: Fictional Beta\nRole: Analyst\nDates: 2022-2025",
    ]);
    expect(model.targets.observations).toHaveLength(6);
    const before = structuredClone(model);
    const { html, context } = await renderReview(model);
    const associations: AssociationReview =
      context.loaderData.evidence.entryAssociations;
    const entries = entryReviewHtml(html);
    expect(html).toContain("Possible employment entries");
    expect(associations.state).toBe("review_required");
    expect(associations.groups).toHaveLength(2);
    expect(associations.persistable).toBe(false);
    expect(associations.scoringEligible).toBe(false);
    expect(associations.qualificationAssessment).toBe("not_assessed");
    expect(
      entries.match(/aria-label="Possible employment entry"/gu),
    ).toHaveLength(2);
    for (const group of associations.groups) {
      expect(entries).toContain(`id="entry-${group.id}"`);
      for (const member of group.members) {
        const observation = model.targets.observations.find(
          (item) => item.id === member.observationId,
        );
        expect(observation?.targetId).toBe(member.targetId);
        expect(observation?.citations).toEqual(member.citations);
        expect(entries).toContain(observation?.value);
        for (const citation of member.citations)
          expect(entries).toContain(
            `#source-${encodeURIComponent(citation.spanId)}`,
          );
      }
    }
    expect(entries).toContain("Unverified entry link");
    expect(entries).toContain(
      "They do not change your structured resume or confirm work history.",
    );
    expect(entries).not.toContain("<form");
    expect(entries).not.toMatch(
      /years of experience|duration credit|evidence-entry-associations\/1\.0\.0/iu,
    );
    expect(html).toContain("Recorded aggregate state: conflict");
    expect(html).toContain(
      "Different employers, roles or dates across separate entries do not by themselves establish a contradiction.",

    expect(model).toEqual(before);

  it("preserves existing same-span candidate identities without duplicating their members", async () => {

      "Employer: Fictional Alpha | Role: Engineer | Dates: 2020-2024",
    ]);
    expect(model.targets.groups).toHaveLength(1);
    const { html, context } = await renderReview(model);
    const associations: AssociationReview =
      context.loaderData.evidence.entryAssociations;
    const entries = entryReviewHtml(html);
    expect(associations.groups).toHaveLength(0);
    expect(associations.unresolved).toHaveLength(0);
    expect(associations.preservedGroupIds).toEqual(
      model.targets.groups.map((group) => group.id),

    expect(entries).toContain(`id="entry-${model.targets.groups[0].id}"`);
    expect(
      entries.match(/aria-label="Possible employment entry"/gu),

    expect(entries.match(/<dd>Fictional Alpha/gu)).toHaveLength(1);

      context.loaderData.evidence.entryAssociations;

  it.each(["target", "citation"])(

      if (kind === "target")
        model.targets.observations[0].targetId =
          model.targets.observations[1].targetId;
      else model.targets.observations[0].citations[0].start += 1;
      const { html } = await renderReview(model);
      const entries = entryReviewHtml(html);
      expect(entries).toContain(
        "This suggested entry could not be matched to the current source.",

      expect(entries).not.toContain("Fictional Alpha");
      expect(html).toContain("Fictional Alpha");

      "Experience\nEmployer: Fictional Alpha\nRole: Engineer",

    model.targets.binding.inputSha256 = "c".repeat(64);

  it("renders exact escaped observation values without copying raw source passages into association data", async () => {
