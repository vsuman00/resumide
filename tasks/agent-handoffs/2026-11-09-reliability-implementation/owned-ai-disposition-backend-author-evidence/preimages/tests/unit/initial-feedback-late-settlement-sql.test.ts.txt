function bodyIn(text: string, name: string) {

  const matches = [
    ...text.matchAll(
      new RegExp(
        `create(?: or replace)? function ${escaped}\\([\\s\\S]*?\\)\\s*returns [\\s\\S]*?\\bas\\s+(\\$[A-Za-z0-9_]*\\$)([\\s\\S]*?)\\1;`,
        "gu",
      ),
    ),
  ];
  return matches.at(-1)?.[2];
}
