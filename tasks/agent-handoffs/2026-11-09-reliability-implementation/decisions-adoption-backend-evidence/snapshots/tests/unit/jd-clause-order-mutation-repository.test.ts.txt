type Result = { data: unknown; error: unknown };
const ok = (data: unknown): Result => ({ data, error: null });
const ready = () => ({
  ...JD_CLAUSE_ORDER_READINESS_PROFILE,
  schemaReady: true,
  writerExecutable: true,
});
type Call = {
  name: string;
  args: Record<string, unknown>;
  signal?: AbortSignal;
};
function fake(
  context: RequirementRelationshipContext,
  override?: (
    call: Call,
    calls: Call[],
  ) => Result | Promise<Result> | undefined,
) {
  const calls: Call[] = [],
    signals: AbortSignal[] = [];
  const rpc = vi.fn((name: string, args: Record<string, unknown>) => {
    const call: Call = { name, args };
    calls.push(call);
    const request = {
      abortSignal(signal: AbortSignal) {
        call.signal = signal;
        signals.push(signal);
        return request;
      },
      then(
        resolve: (result: Result) => unknown,
        reject?: (error: unknown) => unknown,
      ) {
        return Promise.resolve()
          .then(async () => {
            const changed = override?.(call, calls);
            if (changed !== undefined) return changed;
            if (name === "read_requirement_relationship_context")
              return ok(context);
            if (name === "read_jd_clause_order_readiness") return ok(ready());
            if (name === "apply_jd_clause_order_action") {
              const c = parseClauseOrderCommand(args.p_command);
              return ok({
                schemaVersion: "jd-clause-order-receipt/2.0.0",
                status: "applied",
                action: c.operation,
                idempotencyKey: c.idempotencyKey,
                commandSha256: clauseOrderCommandSha256(c),
                draftId: c.draftId,
                baseVersionId: c.binding.canonical.versionId,
                baseRevision: c.binding.canonical.revision,
                requirementVersionId: c.binding.requirements.versionId,
                requirementRevision: c.binding.requirements.revision,
                versionId:
                  c.operation === "accept"
                    ? id(990)
                    : c.binding.canonical.versionId,
                revision:
                  c.binding.canonical.revision +
                  (c.operation === "accept" ? 2 : 0),
                eventId: id(991),
                idempotent: false,
              });
            }
            throw Error("UNEXPECTED_SYNTHETIC_RPC");
          })
          .then(resolve, reject);
      },
    };
    return request;
  });
  // Synthetic SDK boundary only; production uses the supplied actual Supabase client.
  return {
    client: { rpc } as unknown as Parameters<
      typeof loadJdClauseOrderWorkspace
    >[0],
    rpc,
    calls,
    signals,
  };
}

function moved(context: RequirementRelationshipContext) {
  const next = structuredClone(context),
    canonical = reviseCanonical(context.canonical.content, {
      operation: "rename_section",
      sectionId: context.canonical.content.sections[0].id,
      value: "Reviewed experience",
    }),
    jd = createJobRequirementRelations("Requirements\nGit preferred", {
      ownerId: context.source.ownerId,
      organizationId: context.source.organizationId,
      analysisId: context.source.analysisId,
      jobId: context.requirements.content.jobId,
      parentVersionId: context.requirements.versionId,
      locale: "en-IN",
      domain: "software-engineering",
    });
  next.heads = {
    canonicalHeadId: canonical.versionId,
    canonicalHeadRevision: context.canonical.revision + 1,
    requirementHeadId: jd.id,
    requirementHeadRevision: context.requirements.revision + 1,
  };
  return next;
}

it.each([
  "not-ready",
  "rpc-error",
  "throw",
  "conflict",
  "malformed",
  "tampered",
  "extra-conflict",
])("does not regenerate or retry an action after %s", async (kind) => {
  const { f, full, expected } = await fixture(),
    command = buildClauseOrderProposal(f.context, expected, full).acceptCommand;
  const store = fake(f.context, (c) => {
    if (kind === "not-ready" && c.name === "read_jd_clause_order_readiness")
      return ok({ ...ready(), writerExecutable: false });
    if (c.name !== "apply_jd_clause_order_action") return undefined;
    if (kind === "rpc-error")
      return { data: null, error: { message: "SYNTHETIC_PRIVATE_ERROR" } };
    if (kind === "throw") throw Error("SYNTHETIC_PRIVATE_ERROR");
    if (kind === "conflict") return ok({ status: "conflict" });
    if (kind === "extra-conflict")
      return ok({ status: "conflict", extra: true });
    if (kind === "tampered")
      return ok({
        schemaVersion: "jd-clause-order-receipt/2.0.0",
        status: "applied",
        commandSha256: "f".repeat(64),
      });
    return ok({ status: "unknown" });
  });
  await expect(
    applyJdClauseOrderAction(
      store.client,
      f.owned.analysisId,
      f.owned.ownerId,
      command,
      command.idempotencyKey,
    ),
  ).rejects.toMatchObject({
    reason:
      kind === "conflict"
        ? "stale_inputs"
        : ["malformed", "tampered", "extra-conflict"].includes(kind)
          ? "invalid_result"
          : "unavailable",
  });
  expect(
    store.calls.filter((c) => c.name === "apply_jd_clause_order_action"),
  ).toHaveLength(kind === "not-ready" ? 0 : 1);
});
