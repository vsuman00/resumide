function formRequest(form: URLSearchParams | FormData, signal?: AbortSignal) {
  return new Request("https://example.test/analysis/" + id(705) + "/rewrite", {
    method: "POST",
    body: form,
    signal,
    headers: { Origin: "https://example.test" },
  });
}
function encoded(value = JSON.stringify(requestValue)) {
  return new URLSearchParams({ request: value, aiConsent: "accepted" });
}

describe("canonical rewrite actual bounded form parser", () => {
  it.each(["urlencoded", "multipart"])(
    "preserves exact valid request and microseconds from %s",
    async (kind) => {
      const form = kind === "urlencoded" ? encoded() : new FormData();
      if (form instanceof FormData) {
        form.set("request", JSON.stringify(requestValue));
        form.set("aiConsent", "accepted");
      }
      expect(await readCanonicalRewriteForm(formRequest(form))).toEqual(
        requestValue,
      );
    },
  );
  it.each([
    "duplicate form",
    "escaped duplicate JSON",
    "missing",
    "extra",
    "file",
  ])("refuses %s without service IO", async (kind) => {
    let form: URLSearchParams | FormData = encoded();
    if (kind === "duplicate form")
      form.append("request", JSON.stringify(requestValue));
    if (kind === "escaped duplicate JSON")
      form.set(
        "request",
        JSON.stringify(requestValue).replace(
          '"operationId":',
          '"\\u006fperationId":"' +
            requestValue.operationId +
            '","operationId":',
        ),
      );
    if (kind === "missing") form.delete("request");
    if (kind === "extra") form.set("extra", "forbidden");
    if (kind === "file") {
      form = new FormData();
      form.set(
        "request",
        new Blob([JSON.stringify(requestValue)]),
        "request.json",
      );
      form.set("aiConsent", "accepted");
    }
    await expect(
      readCanonicalRewriteForm(formRequest(form)),
    ).rejects.toMatchObject({ status: 422 });
    expect(mocks.admin).not.toHaveBeenCalled();
    expect(mocks.user).not.toHaveBeenCalled();
  });
  it("requires the exact native consent acknowledgement", async () => {
    const form = encoded();
    form.set("aiConsent", "true");
    await expect(
      readCanonicalRewriteForm(formRequest(form)),
    ).rejects.toMatchObject({ status: 403 });
  });
  it("rejects malformed UTF8 and both encoded and decoded byte overflows", async () => {
    const req = (body: Uint8Array<ArrayBuffer> | string) =>
      new Request("https://example.test/rewrite", {
        method: "POST",
        headers: { "Content-Type": "application/x-www-form-urlencoded" },
        body,
      });
    await expect(
      readCanonicalRewriteForm(req(new Uint8Array([0xc3, 0x28]))),
    ).rejects.toMatchObject({ status: 422 });
    await expect(
      readCanonicalRewriteForm(req("x".repeat(65537))),
    ).rejects.toMatchObject({ status: 413 });
    const raw = JSON.stringify(requestValue),
      padded = " ".repeat(16385 - raw.length) + raw;
    await expect(
      readCanonicalRewriteForm(formRequest(encoded(padded))),
    ).rejects.toMatchObject({ status: 422 });
  });
  it.each(["resolve", "reject"])(
    "settles stalled body deadline and releases reader when cancel %s",
    async (kind) => {
      vi.useFakeTimers();
      const cancel = vi.fn(() =>
        kind === "reject"
          ? Promise.reject(new Error("SYNTHETIC_CANCEL_FAILURE"))
          : undefined,
      );
      const stream = new ReadableStream<Uint8Array>({
        pull: () => new Promise<void>(() => {}),
        cancel,
      });
      const request = new Request("https://example.test/rewrite", {
        method: "POST",
        headers: { "Content-Type": "application/x-www-form-urlencoded" },
        body: stream,
        duplex: "half",
      } as RequestInit & { duplex: "half" });
      const pending = readCanonicalRewriteForm(request, 5).catch(
        (error) => error,
      );
      await vi.advanceTimersByTimeAsync(6);
      expect(await pending).toMatchObject({
        reason: "unavailable",
        recovery: "retry_exact",
      });
      expect(cancel).toHaveBeenCalledTimes(1);
      expect(stream.locked).toBe(false);
    },
  );
  it("settles caller cancellation during a pending read and releases its reader", async () => {
    const controller = new AbortController(),
      cancel = vi.fn();
    let enterRead = () => {};
    const reading = new Promise<void>((resolve) => {
      enterRead = resolve;
    });
    const stream = new ReadableStream<Uint8Array>(
      {
        pull: () => {
          enterRead();
          return new Promise<void>(() => {});
        },
        cancel,
      },
      { highWaterMark: 0 },
    );
    const request = new Request("https://example.test/rewrite", {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      signal: controller.signal,
      body: stream,
      duplex: "half",
    } as RequestInit & { duplex: "half" });
    const pending = readCanonicalRewriteForm(request).catch((error) => error);
    await reading;
    expect(stream.locked).toBe(true);
    controller.abort();
    expect(await pending).toMatchObject({
      reason: "unavailable",
      recovery: "retry_exact",
    });
    expect(cancel).toHaveBeenCalledTimes(1);
    expect(stream.locked).toBe(false);
  });
});

async function handlerFixture(
  usageCounter: unknown,
  budgetDenial?: string,
  recorded = { canonicalTokens: "0", unknownClaims: "0" },
) {
  const f = await createRequirementContextFixture({
    edit: "Built database services using PostgreSQL",
    confirm: true,
  });
  const clauseId = f.context.requirements.content.relations.clauses[0].id;
  const patch = buildTypedAuthoringPatch(
    f.context,
    {
      canonicalVersionId: f.context.canonical.versionId,
      requirementVersionId: f.context.requirements.versionId,
      clauseId,
      targetFactId: f.targetFactId,
    },
    "Created",
  );
  const fact = f.context.canonical.content.facts.find(
    (value) => value.id === f.targetFactId,
  )!;
  const row = {
    schemaVersion: "canonical-rewrite-context/1.0.0",
    status: "available",
    relationshipContext: f.context,
    draft: {
      id: id(711),
      updatedAt: "2026-11-15T00:00:00.000001Z",
      contentSha256: "1".repeat(64),
      versionId: null,
      revision: 0,
      headId: null,
    },
    bindingSha256: "2".repeat(64),
    manifest: [
      {
        sourceId: 0,
        factId: fact.id,
        valueSha256: patch.target.currentValueSha256,
        location: {
          sectionId: patch.target.sectionId,
          entryId: patch.target.entryId,
          bulletId: patch.target.bulletId,
        },
        confirmation: fact.confirmation,
        basis: patch.support.basis,
        currentCitations: patch.sourceCitations,
        clauseIds: [clauseId],
        authoringVersion: patch.policies.authoringVersion,
      },
    ],
    saved: {
      state: "legacy_unbound",
      publication: null,
      legacyDraft: { summary: null, bullets: [] },
    },
    readiness: { ...profile, schemaReady: true, mutatorsExecutable: true },
  };
  const workspace = buildCanonicalRewriteWorkspace(row, "unit-model", profile),
    request = createCanonicalRewriteRequest(workspace, id(712));
  let operation: CanonicalRewriteOperation | undefined;
  const control = {
    usagePatch: {} as Record<string, unknown>,
    readinessPatch: {} as Record<string, unknown>,
  };
  const client = {
    rpc: vi.fn(async (name: string, args: Record<string, unknown>) => {
      if (name === "prepare_canonical_rewrite") {
        operation = {
          schemaVersion: "canonical-rewrite-operation/1.0.0",
          operationId: request.operationId,
          ownerId: f.owned.ownerId,
          analysisId: f.owned.analysisId,
          request,
          policy: workspace.binding.policy,
          bindingSha256: workspace.bindingSha256,
          state: "reserved",
          dispatchAllowed: false,
          publication: null,
        };
        return {
          data: {
            status: "prepared",
            operation,
            context: {
              ...row,
              readiness: { ...row.readiness, ...control.readinessPatch },
            },
          },
          error: null,
        };
      }
      if (name === "ai_budget_reserve_next" && budgetDenial)
        return {
          data: {
            schemaVersion: "ai-budget/1.0.0",
            status: "denied",
            code: budgetDenial,
          },
          error: null,
        };
      if (name === "consume_analysis_rate_limit")
        return { data: true, error: null };
      if (name === "get_organization_ai_tokens_used")
        return { data: usageCounter, error: null };
      if (name === "read_canonical_rewrite_usage") {
        // This fake row uses the actual released successor pin when queried.
        // The before-fix caller never queries it: it sees legacy counter zero.
        const usageSha256 = Reflect.get(profile, "usageSha256");
        if (typeof usageSha256 !== "string")
          throw Error("EXPECTED_RELEASED_USAGE_PROFILE");
        if (
          typeof usageCounter !== "number" ||
          !Number.isSafeInteger(usageCounter) ||
          usageCounter < 0
        )
          return { data: usageCounter, error: null };
        const now = new Date(),
          year = now.getUTCFullYear(),
          month = now.getUTCMonth();
        const date = (v: Date) => v.toISOString().slice(0, 10);
        return {
          data: {
            schemaVersion: "canonical-rewrite-usage/1.0.0",
            profileVersion: "canonical-rewrite-readiness-profile/1.0.1",
            usageSha256,
            ownerId: f.owned.ownerId,
            organizationId: f.owned.organizationId,
            analysisId: f.owned.analysisId,
            periodStart: date(new Date(Date.UTC(year, month, 1))),
            periodEnd: date(new Date(Date.UTC(year, month + 1, 1))),
            legacyTokens: String(usageCounter),
            canonicalTokens: recorded.canonicalTokens,
            totalTokens: String(
              BigInt(usageCounter) + BigInt(recorded.canonicalTokens),
            ),
            unknownClaims: recorded.unknownClaims,
            ...control.usagePatch,
          },
          error: null,
        };
      }
      if (name === "claim_canonical_rewrite_dispatch") {
        operation = {
          ...operation!,
          state: "dispatched",
          dispatchAllowed: true,
        };
        return { data: { operation, budget: null }, error: null };
      }
      if (name === "settle_canonical_rewrite_attempt")
        return {
          data: {
            schemaVersion: "canonical-rewrite-settlement/1.0.0",
            operationId: request.operationId,
            bindingSha256: workspace.bindingSha256,
            usage: args.p_usage,
            state: args.p_error_code ? "rejected" : "dispatched",
          },
          error: null,
        };
      if (name === "read_canonical_rewrite_context")
        return { data: row, error: null };
      if (name === "finish_canonical_rewrite") {
        const result = reconstructCanonicalRewrite(
          workspace,
          request.operationId,
          args.p_plan,
        );
        return {
          data: {
            result,
            receipt: {
              schemaVersion: "canonical-rewrite-receipt/1.0.0",
              state: "saved_proposals",
              operationId: request.operationId,
              bindingSha256: workspace.bindingSha256,
              runId: id(713),
              draftId: request.draftId,
              draftVersionId: id(714),
              revision: 1,
              draftUpdatedAt: "2026-11-15T00:00:01.000002Z",
              resultSha256: "3".repeat(64),
              projectionSha256: "4".repeat(64),
              idempotent: false,
            },
          },
          error: null,
        };
      }
      throw new Error("UNEXPECTED_UNIT_RPC");
    }),
  };
  const transport = {
    execute: vi.fn(async () => ({
      output: {
        summarySourceId: null,
        bullets: [
          {
            sourceId: 0,
            clauseId,
            operation: "equivalent_action",
            replacementAction: "Created",
          },
        ],
      },
      usage: { inputTokens: 8, outputTokens: 3 },
    })),
  };
  mocks.user.mockResolvedValue({ id: f.owned.ownerId });
  mocks.server.mockReturnValue({});
  mocks.admin.mockReturnValue(client);
  mocks.consent.mockResolvedValue(undefined);
  const response = () =>
    canonicalRewriteAction(
      new Request(
        "https://example.test/analysis/" + f.owned.analysisId + "/rewrite",
        {
          method: "POST",
          headers: { Origin: "https://example.test" },
          body: new URLSearchParams({
            request: JSON.stringify(request),
            aiConsent: "accepted",
          }),
        },
      ),
      f.owned.analysisId,
      transport,
    );
  return { client, transport, response, control };
}
describe("actual action organization usage admission", () => {
  it.each([
    null,
    false,
    "",
    " ",
    "12",
    -1,
    1.5,
    Number.NaN,
    Number.POSITIVE_INFINITY,
  ])(
    "denies unknown or malformed counter %s before dispatch",
    async (counter) => {
      const f = await handlerFixture(counter),
        response = await f.response();
      expect(response.status).toBe(503);
      expect(f.transport.execute).not.toHaveBeenCalled();
      expect(
        f.client.rpc.mock.calls.some(
          (call) => call[0] === "claim_canonical_rewrite_dispatch",
        ),
      ).toBe(false);
    },
  );
  it.each([0, 12])(
    "retains legitimate nonnegative numeric counter %s",
    async (counter) => {
      const f = await handlerFixture(counter),
        response = await f.response();
      expect(response.status).toBe(200);
      expect((await response.json()).kind).toBe("saved");
      expect(f.transport.execute).toHaveBeenCalledTimes(1);
    },
  );
});

      ).toBe(false);
    },
  );
});
