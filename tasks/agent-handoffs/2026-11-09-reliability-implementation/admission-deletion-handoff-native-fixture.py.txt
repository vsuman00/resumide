migrations = Path("supabase/migrations")
fixtures = Path(__file__).parent

def source(prefix):
    file, = migrations.glob(prefix + "_*.sql")
    return file.read_text()

def section(text, start, end):
    start_at = text.index(start)
    return text[start_at:text.index(end, start_at)]

def literal(file, name):
    return next(
        ast.literal_eval(node.value)
        for node in ast.parse((fixtures / file).read_text()).body
        if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == name for t in node.targets)
    )

baseline = "--baseline" in sys.argv
prepare = "public.prepare_deletion_request" if baseline else "public.prepare_deletion_request_with_admission_handoff"
handoff = """
do $$
declare f fixture_adopted%rowtype; plan jsonb;
begin
  select * into strict f from fixture_adopted;
  begin
    perform PREPARE(f.request_id,'worker',1);
    raise exception 'Recorded worker did not hold deletion';
  exception when sqlstate 'P0A01' then null;end;
  assert exists(select 1 from public.resume_versions where id=f.version_id);
  assert not exists(select 1 from private.privacy_deletion_plans where request_id=f.request_id);
  delete from public.analysis_jobs where analysis_id=f.analysis_id;

  begin
    perform public.prepare_deletion_request(f.request_id,'worker',1);
    raise exception 'The legacy path bypassed the transaction handoff';
  exception when sqlstate 'P0A01' then null;end;
  assert exists(select 1 from public.resume_versions where id=f.version_id);

  begin
    plan:=PREPARE(f.request_id,'worker',1);
  exception when sqlstate 'P0A01' then
    raise exception 'Adopted source could not enter the durable deletion plan';
  end;
  assert plan#>>'{admissionHandoff,adoptedObjects,0,objectId}'=f.object_id::text;
  assert plan#>>'{admissionHandoff,adoptedObjects,0,storageKey}'=f.storage_key;
  assert plan#>>'{admissionHandoff,adoptedObjects,0,verified}'='false';
  assert not exists(select 1 from public.resume_versions where id=f.version_id);
  assert exists(select 1 from private.privacy_deletion_plans where request_id=f.request_id);
  assert exists(select 1 from private.owned_ai_deletion_dispositions where owner_id=f.owner_id);
  assert exists(select 1 from private.analysis_admission_deletion_handoffs where request_id=f.request_id and authorized_xid is null);
  assert (select status='completed' from public.privacy_request_steps where request_id=f.request_id and step_key='database');
  assert not public.complete_analysis_admission_deletion_object(f.request_id,'worker',1,f.upload_id,gen_random_uuid());
  assert public.complete_analysis_admission_deletion_object(f.request_id,'worker',1,f.upload_id,f.object_id);
end;
$$;
""".replace("PREPARE(",prepare+"(")

retry = """
do $$
declare f fixture_adopted%rowtype; plan jsonb;
begin
  select * into strict f from fixture_adopted;
  begin
    perform public.prepare_deletion_request(f.request_id,'worker',1);
    raise exception 'Legacy retry reused the new handoff';
  exception when sqlstate 'P0A01' then null;end;
  begin
    perform public.complete_privacy_request(f.request_id,'worker',1,'{"deleted":true}');
    raise exception 'Account completion passed while the original admission ledger remained';
  exception when sqlstate 'P0A01' then null;end;
  assert (select status='processing' from public.data_requests where id=f.request_id);
  assert exists(select 1 from private.privacy_deletion_plans where request_id=f.request_id);
  update public.data_requests set status='failed' where id=f.request_id;
  assert private.candidate_write_blocked(f.owner_id);
  update public.data_requests set status='processing', attempt_count=2 where id=f.request_id;
  assert not public.complete_analysis_admission_deletion_object(f.request_id,'worker',1,f.upload_id,f.object_id);
  plan:=public.prepare_deletion_request_with_admission_handoff(f.request_id,'worker',2);
  assert plan#>>'{admissionHandoff,adoptedObjects,0,verified}'='true';
  assert plan#>>'{admissionHandoff,attempt}'='2';
  assert private.analysis_admission_export_records(f.owner_id)#>>'{analysisAdmissionRetirementCaptures,0,deletionHandoff,firstAttempt}'='1';
  assert not has_table_privilege('service_role','private.analysis_admission_deletion_handoffs','SELECT,INSERT,UPDATE,DELETE');

$$;

rollback = """
do $$
declare f fixture_adopted%rowtype;
begin
  select * into strict f from fixture_adopted;
  delete from public.analysis_jobs where analysis_id=f.analysis_id;
  perform set_config('fixture.fail_after_plan','on',true);
  begin
    perform public.prepare_deletion_request_with_admission_handoff(f.request_id,'worker',1);
    raise exception 'Expected the injected failure';
  exception when raise_exception then
    assert sqlerrm='Synthetic failure after the plan was inserted';
  end;
  assert exists(select 1 from public.resume_versions where id=f.version_id);
  assert not exists(select 1 from private.privacy_deletion_plans where request_id=f.request_id);
  assert not exists(select 1 from private.analysis_admission_deletion_handoffs where request_id=f.request_id);
  assert not exists(select 1 from private.owned_ai_deletion_dispositions where owner_id=f.owner_id);
  insert into public.analysis_jobs values(f.analysis_id,'recorded-worker',clock_timestamp()-interval '1 hour');
end;
$$;
"""

with TemporaryDirectory(prefix="resumide-adopted-handoff-") as tmp:
    env = {"HOME": tmp, "PATH": "/opt/homebrew/bin:/usr/bin:/bin", "LC_ALL": "C"}
    data = str(Path(tmp) / "data")

    def run(argv, **options):
        return subprocess.run(argv, env=env, timeout=25, text=True, check=True, **options)

    assert run(["postgres","--version"],capture_output=True).stdout.startswith("postgres (PostgreSQL) 17.")
    run(["initdb","-D",data,"--no-sync","--auth=trust","-U","fixture_owner"],stdout=subprocess.DEVNULL)
    try:
        run(["pg_ctl","-w","-D",data,"-l",str(Path(tmp)/"postgres.log"),
             "-o",f"-c listen_addresses='' -c unix_socket_directories={tmp} -c unix_socket_permissions=0700","start"],
            stdout=subprocess.DEVNULL)
        run(["psql","-X","-v","ON_ERROR_STOP=1","-h",tmp,"-U","fixture_owner","-d","postgres","-Atq"],input=sql)
        print("PASS: bound handoff, transactional rollback, prepared-owner barrier, exact object receipt, retry, and continued completion hold.")
    finally:
        if (Path(data)/"postmaster.pid").exists():
            run(["pg_ctl","-w","-D",data,"-m","immediate","stop"],stdout=subprocess.DEVNULL)
