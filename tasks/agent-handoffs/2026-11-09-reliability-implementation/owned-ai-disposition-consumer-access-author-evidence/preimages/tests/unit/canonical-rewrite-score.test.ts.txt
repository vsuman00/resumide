type Result = { data: unknown; error: unknown };
type Read = {
  table: string;
  columns: string;
  filters: Record<string, unknown>;
  single: boolean;
};
const ok = (data: unknown): Result => ({ data, error: null });
function ownedStore(
  context: RequirementRelationshipContext,
  override?: (read: Read, sequence: Read[]) => Result | undefined,
) {
  const c = context.canonical,
    s = context.source,
    r = context.requirements;
  const docs = [c, ...context.ancestors];
  const reads: Read[] = [];
  const root = {
    id: s.resumeId,
    owner_id: s.ownerId,
    organization_id: s.organizationId,
    status: "active",
  };
  const analysis = {
    id: s.analysisId,
    owner_id: s.ownerId,
    organization_id: s.organizationId,
    resume_version_id: s.sourceId,
    job_id: r.content.jobId,
  };
  const source = {
    id: s.sourceId,
    resume_id: s.resumeId,
    owner_id: s.ownerId,
    organization_id: s.organizationId,
    active_resume: root,
    page_count: context.sourceMetadata.pageCount,
  };
  const job = {
    id: r.content.jobId,
    title: "Software Engineer",
    description: r.content.sourceText,
    owner_id: s.ownerId,
    organization_id: s.organizationId,
  };
  const canonicalRows = docs.map((d) => ({
    id: d.versionId,
    resume_id: s.resumeId,
    owner_id: s.ownerId,
    organization_id: s.organizationId,
    source_resume_version_id: s.sourceId,
    parent_version_id: d.content.parentVersionId,
    revision: d.revision,
    content: d.content,
    created_at: d.content.createdAt,
  }));
  const requirementRows = [
    {
      id: r.versionId,
      analysis_id: s.analysisId,
      job_id: r.content.jobId,
      owner_id: s.ownerId,
      organization_id: s.organizationId,
      parent_version_id: r.content.parentVersionId,
      revision: r.revision,
      content: r.content,
      created_at: r.content.createdAt,
    },
  ];
  const rpc = vi.fn<(name: string, args: unknown) => Promise<Result>>(
    async (name) => {
      if (name !== "read_requirement_relationship_context")
        throw Error("UNEXPECTED_WRITE_RPC");
      return ok(context);
    },
  );
  function valueAt(row: Record<string, unknown>, path: string): unknown {
    let value: unknown = row;
    for (const key of path.split(".")) {
      if (typeof value !== "object" || value === null) return undefined;
      value = (value as Record<string, unknown>)[key];
    }
    return value;
  }
  const from = vi.fn((table: string) => {
    const read: Read = { table, columns: "", filters: {}, single: false };
    const query = {
      select(columns: string) {
        read.columns = columns;
        return query;
      },
      eq(key: string, value: unknown) {
        read.filters[key] = value;
        return query;
      },
      neq() {
        return query;
      },
      order() {
        return query;
      },
      limit() {
        return query;
      },
      range() {
        return query;
      },
      maybeSingle() {
        read.single = true;
        return query;
      },
      then(yes: (result: Result) => unknown, no: (error: unknown) => unknown) {
        reads.push(read);
        const replacement = override?.(read, reads);
        if (replacement) return Promise.resolve(replacement).then(yes, no);
        let rows: Record<string, unknown>[];
        if (table === "analyses") rows = [analysis];
        else if (table === "resume_versions") rows = [source];
        else if (table === "memberships")
          rows = [
            {
              organization_id: s.organizationId,
              user_id: s.ownerId,
              role: "owner",
            },
          ];
        else if (table === "data_requests" || table === "authoring_events")
          rows = [];
        else if (table === "resumes") rows = [root];
        else if (table === "jobs") rows = [job];
        else if (table === "canonical_resume_heads")
          rows = [
            {
              resume_id: s.resumeId,
              owner_id: s.ownerId,
              organization_id: s.organizationId,
              version_id: context.heads.canonicalHeadId,
              revision: context.heads.canonicalHeadRevision,
            },
          ];
        else if (table === "canonical_resume_versions") rows = canonicalRows;
        else if (table === "job_requirement_versions") rows = requirementRows;
        else if (table === "document_evidence_attempts")
          rows = [
            {
              id: context.attempt.id,
              analysis_id: s.analysisId,
              owner_id: s.ownerId,
              organization_id: s.organizationId,
              envelope: context.attempt.envelope,
              resume_version_id: s.sourceId,
              input_sha256: s.inputSha256,
              attempt_number: context.attempt.attemptNumber,
            },
          ];
        else throw Error("UNEXPECTED_TABLE_READ");
        rows = rows.filter((row) =>
          Object.entries(read.filters).every(
            ([key, value]) => valueAt(row, key) === value,
          ),
        );
        if (
          table === "canonical_resume_versions" ||
          table === "job_requirement_versions"
        )
          rows = [...rows].sort(
            (a, b) => Number(b.revision) - Number(a.revision),
          );
        return Promise.resolve(ok(read.single ? (rows[0] ?? null) : rows)).then(
          yes,
          no,
        );
      },
    };
    return query;
  });
  return {
    client: { from, rpc } as unknown as Parameters<
      typeof loadScoreDiagnostics
    >[0],
    from,
    rpc,
    reads,
    canonicalRows,
    requirementRows,
    source,
    analysis,
    job,
  };
}

async function setup(declared = false) {
  const f = await createRequirementContextFixture({
    confirm: true,
    ...(declared ? { edit: "Built database services using PostgreSQL" } : {}),
  });
  const authoring = buildTypedAuthoringWorkspace(f.context);
  const patch = buildTypedAuthoringPatch(
    f.context,
    {
      canonicalVersionId: authoring.binding.canonical.versionId,
      requirementVersionId: authoring.binding.requirements.versionId,
      clauseId: authoring.clauses[0].clauseId,
      targetFactId: f.targetFactId,
    },
    "Created",
  );
  const fact = f.context.canonical.content.facts.find(
    (item) => item.id === f.targetFactId,
  );
  if (!fact?.confirmation) throw Error("EXPECTED_ACTUAL_CONFIRMATION");
  const initialRow = {
    schemaVersion: "canonical-rewrite-context/1.0.0",
    status: "available",
    relationshipContext: f.context,
    draft: {
      id: id(1),
      updatedAt: "2026-11-14T00:00:00.123456Z",
      contentSha256: "d".repeat(64),
      versionId: null,
      revision: 0,
      headId: null,
    },
    bindingSha256: "e".repeat(64),
    manifest: [
      {
        sourceId: 0,
        factId: f.targetFactId,
        valueSha256: patch.target.currentValueSha256,
        location: {
          sectionId: patch.target.sectionId,
          entryId: patch.target.entryId,
          bulletId: patch.target.bulletId,
        },
        confirmation: fact.confirmation,
        basis: patch.support.basis,
        currentCitations: patch.sourceCitations,
        clauseIds: [patch.selection.clauseId],
        authoringVersion: patch.policies.authoringVersion,
      },
    ],
    saved: {
      state: "legacy_unbound",
      publication: null,
      legacyDraft: { summary: "Preserved legacy summary", bullets: [] },
    },
    readiness: {
      ...CANONICAL_REWRITE_READINESS_PROFILE,
      schemaReady: true,
      mutatorsExecutable: true,
    },
  };
  let row: unknown = initialRow;
  const workspace = buildCanonicalRewriteWorkspace(
    row,
    "fixture-model",
    CANONICAL_REWRITE_READINESS_PROFILE,
  );
  const store = ownedStore(f.context);
  store.rpc.mockImplementation(async (name) =>
    name === "read_canonical_rewrite_workspace_state"
      ? ok(availableState(row))
      : name === "read_requirement_relationship_context"
        ? ok(f.context)
        : (() => {
            throw Error("UNEXPECTED_RPC");
          })(),
  );
  mocks.user.mockResolvedValue({ id: f.owned.ownerId });
  mocks.admin.mockReturnValue(store.client);
  const args = {
    request: new Request(
      `https://example.test/analysis/${f.owned.analysisId}/score`,
    ),
    params: { id: f.owned.analysisId },
  };
  return {
    ...f,
    workspace,
    store,
    args,
    setRow: (next: unknown) => {
      row = next;
    },
    setRelationshipContext: (next: unknown) => {
      row = { ...initialRow, relationshipContext: next };
    },
  };
}
// Fake RPC metadata supports the actual repositories and producers. Native SQL is not exercised.
it.each([false, true])(
  "loads the actual %s-declaration rewrite workspace without changing diagnostics or numbers",
  async (declared) => {
    const f = await setup(declared),
      expected = buildTypedScoreDiagnostics(f.context),
      response = await loader(f.args);
    expect(response.data.canonicalRewrite).toEqual(f.workspace);
    expect(response.data.canonicalRewrite?.sources[0].basis).toBe(
      declared ? "candidate_declaration" : "source_quote",
    );
    expect(response.data.canonicalRewrite?.saved.legacyDraft?.summary).toBe(
      "Preserved legacy summary",
    );
    expect(response.data.diagnostics).toEqual(expected.diagnostics);
    expect(response.data.typedCoverage).toEqual(expected.typedCoverage);
    expect(
      response.data.diagnostics.score.dimensions.every((d) => d.value === null),
    ).toBe(true);
    expect(response.data.authoring).toBeNull();
    expect(fetch).not.toHaveBeenCalled();
    const call = f.store.rpc.mock.calls.find(
      ([name]) => name === "read_canonical_rewrite_workspace_state",
    );
    expect(call?.[1]).toMatchObject({
      p_user_id: f.owned.ownerId,
      p_analysis_id: f.owned.analysisId,
      p_selection: {
        canonicalVersionId: f.context.canonical.versionId,
        requirementVersionId: f.context.requirements.versionId,
      },
    });
    expect(new Headers(response.init?.headers).get("Cache-Control")).toBe(
      "private, no-store",
    );
  },
);

it("permits the independent rewrite read when the typed wording-change flag is off", async () => {
  const f = await setup();
  mocks.flags.TYPED_AUTHORING_ENABLED = false;
  const response = await loader(f.args);
  expect(response.data.typedAuthoring).toBeNull();
  expect(response.data.canonicalRewrite).toEqual(f.workspace);
  mocks.flags.AI_ENABLED = false;
  const readOnly = await loader(f.args);
  expect(readOnly.data.canonicalRewrite).toEqual({

it.each(["foreign", "malformed", "missing"])(
  "fails closed on %s rewrite context while diagnostics remain readable",
  async (kind) => {
    const f = await setup();
    if (kind === "missing")
      f.store.rpc.mockImplementation(async (name) =>
        name === "read_canonical_rewrite_workspace_state"
          ? ok(null)
          : ok(f.context),
      );
    else if (kind === "malformed")
      f.setRow({ status: "available", private: "PRIVATE_BODY" });
    else {
      const foreign = await createRequirementContextFixture({
        confirm: true,
        clauses: ["Built alternate services using PostgreSQL"],
      });
      f.setRelationshipContext(foreign.context);
    }
    const response = await loader(f.args);
    expect(response.data.canonicalRewrite).toBeNull();
    expect(response.data.diagnostics).toEqual(
      buildTypedScoreDiagnostics(f.context).diagnostics,
    );
    expect(JSON.stringify(response.data)).not.toContain("PRIVATE_BODY");
    expect(fetch).not.toHaveBeenCalled();
  },
);
it("preserves authentication, distinct query validation and read-only action/private headers", async () => {
  const f = await setup();
  mocks.user.mockResolvedValueOnce(null);
  await expect(loader(f.args)).rejects.toMatchObject({ status: 302 });
  expect(f.store.rpc).not.toHaveBeenCalled();
  const duplicate = {
    ...f.args,
    request: new Request(f.args.request.url + "?canonical=a&canonical=b"),
  };
  await expect(loader(duplicate)).rejects.toMatchObject({
    init: { status: 422 },
  });
  expect(action()).toMatchObject({
    data: { error: "Scoring diagnostics are read-only." },
    init: { status: 405 },
  });
  const h = await headers({
    parentHeaders: new Headers({ "Set-Cookie": "owned=1" }),
    loaderHeaders: new Headers(),
    actionHeaders: new Headers(),
    errorHeaders: new Headers(),
  } as Parameters<typeof headers>[0]);
  expect(new Headers(h).get("X-Robots-Tag")).toBe("noindex, nofollow");
  expect(new Headers(h).getSetCookie()).toEqual(["owned=1"]);
});

it("offers explicit writing-workspace setup through the actual score loader and panel when no writer exists", async () => {
  const f = await createRequirementContextFixture({ confirm: true });
  const store = ownedStore(f.context);
  // Actual strict bootstrap reader derives the missing descriptor from real RR.
  // No draft ID, timestamp or hash is supplied for this missing state.
  store.rpc.mockImplementation(async (name) => {
    if (name === "read_requirement_relationship_context") return ok(f.context);
    if (name === "read_canonical_rewrite_workspace_state")
      return ok(missingState(f.context));
    throw Error("UNEXPECTED_RPC");
  });
  mocks.user.mockResolvedValue({ id: f.owned.ownerId });
  mocks.admin.mockReturnValue(store.client);
  const handler = createStaticHandler([
    { path: "/analysis/:id/score", Component: ScoreDiagnostics, loader },
  ]);
  const context = await handler.query(
    new Request(`https://example.test/analysis/${f.owned.analysisId}/score`),
  );
  if (context instanceof Response) throw Error("EXPECTED_OWNED_ROUTER_DATA");
  const html = renderToString(
    createElement(StaticRouterProvider, {
      router: createStaticRouter(handler.dataRoutes, context),
      context,
      hydrate: false,
    }),
  ).replaceAll("<!-- -->", "");
  expect(store.rpc.mock.calls.every(([name]) => name.startsWith("read_"))).toBe(
    true,
  );
  expect(store.reads.every((read) => read.table !== "writer_drafts")).toBe(
    true,
  );
  expect(fetch).not.toHaveBeenCalled();
  expect(html).toContain("Set up writing workspace");
});
