-- Independent structural command admission. No caller atoms/hash grant authority.
create function private.grouped_original_normalize(p_envelope jsonb,p_scope jsonb,p_command jsonb) returns jsonb
language plpgsql immutable security invoker set search_path = '' as $$
declare normal jsonb; expected jsonb; selection jsonb; member jsonb; citation jsonb;
  projected jsonb; canonical jsonb; prefix text; key text; value jsonb;
begin
  if private.grouped_import_keys(p_command,array['normalForm','previewHash','idempotencyKey']) is not true then perform private.correction_fail('invalid'); end if;
  if private.grouped_import_uuid(p_command->'idempotencyKey') is not true
    or jsonb_typeof(p_command->'previewHash') is distinct from 'string'
    or (p_command->>'previewHash') collate "C" !~ '^[a-f0-9]{64}$' then perform private.correction_fail('invalid'); end if;
  normal:=p_command->'normalForm'; expected:=normal->'expected';
  if private.grouped_import_keys(normal,array['schemaVersion','operation','binding','expected','acknowledgementPolicy','acknowledgement','unselectedDisposition','selections']) is not true
    or private.grouped_import_keys(expected,array['reviewRevision','reviewHeadId','assignmentRevision','assignmentHeadId','associationRevision','associationHeadId','canonicalHeadId']) is not true then
    perform private.correction_fail('invalid'); end if;
  if normal->>'schemaVersion' is distinct from 'grouped-original-command/1.0.0'
    or normal->>'operation' is distinct from 'approve_and_import_grouped_original'
    or normal->>'acknowledgementPolicy' is distinct from 'relationship_only_not_fact_confirmation'
    or normal->'acknowledgement' is distinct from 'true'::jsonb
    or normal->>'unselectedDisposition' is distinct from 'retain_original'
    or expected->'canonicalHeadId' is distinct from 'null'::jsonb then perform private.correction_fail('invalid'); end if;
  foreach prefix in array array['review','assignment','association'] loop
    value:=expected->(prefix||'Revision');
    if private.evidence_integer(value,0,2147483647) is not true then perform private.correction_fail('invalid'); end if;
    if value='0'::jsonb then
      if expected->(prefix||'HeadId') is distinct from 'null'::jsonb then perform private.correction_fail('invalid'); end if;
    elsif private.grouped_import_uuid(expected->(prefix||'HeadId')) is not true then perform private.correction_fail('invalid'); end if;
  end loop;
  foreach key in array array['ownerId','organizationId','resumeId','analysisId','sourceId','attemptId','inputSha256','evidenceSchemaVersion','fieldExtractorVersion','targetModelVersion'] loop
    if normal->'binding'->key is distinct from p_scope->key then perform private.correction_fail('conflict'); end if;
  end loop;
  if jsonb_typeof(normal->'selections') is distinct from 'array' then perform private.correction_fail('invalid'); end if;
  if jsonb_array_length(normal->'selections') not between 1 and 32 then perform private.correction_fail('invalid'); end if;
  for selection in select x.value from jsonb_array_elements(normal->'selections') x loop
    if private.grouped_import_keys(selection,array['pageId','method','members']) is not true
      or jsonb_typeof(selection->'pageId') is distinct from 'string'
      or private.grouped_import_utf16_length(selection->>'pageId') not between 1 and 200
      or selection->>'method' not in ('native_text','local_ocr') or jsonb_typeof(selection->'method') is distinct from 'string'
      or jsonb_typeof(selection->'members') is distinct from 'array' then perform private.correction_fail('invalid'); end if;
    if jsonb_array_length(selection->'members') not between 2 and 3 then perform private.correction_fail('invalid'); end if;
    for member in select x.value from jsonb_array_elements(selection->'members') x loop
      if private.grouped_import_keys(member,array['assertionId','field','citations']) is not true
        or jsonb_typeof(member->'assertionId') is distinct from 'string'
        or private.grouped_import_utf16_length(member->>'assertionId') not between 1 and 200
        or jsonb_typeof(member->'field') is distinct from 'string' or member->>'field' not in ('employer','role','dates')
        or jsonb_typeof(member->'citations') is distinct from 'array' then perform private.correction_fail('invalid'); end if;
      if jsonb_array_length(member->'citations')<>1 then perform private.correction_fail('invalid'); end if;
      citation:=member->'citations'->0;
      if private.grouped_import_keys(citation,array['attemptId','spanId','pageId','method','start','end','pageStart','pageEnd']) is not true
        or private.grouped_import_uuid(citation->'attemptId') is not true
        or jsonb_typeof(citation->'spanId') is distinct from 'string' or jsonb_typeof(citation->'pageId') is distinct from 'string'
        or private.grouped_import_utf16_length(citation->>'spanId') not between 1 and 200
        or private.grouped_import_utf16_length(citation->>'pageId') not between 1 and 200
        or jsonb_typeof(citation->'method') is distinct from 'string' or citation->>'method' not in ('native_text','local_ocr') then
        perform private.correction_fail('invalid'); end if;
      foreach key in array array['start','end','pageStart','pageEnd'] loop
        if private.evidence_integer(citation->key,0,200000) is not true then perform private.correction_fail('invalid'); end if;
      end loop;
    end loop;
  end loop;
  if private.grouped_import_keys(normal->'binding',array['ownerId','organizationId','resumeId','analysisId','sourceId','attemptId','inputSha256',
    'evidenceSchemaVersion','fieldExtractorVersion','targetModelVersion','associationModelVersion','producerVersion','admissionPolicyVersion',
    'structuralTextProfileVersion','correctionProjectionModelVersion','correctionAdmissionPolicyVersion','effectiveLocale']) is not true then
    perform private.correction_fail('invalid'); end if;
  for key in select jsonb_object_keys(normal->'binding') loop
    if jsonb_typeof(normal->'binding'->key) is distinct from 'string'
      or private.grouped_import_utf16_length(normal->'binding'->>key)>200 then perform private.correction_fail('invalid'); end if;
  end loop;
  if octet_length(normal::text)>131072 then perform private.correction_fail('invalid'); end if;
  -- The source projector validates every binding/model/locale and all source references.
  projected:=private.grouped_import_project_candidates(p_envelope,normal->'binding');
  select coalesce(jsonb_agg(x.value order by x.ordinality),'[]'::jsonb) into canonical
    from jsonb_array_elements(projected) with ordinality x(value,ordinality)
    where x.value in (select y.value from jsonb_array_elements(normal->'selections') y);
  if canonical is distinct from normal->'selections' then perform private.correction_fail('invalid'); end if;
  return normal;
end;
$$;

-- Fixed field order is the frozen A02OP1 contract, not jsonb text serialization.
create function private.grouped_original_atoms(p_normal jsonb) returns text[]
language plpgsql immutable security invoker set search_path = '' as $$
declare atoms text[]:=array[p_normal->>'schemaVersion',p_normal->>'operation']; key text; expected_value jsonb;
  selection jsonb; member jsonb; citation jsonb;
begin
  foreach key in array array['ownerId','organizationId','resumeId','analysisId','sourceId','attemptId','inputSha256',
    'evidenceSchemaVersion','fieldExtractorVersion','targetModelVersion','associationModelVersion','producerVersion',
    'admissionPolicyVersion','structuralTextProfileVersion','correctionProjectionModelVersion','correctionAdmissionPolicyVersion','effectiveLocale'] loop
    atoms:=array_append(atoms,p_normal->'binding'->>key);
  end loop;
  foreach key in array array['reviewRevision','reviewHeadId','assignmentRevision','assignmentHeadId','associationRevision','associationHeadId','canonicalHeadId'] loop
    expected_value:=p_normal->'expected'->key;
    atoms:=array_append(atoms,case when expected_value='null'::jsonb then null when jsonb_typeof(expected_value)='number' then (expected_value#>>'{}')::numeric::bigint::text else expected_value#>>'{}' end);
  end loop;
  atoms:=atoms||array[p_normal->>'acknowledgementPolicy','1',p_normal->>'unselectedDisposition',jsonb_array_length(p_normal->'selections')::text];
  for selection in select value from jsonb_array_elements(p_normal->'selections') loop
    atoms:=atoms||array[selection->>'pageId',selection->>'method',jsonb_array_length(selection->'members')::text];
    for member in select value from jsonb_array_elements(selection->'members') loop
      atoms:=atoms||array[member->>'assertionId',member->>'field',jsonb_array_length(member->'citations')::text];
      for citation in select value from jsonb_array_elements(member->'citations') loop
        atoms:=atoms||array[citation->>'attemptId',citation->>'spanId',citation->>'pageId',citation->>'method',
          (citation->>'start')::numeric::bigint::text,(citation->>'end')::numeric::bigint::text,
          (citation->>'pageStart')::numeric::bigint::text,(citation->>'pageEnd')::numeric::bigint::text];
      end loop;
    end loop;
  end loop;
  return atoms;
end;
$$;

create function private.grouped_original_scope(p_user_id uuid,p_analysis_id uuid) returns jsonb
language plpgsql volatile security invoker set search_path = '' as $$
declare scope jsonb:=private.correction_scope(p_user_id,p_analysis_id); prefix jsonb; review_head uuid; assignment_head uuid;
begin
  select id into review_head from public.evidence_corrections where analysis_id=p_analysis_id and revision=(scope->>'reviewRevision')::bigint;
  select id into assignment_head from public.candidate_correction_assignments where analysis_id=p_analysis_id and assignment_revision=(scope->>'assignmentRevision')::bigint;
  prefix:=private.entry_association_prefix(scope);
  return scope||jsonb_build_object('reviewHeadId',review_head,'assignmentHeadId',assignment_head)||prefix;
end;
$$;

create function private.grouped_original_receipt(p_batch_id uuid,p_replayed boolean) returns jsonb
language sql volatile security invoker set search_path = '' as $$
  select jsonb_build_object('schemaVersion','grouped-original-result/1.0.0','status','applied','versionId',result_version_id,
    'revision',result_revision,'batchId',id,'associationRevision',association_revision,'associationHeadId',id,
    'idempotent',p_replayed,'scoringEligible',false) from public.candidate_entry_association_batches where id=p_batch_id;
$$;

create function public.import_canonical_grouped_original(p_user_id uuid,p_analysis_id uuid,p_command jsonb,p_original jsonb,p_request_id text) returns jsonb
language plpgsql volatile security definer set search_path = '' as $$
declare scope jsonb; evidence public.document_evidence_attempts%rowtype; normal jsonb; atoms text[]; bytes bytea; digest text;
  prior public.candidate_entry_association_batches%rowtype; expected jsonb; placement jsonb; document jsonb; event jsonb; mapping jsonb; member jsonb;
  batch_id uuid; group_id uuid; storage_key uuid; group_order integer:=0; entry_order integer; source_order integer;
  storage_command jsonb; association_revision integer; key text;
begin
  if p_user_id is null or p_analysis_id is null or p_request_id is null or char_length(p_request_id) not between 1 and 100 then perform private.correction_fail('invalid'); end if;
  -- correction_scope takes owner-write:v1 before row locks and rechecks privacy/current source.
  scope:=private.grouped_original_scope(p_user_id,p_analysis_id);
  if private.canonical_correction_write_ready() is not true then perform private.correction_fail('unavailable'); end if;
  select * into strict evidence from public.document_evidence_attempts where id=(scope->>'attemptId')::uuid
    and owner_id=p_user_id and analysis_id=p_analysis_id and resume_version_id=(scope->>'sourceId')::uuid and input_sha256=scope->>'inputSha256';
  normal:=private.grouped_original_normalize(evidence.envelope,scope,p_command);
  atoms:=private.grouped_original_atoms(normal); bytes:=private.grouped_import_command_bytes(atoms);
  digest:=private.grouped_import_command_sha256(atoms);
  if p_command->>'previewHash' is distinct from digest then perform private.correction_fail('invalid'); end if;
  -- ORIGINAL captured prefixes/locale are compared here. No fresh CAS or raw-body validation before replay.
  select * into prior from public.candidate_entry_association_batches where resume_id=(scope->>'resumeId')::uuid
    and idempotency_key=(p_command->>'idempotencyKey')::uuid;
  if found then
    if prior.command is distinct from p_command or prior.command_bytes is distinct from bytes or prior.command_sha256 is distinct from digest
      or prior.codec_version is distinct from 'a02-command-bytes/1.0.0'
      or (prior.owner_id,prior.organization_id,prior.analysis_id,prior.source_resume_version_id,prior.evidence_attempt_id,prior.input_sha256)
        is distinct from (p_user_id,(scope->>'organizationId')::uuid,p_analysis_id,(scope->>'sourceId')::uuid,(scope->>'attemptId')::uuid,scope->>'inputSha256') then
      perform private.correction_fail('conflict'); end if;
    perform private.entry_association_validate_head(prior.result_version_id);
    return private.grouped_original_receipt(prior.id,true);
  end if;
  expected:=normal->'expected';
  if scope->'headId' is distinct from 'null'::jsonb or expected->'canonicalHeadId' is distinct from scope->'headId' then perform private.correction_fail('conflict'); end if;
  foreach key in array array['reviewRevision','reviewHeadId','assignmentRevision','assignmentHeadId','associationRevision','associationHeadId'] loop
    if expected->key is distinct from scope->key then perform private.correction_fail('conflict'); end if;
  end loop;
  if (scope->>'associationRevision')::bigint>=1000 or private.correction_original_import_allowed(scope) is not true then perform private.correction_fail('conflict'); end if;
  -- Output locale is original submitted intent; the source validator enforces the frozen canonical syntax.
  placement:=private.grouped_import_place_original(evidence.envelope,normal->'binding',p_original,normal->'selections');
  document:=placement->'document'; event:=document->'changes'->0;
  -- Technical IDs below are allocated only after an exact replay misses.
  batch_id:=gen_random_uuid(); storage_key:=gen_random_uuid(); association_revision:=(scope->>'associationRevision')::integer+1;
  storage_command:=jsonb_build_object('schemaVersion','grouped-original-storage/1.0.0','operation','grouped_original_import',
    'batchId',batch_id,'semanticSha256',digest,'storageKey',storage_key);
  insert into public.candidate_entry_association_batches(id,owner_id,organization_id,resume_id,analysis_id,source_resume_version_id,
    evidence_attempt_id,input_sha256,association_revision,previous_batch_id,idempotency_key,command,command_bytes,command_sha256,codec_version,
    result_version_id,result_revision,storage_key,result_content_sha256,created_at)
    values(batch_id,p_user_id,(scope->>'organizationId')::uuid,(scope->>'resumeId')::uuid,p_analysis_id,(scope->>'sourceId')::uuid,
      evidence.id,scope->>'inputSha256',association_revision,(scope->>'associationHeadId')::uuid,(p_command->>'idempotencyKey')::uuid,p_command,bytes,digest,
      'a02-command-bytes/1.0.0',(document->>'versionId')::uuid,1,storage_key,encode(extensions.digest(convert_to(document::text,'UTF8'),'sha256'),'hex'),now());
  insert into public.canonical_resume_versions(id,resume_id,source_resume_version_id,organization_id,owner_id,locale,state,content,created_at,
    parent_version_id,source_evidence_version_id,revision,idempotency_key,command,request_id)
    values((document->>'versionId')::uuid,(scope->>'resumeId')::uuid,(scope->>'sourceId')::uuid,(scope->>'organizationId')::uuid,p_user_id,
      document->>'locale',document->>'state',document,(document->>'createdAt')::timestamptz,null,evidence.id,1,storage_key,storage_command,p_request_id);
  insert into public.resume_facts(canonical_version_id,id,organization_id,owner_id,type,value,state,sources,confirmation)
    select (document->>'versionId')::uuid,(value->>'id')::uuid,(scope->>'organizationId')::uuid,p_user_id,
      value->>'type',value->>'value',value->>'state',value->'sources',null from jsonb_array_elements(document->'facts');
  for mapping in select value from jsonb_array_elements(placement->'mappings') loop
    group_order:=group_order+1; group_id:=gen_random_uuid();
    insert into public.candidate_entry_association_groups(id,batch_id,owner_id,organization_id,resume_id,source_resume_version_id,result_version_id,
      group_order,page_id,method,section_id,entry_id,source_selection)
      values(group_id,batch_id,p_user_id,(scope->>'organizationId')::uuid,(scope->>'resumeId')::uuid,(scope->>'sourceId')::uuid,(document->>'versionId')::uuid,
        group_order,mapping->'selection'->>'pageId',mapping->'selection'->>'method',(mapping->>'sectionId')::uuid,(mapping->>'entryId')::uuid,mapping->'selection');
    entry_order:=0;
    for member in select value from jsonb_array_elements(mapping->'members') loop
      entry_order:=entry_order+1;
      select ordinality::integer into strict source_order from jsonb_array_elements(mapping->'selection'->'members') with ordinality
        where value->'assertionId'=member->'assertionId';
      insert into public.candidate_entry_association_members(id,group_id,batch_id,owner_id,organization_id,resume_id,source_resume_version_id,result_version_id,
        source_order,entry_order,assertion_id,field,fact_id,original_entry_id,citations)
        values(gen_random_uuid(),group_id,batch_id,p_user_id,(scope->>'organizationId')::uuid,(scope->>'resumeId')::uuid,(scope->>'sourceId')::uuid,
          (document->>'versionId')::uuid,source_order,entry_order,member->>'assertionId',member->>'field',(member->>'factId')::uuid,
          (member->>'originalEntryId')::uuid,member->'citations');
    end loop;
  end loop;
  insert into public.change_events(canonical_version_id,id,owner_id,organization_id,actor_id,operation,event,created_at)
    values((document->>'versionId')::uuid,(event->>'id')::uuid,p_user_id,(scope->>'organizationId')::uuid,p_user_id,'import',event,(document->>'createdAt')::timestamptz);
  insert into public.canonical_resume_heads(resume_id,version_id,revision,owner_id,organization_id)
    values((scope->>'resumeId')::uuid,(document->>'versionId')::uuid,1,p_user_id,(scope->>'organizationId')::uuid);
  perform private.write_correction_snapshot((document->>'versionId')::uuid,storage_command);
  perform private.entry_association_validate_head((document->>'versionId')::uuid);
  insert into public.audit_events(organization_id,actor_id,action,target_type,target_id,outcome,request_id)
    values((scope->>'organizationId')::uuid,p_user_id,'canonical.grouped_import','canonical_resume_version',(document->>'versionId')::uuid,'success',p_request_id);
  return private.grouped_original_receipt(batch_id,false);
exception
  when insufficient_privilege then return jsonb_build_object('schemaVersion','grouped-original-result/1.0.0','status','not_found','scoringEligible',false);
  when sqlstate 'P0C01' or unique_violation then return jsonb_build_object('schemaVersion','grouped-original-result/1.0.0','status','conflict','scoringEligible',false);
  when invalid_parameter_value or invalid_text_representation or numeric_value_out_of_range then return jsonb_build_object('schemaVersion','grouped-original-result/1.0.0','status','invalid','scoringEligible',false);
  when others then return jsonb_build_object('schemaVersion','grouped-original-result/1.0.0','status','unavailable','scoringEligible',false);
end;
$$;

]$patch_manifest$::jsonb; item jsonb; change jsonb; routine pg_catalog.pg_proc%rowtype;
  source text; next_source text; definition text; prepared text[]:='{}'; statement text; role_name text;
begin
  for item in select value from jsonb_array_elements(manifest) loop
    select * into routine from pg_catalog.pg_proc where oid=to_regprocedure(item->>'signature');
    if not found or routine.prokind<>'f' or routine.proretset or routine.prorettype<>to_regtype(item->>'resultType')
      or routine.prosecdef is distinct from (item->>'definer')::boolean or routine.provolatile<>item->>'volatility'
      or cardinality(routine.proconfig) is distinct from 1
      or not exists(select 1 from pg_options_to_table(routine.proconfig) where option_name='search_path' and option_value in ('','""'))
      or encode(extensions.digest(convert_to(routine.prosrc,'UTF8'),'sha256'),'hex')<>item->>'sha256' then
      raise exception 'A02_COMPATIBILITY_DRIFT' using errcode='23514'; end if;
    foreach role_name in array array['anon','authenticated','service_role'] loop
      if has_function_privilege(role_name,routine.oid,'EXECUTE') is distinct from (role_name='service_role' and (item->>'serviceExecute')::boolean) then
        raise exception 'A02_COMPATIBILITY_ACL_DRIFT' using errcode='23514'; end if;
    end loop;
    source:=routine.prosrc; next_source:=source;
    for change in select value from jsonb_array_elements(item->'patches') loop
      if (length(next_source)-length(replace(next_source,change->>'old','')))/length(change->>'old')<>1 then
        raise exception 'A02_COMPATIBILITY_CONTEXT_DRIFT' using errcode='23514'; end if;
      next_source:=replace(next_source,change->>'old',change->>'new');
    end loop;
    if encode(extensions.digest(convert_to(next_source,'UTF8'),'sha256'),'hex')<>item->>'newSha256' then raise exception 'A02_COMPATIBILITY_RESULT_DRIFT' using errcode='23514'; end if;
    definition:=pg_get_functiondef(routine.oid);
    if (length(definition)-length(replace(definition,source,'')))/length(source)<>1 then raise exception 'A02_COMPATIBILITY_DEFINITION_DRIFT' using errcode='23514'; end if;
    prepared:=array_append(prepared,replace(definition,source,next_source));
  end loop;
  foreach statement in array prepared loop execute statement; end loop;
end;
