const hash = (s: string) =>
  createHash("sha256").update(s, "utf8").digest("hex");
const esc = (s: string) => s.replace(/[.*+?^${}()|[\]\\]/gu, "\\$&");
// Delimited text and JSON patch data only. Never a SQL parser or database call.
function body(s: string, name: string) {
  const m = new RegExp(
    `create(?: or replace)? function ${esc(name)}\\([^;]*?\\bas\\s+(\\$[A-Za-z0-9_]*\\$)([\\s\\S]*?)\\1;`,
  ).exec(s);
  if (!m) throw Error("MISSING_SOURCE_BODY");
  return m[2];
}
function literal(s: string, name: string) {
  const m = new RegExp(`\\$${name}\\$([\\s\\S]*?)\\$${name}\\$`).exec(s);
  if (!m) throw Error("MISSING_SOURCE_LITERAL");
  return m[1];
}
function replace(s: string, old: string, next: string) {
  expect(s.split(old)).toHaveLength(2);
  return s.replace(old, next);
}
type Change = { old: string; new: string };
type HistoricalPatch = {
  signature: string;
  sha256?: string;
  before?: string;
  newSha256?: string;
  after?: string;
  patches?: Change[];
  changes?: Change[];
};
function apply(source: string, patch: HistoricalPatch) {
  expect(hash(source)).toBe(patch.sha256 ?? patch.before);
  for (const c of patch.patches ?? patch.changes ?? [])
    source = replace(source, c.old, c.new);
  expect(hash(source)).toBe(patch.newSha256 ?? patch.after);
  return source;
}
const file = (name: string) => read(`supabase/migrations/${name}.sql`);
type Patch = {
  signature: string;
  before: string;
  after: string;
  old: string;
  new: string;
  setof: boolean;
};

const byName = (name: string) => {
  const p = patches.find((p) => p.signature.startsWith(name + "("));
  if (!p) throw Error("MISSING_FORWARD_PATCH");
  return p;
};
