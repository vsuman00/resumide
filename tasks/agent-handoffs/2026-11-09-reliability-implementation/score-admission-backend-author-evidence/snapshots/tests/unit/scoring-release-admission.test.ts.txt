function withheld(
  d: Pick<ScoreDiagnostics, "score" | "readinessAggregate" | "policyState">,
) {
  expect(d.score.state).toBe("withheld");
  expect(d.score.calibrationReceipt).toBeNull();
  expect(d.readinessAggregate).toBeNull();
  expect(d.policyState).toBe("draft");
  expect(d.score.dimensions.map((x) => x.id).sort()).toEqual(
    [...ScoreDimensionId.options].sort(),
  );
  for (const dimension of d.score.dimensions) {
    expect(dimension.value).toBeNull();
    expect(dimension.eligible).toBe(false);
    expect(dimension.coverage).toBeNull();
    expect(dimension.blockers).toContain("calibration_not_approved");
  }
  expect(WithheldScoreV2Schema.safeParse(d.score).success).toBe(true);
}

  function storedRead() {
    const input = scoreFixture();
    const snapshot: Awaited<ReturnType<typeof loadCanonical>> = {
      analysisId: input.canonical.analysisId,
      resumeId: input.canonical.resumeId,
      current: {
        document: input.canonical,
        revision: 1,
        sourceId: input.evidence.sourceId,
      },
      headId: input.canonical.versionId,
      history: [],
      nextOffset: null,
      comparison: null,
      scoringEligible: false,
    };
    readers.canonical.mockResolvedValue(snapshot);
    const evidenceRow = {
      envelope: input.evidence,
      resume_version_id: input.evidence.sourceId,
      input_sha256: input.evidence.inputSha256,
    };
    const rows: Record<string, { data: unknown; error: unknown }> = {
      analyses: { data: { job_id: null }, error: null },
      document_evidence_attempts: { data: evidenceRow, error: null },
      resume_versions: { data: { page_count: 1 }, error: null },
    };
    const client = { from: vi.fn(), rpc: vi.fn() } satisfies Parameters<
      typeof loadScoreDiagnostics
    >[0];
    client.from.mockImplementation((table: string) => {
      if (!rows[table]) throw Error("UNEXPECTED_OWNED_READ");
      return {
        select: vi.fn().mockReturnThis(),
        eq: vi.fn().mockReturnThis(),
        maybeSingle: vi.fn(async () => rows[table]),
      };
    });
    return {
      input,
      snapshot,
      evidenceRow,
      rows,
      client,
      read: (selection: unknown = {}) =>
        loadScoreDiagnostics(
          client,
          input.canonical.analysisId,
          input.canonical.ownerId,
          selection,
          snapshot,
        ),
    };
  }
  it("owned read performs final source/head recheck while returning only withheld diagnostics", async () => {
    const f = storedRead();
    const result = await f.read();
    withheld(result.diagnostics);
    expect(readers.canonical).toHaveBeenCalledExactlyOnceWith(
      f.client,
      f.input.canonical.analysisId,
      f.input.canonical.ownerId,
      { versionId: f.input.canonical.versionId },
    );
    expect(
      f.client.from.mock.calls.filter(([table]) => table === "analyses"),
    ).toHaveLength(2);
    expect(result.diagnostics.readiness.blockers).toContain(
      "job_description_missing",
    );
  });
  it.each([
    "read-error",
    "source-version",
    "source-hash",
    "unknown-receipt",
  ] as const)("owned read refuses %s without number fallback", async (mode) => {
    const f = storedRead();

      f.evidenceRow.resume_version_id = requirementFixtureId(908);

    await expect(
      f.read(
        mode === "unknown-receipt" ? { calibrationReceipt: "synthetic" } : {},
      ),
    ).rejects.toThrow();
  });
});

function calibrationFixture() {
  const { partitionInput } = createEvaluationWorkflowFixture();
  const lock = createEvaluationPartitionLock(partitionInput).lock;
  if (!lock) throw Error("SYNTHETIC_PARTITION_LOCK_UNAVAILABLE");
  const candidates = ScoreDimensionId.options.map((dimension) => ({
    version: "candidate_synthetic",
    dimension,
    weights: [{ featureId: "signal_synthetic", weight: 1 }],
    minimumCoverage: 1,
    threshold: 70,
  }));
  const input = {
    schemaVersion: 1,
    proposalVersion: "proposal_synthetic",
    previousPolicyVersion: getDraftScorePolicy().id,
    rationaleReference: "rationale_synthetic",
    partition: "calibration",
    actorGroups: ["workflow_calibration_group"],
    lock,
    currentPartitionInput: partitionInput,
    candidates,
  };
  const observations = lock.records
    .filter((r) => r.partition === "calibration")
    .flatMap((record) =>
      ScoreDimensionId.options.map((dimension) => ({
        sourceId: record.id,
        sourceHash: record.contentHash,
        dimension,
        features: [{ featureId: "signal_synthetic", credit: 0.8 }],
        referenceScore: 80,
        referencePositive: true,
        state: "evaluated",
      })),
    );
  return { input, observations, lock, partitionInput };
}
