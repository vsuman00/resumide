      ["settleSha256", "public.settle_initial_feedback_attempt"],
      ["metadataSha256", "public.read_initial_feedback_operation_readiness"],
    ] as const;
    for (const [key, name] of pinned)
      expect(profile[key]).toBe(hash(body(name)));
    expect(catalog).toHaveLength(44);
    for (const name of names.filter(
      (x) => x !== "private.initial_feedback_operation_ready",
    ))
      expect(
        catalog.find((x) => x.signature.split("(")[0] === name)?.sha256,
      ).toBe(hash(body(name)));

      catalog.some(
        (x) => x.signature === "private.initial_feedback_operation_ready()",
      ),
    ).toBe(false);
  });

  it("closes current R19 budget and lease postimages rather than their old source bodies", () => {
    const r19 = read(
      "supabase/migrations/20261110085512_privacy_write_barrier.sql",
    );
    const patches = jsonTag(r19, "manifest") as Array<{
      signature: string;
      patches: Array<{ old: string; new: string; count: number }>;
    }>;
    for (const [signature, file] of [
      [
        "private.ai_budget_transition(uuid,uuid,jsonb,text)",
        "20261110062548_technical_ai_budget.sql",
      ],
      [
        "private.lock_analysis_stage_lease(uuid,text,integer,uuid)",
        "20261110082455_analysis_stage_write_fencing.sql",
      ],
    ]) {
      const name = signature.split("(")[0];
      let text = bodyIn(read(`supabase/migrations/${file}`), name);
      for (const patch of patches.find((x) => x.signature === signature)!
        .patches) {
        expect(text.split(patch.old)).toHaveLength(patch.count + 1);
        text = text.replaceAll(patch.old, patch.new);
      }
      expect(catalog.find((x) => x.signature === signature)?.sha256).toBe(
        hash(text),
      );
    }
    expect(
      catalog.find((x) => x.signature === "private.legacy_scoring_ready()")
        ?.sha256,
    ).toBe("56a92f65d533a2f1b053078dbb372b1271728224e5df1c74b201b4cffb21a8d3");
    expect(
      catalog.find(
        (x) =>
          x.signature ===
          "private.legacy_scoring_stored_source(private.analysis_scoring_inputs)",
      )?.sha256,
    ).toBe("abe9dc88974b51a20422def9f71810341e5c4fe2a7bb4e7705cd394c67f5b178");
  });
