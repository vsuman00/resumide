export const CanonicalRewriteHttpResultSchema = z.union([
  z.strictObject({
    kind: z.literal("saved"),
    publication: CanonicalRewritePublicationSchema,
  }),
  z.strictObject({
    kind: z.literal("pending"),
    operationId: z.uuid(),
    state: z.enum(["reserved", "dispatched", "unknown"]),
  }),
]);
export const canonicalRewritePrivateHeaders = () =>
  new Headers({
    "Cache-Control": "private, no-store",
    "X-Content-Type-Options": "nosniff",
    "X-Robots-Tag": "noindex, nofollow",
    Vary: "Cookie, Origin",
  });
class HttpError extends Error {
  constructor(readonly status: number) {
    super("Rewrite request refused.");
  }
}
function parseUniqueJson(text: string) {
  let at = 0,
    count = 0;
  const space = () => {
    while (/[\t\r\n ]/u.test(text[at] ?? "") && at < text.length) at++;
  };
  const string = () => {
    const begin = at++;
    for (; at < text.length; at++) {
      if (text[at] === "\\") {
        at++;
        continue;
      }
      if (text[at] === '"') {
        at++;
        return JSON.parse(text.slice(begin, at)) as string;
      }
    }
    throw new HttpError(422);
  };
  const value = (depth: number): void => {
    if (depth > 8 || ++count > 4096) throw new HttpError(422);
    space();
    const c = text[at];
    if (c === '"') {
      string();
      return;
    }
    if (c === "{") {
      at++;
      space();
      const keys = new Set<string>();
      if (text[at] === "}") {
        at++;
        return;
      }
      for (;;) {
        space();
        if (text[at] !== '"') throw new HttpError(422);
        const key = string();
        if (keys.has(key)) throw new HttpError(422);
        keys.add(key);
        space();
        if (text[at++] !== ":") throw new HttpError(422);
        value(depth + 1);
        space();
        const end = text[at++];
        if (end === "}") return;
        if (end !== ",") throw new HttpError(422);
      }
    }
    if (c === "[") {
      at++;
      space();
      if (text[at] === "]") {
        at++;
        return;
      }
      for (;;) {
        value(depth + 1);
        space();
        const end = text[at++];
        if (end === "]") return;
        if (end !== ",") throw new HttpError(422);
      }
    }
    const match =
      /^(?:true|false|null|-?(?:0|[1-9][0-9]*)(?:\.[0-9]+)?(?:[eE][+-]?[0-9]+)?)/u.exec(
        text.slice(at),
      );
    if (!match) throw new HttpError(422);
    at += match[0].length;
  };
  value(0);
  space();
  if (at !== text.length) throw new HttpError(422);
  return JSON.parse(text) as unknown;
}
export async function readCanonicalRewriteForm(
  request: Request,
  timeoutMs = 5000,
): Promise<CanonicalRewriteRequest> {
  const type = request.headers.get("content-type") ?? "";
  if (
    !/^(application\/x-www-form-urlencoded|multipart\/form-data)(?:;|$)/iu.test(
      type,
    )
  )
    throw new HttpError(415);
  if (!Number.isInteger(timeoutMs) || timeoutMs < 1 || timeoutMs > 5000)
    throw new HttpError(422);
  let reader: ReadableStreamDefaultReader<Uint8Array> | undefined;
  const deadline = Date.now() + timeoutMs;
  try {
    return await runProviderAttempt({
      timeoutMs,
      signal: request.signal,
      failure: () => new CanonicalRewriteError("unavailable", "retry_exact"),
      execute: async (signal) => {
        if (!request.body) throw new HttpError(422);
        reader = request.body.getReader();
        let total = 0;
        const parts: Uint8Array[] = [];
        for (;;) {
          const next = await reader.read();
          if (signal.aborted || Date.now() >= deadline)
            throw new CanonicalRewriteError("unavailable", "retry_exact");
          if (next.done) break;
          total += next.value.length;
          if (total > 65536) throw new HttpError(413);
          parts.push(next.value);
        }
        const bytes = Buffer.concat(parts),
          text = new TextDecoder("utf-8", { fatal: true }).decode(bytes);
        const form = /^application\/x-www-form-urlencoded/iu.test(type)
          ? new URLSearchParams(text)
          : await new Response(bytes, {
              headers: { "Content-Type": type },
            }).formData();
        if (signal.aborted || Date.now() >= deadline)
          throw new CanonicalRewriteError("unavailable", "retry_exact");
        const entries = [...form.entries()];
        if (
          entries.length !== 2 ||
          form.getAll("request").length !== 1 ||
          form.getAll("aiConsent").length !== 1 ||
          entries.some(
            ([key, value]) =>
              !["request", "aiConsent"].includes(key) ||
              typeof value !== "string",
          )
        )
          throw new HttpError(422);
        if (form.get("aiConsent") !== "accepted") throw new HttpError(403);
        const raw = form.get("request");
        if (typeof raw !== "string" || Buffer.byteLength(raw, "utf8") > 16384)
          throw new HttpError(422);
        const parsed = CanonicalRewriteRequestSchema.safeParse(
          parseUniqueJson(raw),
        );
        if (!parsed.success) throw new HttpError(422);
        return parsed.data;
      },
    });
  } catch (error) {
    if (error instanceof HttpError || error instanceof CanonicalRewriteError)
      throw error;
    throw new HttpError(422);
  } finally {
    if (reader) releaseReader(reader);
  }
}
// A pending read may prevent lock release; cleanup must not replace its error.
function releaseReader(
  reader: ReadableStreamDefaultReader<Uint8Array>,
): boolean {
  const release = () => {
    try {
      reader.releaseLock();
      return true;
    } catch {
      return false;
    }
  };
  try {
    void reader.cancel().then(
      () => true,
      () => false,
    );
  } catch {
    return release();
  }
  return release();
}
export async function canonicalRewriteAction(
  request: Request,
  analysisId: string | undefined,
  transport?: CanonicalRewriteTransport,
): Promise<Response> {
  const headers = canonicalRewritePrivateHeaders();
  const respond = (body: unknown, status = 200) =>
    Response.json(body, { status, headers });
  if (request.method !== "POST") {
    headers.set("Allow", "POST");
    return respond(
      { kind: "error", reason: "invalid_input", recovery: "unavailable" },
      405,
    );
  }
  try {
    const config = getServerConfig();
    if (
      !config.CANONICAL_REWRITE_ENABLED ||
      !config.AUTHORING_ENABLED ||
      !config.SCORING_V2_ENABLED
    )
      throw new CanonicalRewriteError("disabled", "unavailable");
    if (!config.AI_ENABLED)
      throw new CanonicalRewriteError("unavailable", "unavailable");
    const origin = new URL(config.APP_ORIGIN).origin;
    if (
      new URL(request.url).origin !== origin ||
      request.headers.get("Origin") !== origin
    )
      throw new HttpError(403);
    const user = await getAuthenticatedUser(
      createSupabaseServerClient(request, headers),
    );
    if (!user) throw new HttpError(401);
    if (
      !z.uuid().safeParse(analysisId).success ||
      !z.uuid().safeParse(user.id).success
    )
      throw new HttpError(404);
    const input = await readCanonicalRewriteForm(request);
    const admin = createSupabaseAdminClient();
    const result = await runCanonicalRewrite(
      admin,
      analysisId!,
      user.id,
      input,
      {
        model: config.OPENAI_MODEL,
        expectedProfile: CANONICAL_REWRITE_READINESS_PROFILE,
        timeoutMs: config.OPENAI_TIMEOUT_MS,
        signal: request.signal,
        getTransport: () =>
          transport ?? createCanonicalRewriteTransport(config),
        budget: config.AI_BUDGET_ENABLED
          ? {
              policyVersion: config.AI_BUDGET_POLICY_VERSION,
              maxOutputTokens: config.OPENAI_MAX_OUTPUT_TOKENS,
            }
          : undefined,
        authorize: async (workspace) => {
          await requireAndRecordAiConsent({
            userId: user.id,
            accepted: true,
            requestId: input.operationId,
          });
          const window = Math.ceil(config.OPENAI_TIMEOUT_MS / 1000) + 60;
          for (const [key, limit, seconds] of [
            [
              `rewrite:${createHash("sha256").update(analysisId!).digest("hex")}`,
              1,
              window,
            ],
            [
              `rewrite-user:${createHash("sha256").update(user.id).digest("hex")}`,
              3,
              300,
            ],
          ] as const) {
            const allowed = await admin.rpc("consume_analysis_rate_limit", {
              p_key: key,
              p_limit: limit,
              p_window_seconds: seconds,
            });
            if (allowed.error || allowed.data !== true)
              throw new CanonicalRewriteError("unavailable", "retry_exact");
          }
          if (!config.AI_BUDGET_ENABLED) {
            const usage = await readCanonicalRewriteUsage(
              admin,
              {
                ownerId: user.id,
                organizationId:
                  workspace.binding.authoring.source.organizationId,
                analysisId: analysisId!,
              },
              { signal: request.signal },
            );
            const reserve =
              Buffer.byteLength(
                JSON.stringify(canonicalRewritePayload(workspace)),
                "utf8",
              ) +
              4000 +
              config.OPENAI_MAX_OUTPUT_TOKENS;
            if (
              BigInt(usage.totalTokens) + BigInt(reserve) >
              BigInt(config.AI_MONTHLY_TOKEN_LIMIT)
            )
              throw new CanonicalRewriteError("unavailable", "review_again");
          }
        },
      },
    );
    return respond(CanonicalRewriteHttpResultSchema.parse(result));
  } catch (error) {
    if (error instanceof HttpError)
      return respond(
        {
          kind: "error",
          reason: error.status === 404 ? "not_found" : "invalid_input",
          recovery: [401, 403, 404].includes(error.status)
            ? "unavailable"
            : "review_again",
        },
        error.status,
      );
    if (error instanceof CanonicalRewriteError)
      return respond(
        { kind: "error", reason: error.reason, recovery: error.recovery },
        error.status,
      );
    if (error instanceof AiBudgetError) {
      const exhausted = ["AI_BUDGET_EXHAUSTED", "AI_BUDGET_BLOCKED"].includes(
        error.code,
      );
      const denied = ["AI_BUDGET_DENIED", "AI_BUDGET_DISABLED"].includes(
        error.code,
      );
      return respond(
        {
          kind: "error",
          reason: "unavailable",
          recovery: exhausted
            ? "review_again"
            : denied
              ? "unavailable"
              : "retry_exact",
        },
        exhausted ? 429 : 503,
      );
    }
    return respond(
      { kind: "error", reason: "unavailable", recovery: "retry_exact" },
      503,
    );
  }
}
