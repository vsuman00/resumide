const read = (path: string) =>
  readFileSync(new URL(`../../${path}`, import.meta.url), "utf8");
const candidate = read(
  "supabase/migrations/20261115100607_owned_ai_retirement.sql",
);
const source =
  candidate || read("supabase/migrations/20261115082125_owned_ai_usage.sql");
const hash = (value: string) =>
  createHash("sha256").update(value, "utf8").digest("hex");
const body = (text: string, name: string) => {
  const escaped = name.replace(/[.*+?^${}()|[\]\\]/gu, "\\$&");
  const match = new RegExp(
    `create(?: or replace)? function ${escaped}\\([\\s\\S]*?\\)\\s*returns [\\s\\S]*?\\bas\\s+(\\$[A-Za-z0-9_]*\\$)([\\s\\S]*?)\\1;`,
    "u",
  ).exec(text);
  if (!match) throw new Error(`Missing actual source ${name}`);
  return match[2];
};

describe("owned AI retirement source contract; SQL unparsed/unexecuted", () => {
  it("requires actual retirement authority before accepting a missing publication", () => {
    // This is an SQL source-admission regression, not a Native behavior test.
    expect(source).toContain("retirement_scope as materialized");
    expect(source).toContain(
      "not exists(select 1 from private.initial_feedback_publications",
    );
    expect(source).toContain("s.publication_id=o.publication_id");
    expect(source).toContain("s.binding_sha256=o.binding_sha256");
  });
  it("captures within the two real producer transactions before parent erasure", () => {
    const prepare = body(source, "public.prepare_deletion_request");
    const original = body(
      read(
        "supabase/migrations/20260911130000_durable_privacy_deletion_plan.sql",
      ),
      "public.prepare_deletion_request",
    );
    expect(
      prepare.replace(
        "  perform private.retire_initial_feedback_publications(request_row.user_id,null,request_row.id,null);\n",
        "",
      ),
    ).toBe(original);
    expect(
      prepare.indexOf("insert into private.privacy_deletion_plans"),
    ).toBeLessThan(
      prepare.indexOf("perform private.retire_initial_feedback_publications"),
    );
    expect(
      prepare.indexOf("perform private.retire_initial_feedback_publications"),
    ).toBeLessThan(prepare.indexOf("delete from public.analyses"));
    const complete = body(source, "public.complete_retention_item");
    expect(
      complete.indexOf("perform private.retire_initial_feedback_publications"),
    ).toBeLessThan(complete.indexOf("delete from public.resumes"));
    expect(complete).toContain(
      "r.owner_id=item_row.owner_id and r.status='deleted'",
    );
    expect(complete).toContain("ri.status<>'deleted'");
  });
  it("requires live durable authority and actual immutable source/publication/run rows", () => {
    const capture = body(
      source,
      "private.retire_initial_feedback_publications",
    );
    for (const text of [
      "private.try_lock_candidate_owners(array[p_owner_id])",
      "private.privacy_deletion_plans",
      "d.lease_expires_at>clock_timestamp()",

      "for update of a skip locked",

      "run.input_hash<>op.binding_sha256",
      "private.ifo_sha(private.ifpub_run_json(run)::text)<>pub.run_content_sha256",
    ])
      expect(capture).toContain(text);
    expect(capture).not.toContain("on conflict");
    expect(capture).not.toMatch(/set_config|p_supported|p_grounded|p_proof/gu);
    const guard = body(source, "private.owned_ai_retirement_guard");
    expect(guard).toContain("if tg_op<>'INSERT'");
    expect(guard).toContain(
      "(new.attempt_id,new.run_id,new.publication_state) is distinct from (pub.attempt_id,pub.run_id,pub.state)",

      usage.indexOf("with canonical_scope"),
      usage.indexOf("), legacy as"),

    expect(prefix).not.toMatch(/period_start|period_end/gu);
    expect(prefix).toContain(
      "s.publication_id is distinct from o.publication_id",

    expect(prefix).toContain("t.state='published' and t.error_code is null");
    expect(prefix).toContain("isfinite(t.claimed_at)");

    expect(usage).toContain("legacy_tokens+canonical_tokens+feedback_tokens");
    expect(usage.match(/x.claimed_at>=period_start/gu)).toHaveLength(2);
  });
  it("pins real composed source bodies without a self-hash cycle", () => {
    expect(Object.keys(full)).toHaveLength(11);
    expect(Object.keys(retirement)).toHaveLength(6);
    for (const [name, value] of [
      ["private.initial_feedback_ready", full.checkerSha256],

it("adopts exact current retirement/full checker successors while retaining every historical100607 body", () => {
  const text = read(
    "supabase/migrations/20261115131621_owned_ai_disposition_clocks.sql",
  );
  const rows = JSON.parse(text.split("$disposition_patches$")[1]) as Array<{
    signature: string;
    before: string;
    after: string;
    old: string;
    new: string;
  }>;
  for (const [name, pin, oldPin] of [
    [
      "private.initial_feedback_ready",
      currentFull.checkerSha256,
      full.checkerSha256,
    ],
    [
      "private.owned_ai_retirement_ready",
      currentRetirement.checkerSha256,
      retirement.checkerSha256,
    ],
  ]) {
    const p = rows.find((x) => x.signature === name + "()")!;
    expect(p.old).toBe(body(source, name));
    expect(p.before).toBe(oldPin);
    expect(hash(p.new)).toBe(pin);
    expect(p.after).toBe(pin);
  }
});
