create function pg_temp.a02_child(p_owner uuid,p_source uuid,p_parent uuid,p_fact uuid,p_value text,p_operation text,p_target uuid default null)
returns uuid language plpgsql set search_path='' as $$
declare body jsonb; prior jsonb; command jsonb; result jsonb; now_text text; new_id uuid:=gen_random_uuid();
begin
  select content into strict body from public.canonical_resume_versions where id=coalesce(p_target,p_parent);
  now_text:=to_char(clock_timestamp() at time zone 'UTC','YYYY-MM-DD"T"HH24:MI:SS.MS"Z"');
  body:=body||jsonb_build_object('versionId',new_id,'parentVersionId',p_parent,'createdAt',now_text);
  if p_operation='rollback' then
    body:=body||jsonb_build_object('changes',jsonb_build_array(jsonb_build_object('id',gen_random_uuid(),'actorId',p_owner,'action','rollback','targetId',new_id,'priorValue',null,'createdAt',now_text)));
  elsif p_operation='rename_section' then
    select value->'title' into strict prior from jsonb_array_elements(body->'sections') where value->>'id'=p_fact::text;
    body:=jsonb_set(body,'{sections}',(select jsonb_agg(case when value->>'id'=p_fact::text then jsonb_set(value,'{title}',to_jsonb(p_value)) else value end order by ordinality) from jsonb_array_elements(body->'sections') with ordinality));
    body:=body||jsonb_build_object('changes',jsonb_build_array(jsonb_build_object('id',gen_random_uuid(),'actorId',p_owner,'action','edit','targetId',p_fact,'priorValue',prior,'createdAt',now_text)));
  else
    select value into strict prior from jsonb_array_elements(body->'facts') where value->>'id'=p_fact::text;
    body:=jsonb_set(body,'{facts}',(select jsonb_agg(case when value->>'id'=p_fact::text then
      case when p_operation='confirm' then value||jsonb_build_object('state','candidate_confirmed','confirmation',jsonb_build_object('actorId',p_owner,'confirmedAt',now_text,'value',value->>'value'))
      else value||jsonb_build_object('value',p_value,'state','unconfirmed','confirmation',null,'sources','[]'::jsonb) end else value end order by ordinality) from jsonb_array_elements(body->'facts') with ordinality));
    body:=body||jsonb_build_object('changes',jsonb_build_array(jsonb_build_object('id',gen_random_uuid(),'actorId',p_owner,'action','edit','targetId',p_fact,'priorValue',prior->'value','createdAt',now_text)));
  end if;
  command:=jsonb_build_object('operation',p_operation,'expectedVersionId',p_parent,'idempotencyKey',new_id,
    'factId',case when p_operation in ('edit','confirm') then p_fact else null end,'value',case when p_operation in ('edit','rename_section') then p_value else null end,'targetVersionId',p_target);
  if p_operation='rename_section' then command:=command||jsonb_build_object('sectionId',p_fact); end if;
  result:=public.save_canonical_version(p_owner,p_source,body,command,'a02-native-owned');
  perform pg_temp.a02_assert(result->>'status'='applied','ordinary_child_commits');
  return new_id;
end;
$$;
