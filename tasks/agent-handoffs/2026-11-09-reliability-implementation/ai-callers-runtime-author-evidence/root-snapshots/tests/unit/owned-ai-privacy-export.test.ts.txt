  Object.assign(op, {

    binding_sha256: hashAiBudgetBinding(binding),

  f.sources.sources.push({
    operationId,
    ownerId: owner,
    organizationId: binding.organizationId,

    resumeId: id(72),
    resumeVersionId: source.resumeVersionId,
    requestNonce: source.requestNonce,
    bindingSha256: op.binding_sha256,
    registeredAt: stamp,
  });
}
