it("frames null, empty and multibyte atoms in exact A11OP2 UTF8 bytes", () => {
  const bytes = encodeClauseOrderCommandAtoms([null, "", "é", "🙂"]);
  expect(
    Array.from(bytes, (b) => b.toString(16).padStart(2, "0")).join(""),
  ).toBe("4131314f50320a4e3b54303a3b54323ac3a93b54343af09f99823b");
});
it("preserves atom boundaries and Unicode bytes without normalizing text", () => {
  expect(encodeClauseOrderCommandAtoms(["a", "bc"])).not.toEqual(
    encodeClauseOrderCommandAtoms(["ab", "c"]),
  );
  expect(encodeClauseOrderCommandAtoms([null])).not.toEqual(
    encodeClauseOrderCommandAtoms([""]),
  );
  expect(encodeClauseOrderCommandAtoms(["é"])).not.toEqual(
    encodeClauseOrderCommandAtoms(["e\u0301"]),
  );
});
it.each<string>(["\u0000", "\ud800", "\udfff", "x\ud800y"])(
  "rejects non-Postgres text %j before UTF8 replacement",
  (atom) => {
    expect(() => encodeClauseOrderCommandAtoms([atom])).toThrow(
      "CLAUSE_ORDER_CODEC_TEXT",
    );
  },
);
it("enforces exact atom and framed-byte bounds", () => {
  expect(encodeClauseOrderCommandAtoms(Array(256).fill(null))).toHaveLength(
    519,
  );
  expect(() => encodeClauseOrderCommandAtoms(Array(257).fill(null))).toThrow();
  expect(encodeClauseOrderCommandAtoms(["x".repeat(16369)])).toHaveLength(
    16384,
  );
  expect(() => encodeClauseOrderCommandAtoms(["x".repeat(16370)])).toThrow();
});

it.each(vectors.vectors)(
  "matches independent Python framing and SHA $sha256",
  async (vector) => {
    const bytes = encodeClauseOrderCommandAtoms(vector.atoms);
    expect(
      Array.from(bytes, (b) => b.toString(16).padStart(2, "0")).join(""),
    ).toBe(vector.framedHex);
    const digest = await webcrypto.subtle.digest("SHA-256", bytes);
    expect(
      Array.from(new Uint8Array(digest), (b) =>
        b.toString(16).padStart(2, "0"),
      ).join(""),
    ).toBe(vector.sha256);
  },
);

async function actualProposal() {
  const f = await createJdClauseOrderFixture();
  return buildClauseOrderProposal(
    f.context,
    { analysisId: f.owned.analysisId, ownerId: f.owned.ownerId },
    {
      canonicalVersionId: f.context.canonical.versionId,
      requirementVersionId: f.context.requirements.versionId,
      clauseId: f.context.requirements.content.relations.clauses[0].id,
      targetFactId: f.targetFactId,
    },
    { acceptKey: id(90), rejectKey: id(91) },
  );
}
it("validates actual Node producer output without changing ordered draft identity", async () => {
  const p = await actualProposal();
  expect(await validateClauseOrderProposal(p)).toEqual(p);
  expect(JdClauseOrderProposalSchema.parse(p).draft.sourceClauses[0]).toEqual(
    p.draft.sourceClauses[0],
  );
  expect(
    Object.keys(JdClauseOrderProposalSchema.parse(p).draft.sourceClauses[0]),
  ).toEqual(["index", "start", "end", "text", "sha256"]);
});
it("agrees with the Node byte hash and includes the key in exact 53+8N command atoms", async () => {
  const p = await actualProposal(),
    c = p.acceptCommand;
  const atoms = clauseOrderCommandAtoms(c);
  expect(atoms).toHaveLength(53 + 8 * c.sourceCitations.length);
  expect(atoms.slice(0, 5)).toEqual([
    c.schemaVersion,
    c.operation,
    c.idempotencyKey,
    "true",
    c.draftId,
  ]);
  expect(atoms[32]).toBe(c.target.bulletId);
  expect(atoms.slice(43, 47)).toEqual([
    "0",
    String(c.sourceClauses[0].start),
    String(c.sourceClauses[0].end),
    c.sourceClauses[0].sha256,
  ]);
  expect(await clauseOrderCommandSha256(c)).toBe(nodeCommandSha256(c));
  expect(await clauseOrderCommandSha256(c)).toBe(p.acceptCommandSha256);
  expect(
    await clauseOrderCommandSha256({ ...c, idempotencyKey: id(92) }),
  ).not.toBe(p.acceptCommandSha256);
});

function acceptedReceipt(p: JdClauseOrderProposal) {
  const c = p.acceptCommand;
  return {
    schemaVersion: "jd-clause-order-receipt/2.0.0",
    status: "applied",
    action: "accept",
    idempotencyKey: c.idempotencyKey,
    commandSha256: p.acceptCommandSha256,
    draftId: c.draftId,
    baseVersionId: c.binding.canonical.versionId,
    baseRevision: c.binding.canonical.revision,
    requirementVersionId: c.binding.requirements.versionId,
    requirementRevision: c.binding.requirements.revision,
    versionId: id(94),
    revision: c.binding.canonical.revision + 2,
    eventId: id(95),
    idempotent: true,
  };
}
it("accepts a complete exact replay receipt and a zero-head-movement rejection receipt", async () => {
  const p = await actualProposal(),
    a = acceptedReceipt(p),
    c = p.rejectCommand;
  expect(await validateClauseOrderReceipt(a, p.acceptCommand)).toEqual(a);
  const rejected = {
    ...a,
    action: "reject",
    idempotencyKey: c.idempotencyKey,
    commandSha256: p.rejectCommandSha256,
    versionId: c.binding.canonical.versionId,
    revision: c.binding.canonical.revision,
  };
  expect(await validateClauseOrderReceipt(rejected, c)).toEqual(rejected);
});

it("matches the independent full-command 61-atom golden without granting semantic admission", async () => {
  const vector = vectors.commandVectors[0];
  const command = JdClauseOrderCommandSchema.parse(vector.command);
  const atoms = clauseOrderCommandAtoms(command);
  expect(atoms).toHaveLength(61);
  expect(atoms).toEqual(vector.atoms);
  const bytes = encodeClauseOrderCommandAtoms(atoms);
  expect(bytes).toHaveLength(1947);
  expect(
    Array.from(bytes, (b) => b.toString(16).padStart(2, "0")).join(""),
  ).toBe(vector.framedHex);
  expect(await clauseOrderCommandSha256(command)).toBe(vector.sha256);
  expect(vector.sha256).toBe(
    "1874dac7cdf2dd07ee238002a90b3c78ec723d69e6727452a995a91d9f9e134b",
  );
});
