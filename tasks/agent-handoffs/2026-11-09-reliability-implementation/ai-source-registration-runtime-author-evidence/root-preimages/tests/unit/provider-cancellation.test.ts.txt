it("rejects an already aborted worker signal before reads or provider dispatch", async () => {
  const execute = vi.fn().mockResolvedValue(qualitativeOutput());
  await expect(
    Reflect.apply(processQualitativeStage, undefined, [
      analysisId,
      1,
      { execute },
      { signal: AbortSignal.abort(), lease },
    ]),
  ).rejects.toMatchObject({ code: "ANALYSIS_CANCELLED" });
  await expect(
    Reflect.apply(processQualitativeStage, undefined, [
      analysisId,
      2,
      { execute },
      { lease },
    ]),
  ).rejects.toMatchObject({ code: "ANALYSIS_LEASE_LOST" });
  expect(execute).not.toHaveBeenCalled();
  expect(mocks.from).not.toHaveBeenCalled();
  expect(mocks.rpc).not.toHaveBeenCalled();
});

it("propagates a worker lease or shutdown abort and rejects an ignoring provider's late result", async () => {
  const controller = new AbortController();
  let started!: () => void;
  const dispatched = new Promise<void>((resolve) => {
    started = resolve;
  });
  let finish!: (value: ReturnType<typeof qualitativeOutput>) => void;
  let child: AbortSignal | undefined;
  const execute = vi.fn((request) => {
    child = request.signal;
    started();
    return new Promise<ReturnType<typeof qualitativeOutput>>((resolve) => {
      finish = resolve;
    });
  });
  const pending = Reflect.apply(processQualitativeStage, undefined, [
    analysisId,
    1,
    { execute },
    { signal: controller.signal, lease },
  ]);
  await dispatched;
  controller.abort();
  finish(qualitativeOutput());
  const outcome = await pending.catch((error: unknown) => error);
  expect(child?.aborted).toBe(true);
  expect(outcome).toMatchObject({ code: "ANALYSIS_CANCELLED" });
  expect(
    mocks.rpc.mock.calls.some(([name]) => name.startsWith("persist_")),
  ).toBe(false);
});

it("settles known cancelled usage without turning cancellation into partial feedback", async () => {
  mocks.budget = true;
  const failure = new QualitativeAiFailure("AI_CANCELLED", false, {
    inputTokens: 10,
    outputTokens: 20,
  });
  await expect(
    Reflect.apply(processQualitativeStage, undefined, [
      analysisId,
      1,
      { execute: vi.fn().mockRejectedValue(failure) },
      { lease },
    ]),
  ).rejects.toBe(failure);
  expect(mocks.rpc).toHaveBeenCalledWith(
    "ai_budget_settle",
    expect.objectContaining({
      p_request: expect.objectContaining({
        usage: { inputTokens: "10", outputTokens: "20" },
      }),
    }),
  );
  expect(
    mocks.rpc.mock.calls.some(([name]) => name.startsWith("persist_")),
  ).toBe(false);
});

it("keeps accounting but preserves the previous draft when the final source read observes disconnect", async () => {
  mocks.budget = true;
  const controller = new AbortController();
  abortOnSourceRead = () => controller.abort();
  const response = await rewriteAction(
    new Request(`https://example.test/resume/${analysisId}/rewrite`, {
      method: "POST",
      headers: {
        Origin: "https://example.test",
        "Content-Type": "application/x-www-form-urlencoded",
        "Idempotency-Key": analysisId,
      },
      body: "aiConsent=accepted",
      signal: controller.signal,
    }),
    analysisId,
    {
      execute: vi.fn().mockResolvedValue({
        output: {
          summary: null,
          bullets: [
            {
              sourceId: 0,
              rewrite: "Created a Python API for event ingestion.",
              reasoning: "Clarifies the contribution.",
            },
          ],
        },
        usage: { inputTokens: 10, outputTokens: 20 },
      }),
    },
  );
  expect(response.status).toBe(409);
  expect(writes).toEqual(["analysis_ai_runs:insert"]);
  expect(
    mocks.rpc.mock.calls.filter(([name]) => name === "ai_budget_settle"),
  ).toHaveLength(1);
});
