"""Verify adopted-source binding with the real review-handoff predicate in PostgreSQL 17."""
import ast
from pathlib import Path
import subprocess
import sys
from tempfile import TemporaryDirectory

migrations = Path("supabase/migrations")

def source(prefix):
    file, = migrations.glob(prefix + "_*.sql")
    return file.read_text()

def section(text, start, end):
    start_at = text.index(start)
    return text[start_at:text.index(end, start_at)]

previous = Path(__file__).with_name("admission-cleanup-recovery-native-fixture.py.txt")
scaffold = next(
    ast.literal_eval(node.value)
    for node in ast.parse(previous.read_text()).body
    if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == "scaffold" for t in node.targets)
)
scaffold += """
alter table public.data_requests add column next_attempt_at timestamptz, add column last_error_code text;
alter table public.analyses add column status text not null default 'completed';
alter table public.resumes add column status text not null default 'active';
create table public.privacy_request_steps(request_id uuid, status text, attempt_count integer, last_error_code text);
create table public.document_evidence_attempts(id uuid primary key, analysis_id uuid, owner_id uuid,
  organization_id uuid,resume_version_id uuid,input_sha256 text,attempt_number integer,envelope jsonb);
"""

checks = """
do $$
declare
  candidate uuid:=gen_random_uuid();
  outsider uuid:=gen_random_uuid();
  org uuid:=gen_random_uuid();
  other_org uuid:=gen_random_uuid();
  request_id_value uuid:=gen_random_uuid();
  operation_id uuid;
  key text;
  upload_id uuid;
  analysis_id uuid;
  resume_id uuid;
  version_id uuid;
  job_id uuid;
  evidence_id uuid;
  owner_id uuid;
  org_id uuid;
  ordinary_upload uuid;
  ordinary_analysis uuid;
  ordinary_version uuid;
  review_upload uuid;
  first_receipt jsonb;
  view_data jsonb;
begin
  insert into auth.users values(candidate),(outsider);
  insert into public.organizations values(org,candidate),(other_org,outsider);
  insert into public.data_requests(id,user_id,kind,status,lease_owner,lease_expires_at,attempt_count,privacy_generation)
    values(request_id_value,candidate,'deletion','processing','worker',clock_timestamp()+interval '2 minutes',1,1);

  for kind in 1..9 loop
    operation_id:=gen_random_uuid();
    upload_id:=gen_random_uuid();
    analysis_id:=gen_random_uuid();
    resume_id:=gen_random_uuid();
    version_id:=gen_random_uuid();
    job_id:=gen_random_uuid();
    evidence_id:=gen_random_uuid();
    owner_id:=case when kind=9 then outsider else candidate end;
    org_id:=case when kind=9 then other_org else org end;
    key:='organizations/'||org_id||'/resumes/'||resume_id||'/versions/'||version_id||'.pdf';
    insert into private.analysis_admission_operations(id,organization_id,owner_id,idempotency_key,binding_sha256)
      values(operation_id,org_id,owner_id,gen_random_uuid(),repeat('a',64));
    insert into private.analysis_admission_uploads(id,operation_id,analysis_id,resume_id,version_id,job_id,storage_key)
      values(upload_id,operation_id,analysis_id,resume_id,version_id,job_id,key);
    if kind<>3 then
      insert into private.analysis_admission_upload_transport(upload_id,bucket,dispatched_at,acknowledged_at,object_id)
        values(upload_id,'resumes',clock_timestamp(),clock_timestamp(),gen_random_uuid());
    end if;
    update private.analysis_admission_uploads set state='adopted' where id=upload_id;
    update private.analysis_admission_operations set adopted_upload_id=upload_id where id=operation_id;
    insert into public.resumes values(resume_id,owner_id,org_id,'active');
    insert into public.resume_versions values(version_id,case when kind=5 then outsider else owner_id end,
      org_id,resume_id,key,100,repeat('b',64));
    insert into public.jobs values(job_id,owner_id,org_id);
    insert into public.analyses values(analysis_id,owner_id,org_id,version_id,job_id,
      case when kind in (2,8) then 'needs_ocr' when kind=7 then 'extracting' else 'completed' end);
    if kind=4 then
      insert into public.analysis_jobs values(analysis_id,'old-worker',clock_timestamp()-interval '1 hour');
    elsif kind=6 then
      insert into public.analyses values(gen_random_uuid(),owner_id,org_id,version_id,job_id,'partial');
    end if;
    if kind in (2,8) then
      insert into public.document_evidence_attempts values(evidence_id,analysis_id,owner_id,org_id,version_id,repeat('b',64),1,
        jsonb_build_object('schemaVersion','evidence/2.0.0','id',evidence_id,'analysisId',analysis_id,'ownerId',owner_id,
          'sourceId',case when kind=8 then gen_random_uuid() else version_id end,'inputSha256',repeat('b',64),
          'scoringEligible',false,'confidence','{"kind":"uncalibrated","reason":"calibration_not_approved"}'::jsonb,'status','review_required'));
    end if;
    if kind=1 then ordinary_upload:=upload_id; ordinary_analysis:=analysis_id; ordinary_version:=version_id; end if;
    if kind=2 then review_upload:=upload_id; end if;
  end loop;

  assert public.capture_analysis_admission_retirement(request_id_value,'other',1) is null;
  assert public.capture_analysis_admission_retirement(request_id_value,'worker',2) is null;
  assert public.capture_analysis_admission_retirement(request_id_value,'worker',1)->>'operations'='8';
  view_data:=private.analysis_admission_export_records(candidate);
  first_receipt:=view_data#>'{analysisAdmissionRetirementCaptures,0,adoptedSourceBindings}';
  assert jsonb_array_length(first_receipt)=2, 'Adopted binding snapshot is missing';
  assert exists(select 1 from jsonb_array_elements(first_receipt) where value->>'uploadId'=ordinary_upload::text);
  assert exists(select 1 from jsonb_array_elements(first_receipt) where value->>'uploadId'=review_upload::text);
  assert (select count(*)=2 from private.analysis_admission_adopted_source_bindings);
  assert not has_table_privilege('service_role','private.analysis_admission_adopted_source_bindings','SELECT,INSERT,UPDATE,DELETE');
  assert not has_function_privilege('service_role','private.analysis_admission_live_adopted_bindings(uuid,uuid)','execute');
  assert not has_function_privilege('authenticated','public.capture_analysis_admission_retirement(uuid,text,integer)','execute');

  update public.data_requests set attempt_count=2 where id=request_id_value;
  perform public.capture_analysis_admission_retirement(request_id_value,'worker',2);
  assert private.analysis_admission_export_records(candidate)#>'{analysisAdmissionRetirementCaptures,0,adoptedSourceBindings}'=first_receipt;
  assert private.analysis_admission_export_records(outsider)->'analysisAdmissionRetirementCaptures'='[]'::jsonb;

  begin
    insert into public.analysis_jobs values(ordinary_analysis,'unexpected',clock_timestamp()+interval '1 minute');
    perform public.capture_analysis_admission_retirement(request_id_value,'worker',2);
    raise exception 'Drifted worker state was accepted';
  exception when check_violation then
    assert sqlerrm='ADMISSION_ADOPTED_SOURCE_CHANGED';
  end;
  begin
    update public.resume_versions set checksum=repeat('c',64) where id=ordinary_version;
    perform public.capture_analysis_admission_retirement(request_id_value,'worker',2);
    raise exception 'Drifted source was accepted';
  exception when check_violation then
    assert sqlerrm='ADMISSION_ADOPTED_SOURCE_CHANGED';
  end;
  assert private.analysis_admission_export_records(candidate)#>'{analysisAdmissionRetirementCaptures,0,adoptedSourceBindings}'=first_receipt;
  assert (select count(*)=9 from public.resume_versions);
  assert not exists(select 1 from private.analysis_admission_cleanup_receipts);
end;
$$;
"""

versions = ["20261121130718","20261121132235","20261121203700","20261121205500","20261121211200"]
if "--baseline" not in sys.argv:
    versions.append("20261121212400")
sql = "\n".join([
    scaffold,
    section(source("20261110080809"),"create table private.analysis_admission_operations","create function private.analysis_admission_denied"),
    section(source("20261109164851"),"create function private.candidate_owner_lock_keys","-- A false try-lock"),
    section(source("20261116075250"),"create function private.privacy_generation_claim","create function public.complete_privacy_request"),
    section(source("20261110080810"),"create function private.analysis_job_v2_review_handoff","create function public.claim_analysis_job_v2"),
    *(source(version) for version in versions), checks,
])

with TemporaryDirectory(prefix="resumide-adopted-binding-") as tmp:
    env = {"HOME": tmp, "PATH": "/opt/homebrew/bin:/usr/bin:/bin", "LC_ALL": "C"}
    data = str(Path(tmp) / "data")

    def run(argv, **options):
        return subprocess.run(argv, env=env, timeout=25, text=True, check=True, **options)

    assert run(["postgres","--version"],capture_output=True).stdout.startswith("postgres (PostgreSQL) 17.")
    run(["initdb","-D",data,"--no-sync","--auth=trust","-U","fixture_owner"],stdout=subprocess.DEVNULL)
    try:
        run(["pg_ctl","-w","-D",data,"-l",str(Path(tmp)/"postgres.log"),
             "-o",f"-c listen_addresses='' -c unix_socket_directories={tmp} -c unix_socket_permissions=0700","start"],
            stdout=subprocess.DEVNULL)
        run(["psql","-X","-v","ON_ERROR_STOP=1","-h",tmp,"-U","fixture_owner","-d","postgres","-Atq"],input=sql)
        print("PASS: completed and evidence-review adopted sources, seven exclusions, stable replay, source and queue drift refusal, and owner export.")
    finally:
        if (Path(data)/"postmaster.pid").exists():
            run(["pg_ctl","-w","-D",data,"-m","immediate","stop"],stdout=subprocess.DEVNULL)
