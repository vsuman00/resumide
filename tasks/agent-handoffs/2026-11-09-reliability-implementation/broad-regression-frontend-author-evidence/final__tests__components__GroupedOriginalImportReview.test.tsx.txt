const id = (value: number) =>
  `10000000-0000-4000-8000-${String(value).padStart(12, "0")}`;
function sourceFixture(
  canCommit = false,
  text = "Experience\nEmployer: Synthetic North Labs\nRole: Engineer\nDates: 2020 - 2022\nEmployer: Synthetic South Labs\nRole: Analyst\nDates: 2021 - 2023",
) {
  const context = {
    id: id(1),
    ownerId: id(2),
    analysisId: id(3),
    sourceId: id(4),
    inputSha256: "a".repeat(64),
    createdAt: "2026-11-13T00:00:00.000Z",
    retentionPolicyId: "synthetic/1.0.0",
  };
  const envelope = DocumentEvidenceSchema.parse(
    reconcileDocumentEvidence(context, buildNativeEvidence([text])),
  );
  const source = {
    ownerId: context.ownerId,
    organizationId: id(6),
    resumeId: id(5),
    analysisId: context.analysisId,
    sourceId: context.sourceId,
    attemptId: context.id,
    inputSha256: context.inputSha256,
  };
  const captured = {
    reviewRevision: 2,
    reviewHeadId: id(7),
    assignmentRevision: 1,
    assignmentHeadId: id(8),
    associationRevision: 0,
    associationHeadId: null,
    canonicalHeadId: null,
  };
  const readData = {
    schemaVersion: "grouped-original-read/1.0.0",
    status: "available",
    compatibilityReady: true,
    canCommit,
    freshImportAllowed: true,
    scoringEligible: false,
    source,
    captured,
    attempt: { id: context.id, attemptNumber: 1, envelope },
  };
  const rpc = vi.fn().mockResolvedValue({ data: readData, error: null });
  return { envelope, source, captured, rpc, readData };
}
const storageKey = `resumide:grouped-original:pending:${id(2)}:${id(3)}`;
const routers: ReturnType<typeof createMemoryRouter>[] = [];
beforeEach(() => {
  vi.clearAllMocks();
  sessionStorage.clear();
  mocks.user.mockResolvedValue({ id: id(2) });
});
afterEach(() => {
  cleanup();
  for (const router of routers.splice(0)) router.dispose();
  vi.restoreAllMocks();
});
function baseModel() {
  return {
    analysisId: id(3),
    resumeId: id(5),
    current: null,
    headId: null,
    history: [],
    nextOffset: null,
    comparison: null,
    scoringEligible: false,
  };
}
async function workspaceFixture(canCommit = false) {
  const fixture = sourceFixture(canCommit);
  const workspace = await loadGroupedOriginalWorkspace(
    { rpc: fixture.rpc },
    id(3),
    id(2),
  );
  if (workspace.state !== "available") throw Error("EXPECTED_OWNED_WORKSPACE");
  const request = {
    schemaVersion: "grouped-original-command/1.0.0" as const,
    operation: "approve_and_import_grouped_original" as const,
    binding: workspace.binding,
    expected: { ...workspace.captured, canonicalHeadId: null },
    selectedGroupIds: workspace.groups.map((group) => group.id),
    unselectedDisposition: "retain_original" as const,
  };
  const preview = await previewGroupedOriginalImport(
    { rpc: fixture.rpc },
    id(3),
    id(2),
    request,
  );
  return { ...fixture, workspace, preview };
}
function storageRecord(command: GroupedOriginalCommand, savedAt = Date.now()) {
  return JSON.stringify({
    schemaVersion: "pending-grouped-original/1.0.0",
    savedAt,
    command,
  });
}
async function wirePage({
  canCommit = false,
  outcome = "applied",
  text,
}: {
  canCommit?: boolean;
  outcome?: "applied" | "ambiguous" | "conflict";
  text?: string;
} = {}) {
  const fixture = sourceFixture(canCommit, text);
  let saved: ReturnType<typeof CanonicalResumeDraftSchema.parse> | null = null;
  const commands: unknown[] = [];
  mocks.canonicalLoad.mockImplementation(async () =>
    saved
      ? {
          ...baseModel(),
          headId: saved.versionId,
          current: { sourceId: id(4), revision: 1, document: saved },
          history: [
            {
              id: saved.versionId,
              revision: 1,
              parent_version_id: null,
              created_at: saved.createdAt,
            },
          ],
        }
      : baseModel(),
  );
  mocks.rpc.mockImplementation(
    async (name: string, args: Record<string, unknown>) => {
      if (name === "read_grouped_original_workspace")
        return {
          data: {
            ...fixture.readData,
            freshImportAllowed: saved === null,
            captured: {
              ...fixture.captured,
              canonicalHeadId: saved?.versionId ?? null,
            },
          },
          error: null,
        };
      if (name !== "import_canonical_grouped_original")
        throw Error("UNEXPECTED_RPC");
      commands.push(args.p_command);
      const original = CanonicalResumeDraftSchema.parse(args.p_original);
      const stored = JSON.parse(sessionStorage.getItem(storageKey) ?? "null");
      if (
        !args.p_command ||
        typeof args.p_command !== "object" ||
        !("idempotencyKey" in args.p_command)
      )
        throw Error("EXPECTED_WIRE_COMMAND");
      expect(stored.command.idempotencyKey).toBe(args.p_command.idempotencyKey);
      expect(stored).not.toHaveProperty("relationshipAcknowledgement");
      if (outcome === "conflict")
        return {
          data: {
            schemaVersion: "grouped-original-result/1.0.0",
            status: "conflict",
            scoringEligible: false,
          },
          error: null,
        };
      const idempotent = saved !== null;
      if (!saved)
        saved = CanonicalResumeDraftSchema.parse({
          ...original,
          producerVersion: "canonical-engine/1.0.3",
        });
      const receipt = {
        schemaVersion: "grouped-original-result/1.0.0",
        status: "applied",
        versionId: saved.versionId,
        revision: 1,
        batchId: id(40),
        associationRevision: 1,
        associationHeadId: id(40),
        idempotent,
        scoringEligible: false,
      };
      return {
        data:
          outcome === "ambiguous" && commands.length === 1
            ? { ...receipt, unexpected: true }
            : receipt,
        error: null,
      };
    },
  );
  const router = createMemoryRouter(
    [
      {
        path: "/analysis/:id/canonical",
        loader,
        action,
        shouldRevalidate,
        Component: CanonicalResume,
        HydrateFallback: () => null,
      },
    ],
    { initialEntries: [`/analysis/${id(3)}/canonical`] },
  );
  routers.push(router);
  render(<RouterProvider router={router} />);
  await screen.findByRole("heading", {
    name: "Review possible employment entries",
  });
  return { router, commands, fixture };
}
async function previewSelection() {
  fireEvent.click(
    screen.getByRole("checkbox", { name: "Select possible entry 1" }),
  );
  fireEvent.click(
    screen.getByRole("button", { name: "Preview selected relationships" }),
  );
  return screen.findByRole("region", { name: "Selected relationship preview" });
}
function accept(region: HTMLElement) {
  fireEvent.click(
    within(region).getByRole("checkbox", {
      name: /I reviewed these entry relationships/,
    }),
  );
}

describe("grouped original import in the actual canonical page", () => {
  it("shows unverified employment relationships from the real owned workspace DTO", async () => {
    const fixture = sourceFixture();
    const groupedOriginal = await loadGroupedOriginalWorkspace(
      { rpc: fixture.rpc },
      id(3),
      id(2),
    );
    expect(groupedOriginal.state).toBe("available");
    if (groupedOriginal.state !== "available")
      throw Error("EXPECTED_OWNED_WORKSPACE");
    expect(groupedOriginal.groups).toHaveLength(2);
    expect(groupedOriginal.canPreview).toBe(true);
    expect(groupedOriginal.canCommit).toBe(false);
    const router = createMemoryRouter(
      [
        {
          path: "/analysis/:id/canonical",
          loader: () => ({
            analysisId: id(3),
            resumeId: id(5),
            current: null,
            headId: null,
            history: [],
            nextOffset: null,
            comparison: null,
            scoringEligible: false,
            formKey: id(10),
            factKeys: {},
            retainedKeys: {},
            restoreKey: id(11),
            scoringDiagnosticsEnabled: false,
            groupedOwnerId: id(2),
            groupedOriginal,
            entryRegroup: {
              schemaVersion: "entry-regroup-workspace/1.0.0",
              state: "unavailable",
              reason: "unavailable",
              scoringEligible: false,
            } satisfies CanonicalEntryRegroupWorkspace,
          }),
          Component: CanonicalResume,
        },
      ],
      { initialEntries: [`/analysis/${id(3)}/canonical`] },
    );
    render(<RouterProvider router={router} />);
    expect(
      await screen.findByRole("button", { name: "Import resume for review" }),
    ).toBeTruthy();
    expect(
      screen.getByRole("heading", {
        name: "Review possible employment entries",
      }),
    ).toBeTruthy();
  });

  it("previews manually with nonzero captures while denied commit remains disabled", async () => {
    await wirePage();
    const previewButton = screen.getByRole("button", {
      name: "Preview selected relationships",
    });
    expect(previewButton).toHaveProperty("disabled", true);
    expect(
      screen
        .getAllByRole("checkbox")
        .every(
          (input) => !(input instanceof HTMLInputElement) || !input.checked,
        ),
    ).toBe(true);
    expect(screen.getByText("Synthetic North Labs")).toBeTruthy();
    expect(screen.getByText("2020 - 2022")).toBeTruthy();
    expect(
      screen
        .getAllByRole("link", { name: "View cited source (native text)" })[0]
        .getAttribute("href"),
    ).toContain(`/analysis/${id(3)}/evidence#source-`);
    const region = await previewSelection();
    expect(
      within(region).getByRole("button", {
        name: "Save selected entry relationships",
      }),
    ).toHaveProperty("disabled", true);
    expect(within(region).getByRole("checkbox")).toHaveProperty(
      "disabled",
      true,
    );
    expect(
      mocks.rpc.mock.calls.filter(
        (call) => call[0] === "import_canonical_grouped_original",
      ),
    ).toHaveLength(0);
    expect(sessionStorage.getItem(storageKey)).toBeNull();
  });

  it("requires named acceptance, stores the exact metadata command before dispatch, and clears matching success", async () => {
    const page = await wirePage({ canCommit: true });
    const region = await previewSelection();
    expect(
      within(region).getByRole("button", {
        name: "Save selected entry relationships",
      }),
    ).toHaveProperty("disabled", true);
    const checkbox = within(region).getByRole("checkbox");
    expect(checkbox.getAttribute("name")).toBe("relationshipAcknowledgement");
    expect(checkbox.getAttribute("value")).toBe("accepted");
    fireEvent.submit(checkbox.closest("form")!);
    expect(page.commands).toHaveLength(0);
    accept(region);
    fireEvent.click(
      within(region).getByRole("button", {
        name: "Save selected entry relationships",
      }),
    );
    await screen.findByRole("link", { name: "Open the saved version" });
    expect(page.commands).toHaveLength(1);
    await waitFor(() => expect(sessionStorage.getItem(storageKey)).toBeNull());
    expect(
      screen.getByRole("heading", { name: "Facts and confirmation" }),
    ).toBeTruthy();
  });

  it("keeps the original key/captures after an ambiguous save creates a head, reloads unchecked, and replays once manually", async () => {
    const page = await wirePage({ canCommit: true, outcome: "ambiguous" });
    const region = await previewSelection();
    accept(region);
    fireEvent.click(
      within(region).getByRole("button", {
        name: "Save selected entry relationships",
      }),
    );
    let pending = await screen.findByRole("region", {
      name: "Pending original import",
    });
    await screen.findByRole("heading", { name: "Facts and confirmation" });
    expect(page.commands).toHaveLength(1);
    const originalStored = sessionStorage.getItem(storageKey)!;
    const original = GroupedOriginalCommandSchema.parse(
      JSON.parse(originalStored).command,
    );
    expect(original.expected).toMatchObject({
      reviewRevision: 2,
      reviewHeadId: id(7),
      assignmentRevision: 1,
      assignmentHeadId: id(8),
      canonicalHeadId: null,
    });
    expect(within(pending).getByRole("checkbox")).toHaveProperty(
      "checked",
      false,
    );
    cleanup();
    page.router.dispose();
    const router = createMemoryRouter(
      [
        {
          path: "/analysis/:id/canonical",
          loader,
          action,
          shouldRevalidate,
          Component: CanonicalResume,
          HydrateFallback: () => null,
        },
      ],
      { initialEntries: [`/analysis/${id(3)}/canonical`] },
    );
    routers.push(router);
    render(<RouterProvider router={router} />);
    pending = await screen.findByRole("region", {
      name: "Pending original import",
    });
    expect(within(pending).getByRole("checkbox")).toHaveProperty(
      "checked",
      false,
    );
    expect(
      within(pending).getByRole("button", { name: "Retry the same import" }),
    ).toHaveProperty("disabled", true);
    expect(page.commands).toHaveLength(1);
    expect(sessionStorage.getItem(storageKey)).toBe(originalStored);
    accept(pending);
    fireEvent.click(
      within(pending).getByRole("button", { name: "Retry the same import" }),
    );
    await screen.findByText(
      /This import was already saved; your current version may be newer/,
    );
    expect(page.commands).toHaveLength(2);
    expect(page.commands[1]).toEqual(page.commands[0]);
    await waitFor(() => expect(sessionStorage.getItem(storageKey)).toBeNull());
  });

  it("keeps conflict recovery separate from a new preview and never retries automatically", async () => {
    const page = await wirePage({ canCommit: true, outcome: "conflict" });
    const region = await previewSelection();
    accept(region);
    fireEvent.click(
      within(region).getByRole("button", {
        name: "Save selected entry relationships",
      }),
    );
    await screen.findByText(
      /Review the latest source and create a new preview/,
    );
    const pending = await screen.findByRole("region", {
      name: "Pending original import",
    });
    expect(within(pending).getByRole("checkbox")).toHaveProperty(
      "checked",
      false,
    );
    expect(page.commands).toHaveLength(1);
    expect(sessionStorage.getItem(storageKey)).not.toBeNull();
  });

  it("invalidates the preview on a changed selection/locale and on persisted pageshow", async () => {
    await wirePage({ canCommit: true });
    let region = await previewSelection();
    const originalKey = JSON.parse(
      region.querySelector<HTMLInputElement>('input[name="command"]')!.value,
    ).idempotencyKey;
    accept(region);
    fireEvent.change(screen.getByRole("textbox", { name: "Output locale" }), {
      target: { value: "fr-FR" },
    });
    expect(
      screen.queryByRole("region", { name: "Selected relationship preview" }),
    ).toBeNull();
    fireEvent.change(screen.getByRole("textbox", { name: "Output locale" }), {
      target: { value: "en-IN" },
    });
    expect(
      screen.queryByRole("region", { name: "Selected relationship preview" }),
    ).toBeNull();
    fireEvent.click(
      screen.getByRole("button", { name: "Preview selected relationships" }),
    );
    region = await screen.findByRole("region", {
      name: "Selected relationship preview",
    });
    expect(
      JSON.parse(
        region.querySelector<HTMLInputElement>('input[name="command"]')!.value,
      ).idempotencyKey,
    ).not.toBe(originalKey);
    expect(within(region).getByRole("checkbox")).toHaveProperty(
      "checked",
      false,
    );
    accept(region);
    fireEvent.click(
      screen.getByRole("checkbox", { name: "Select possible entry 1" }),
    );
    fireEvent.click(
      screen.getByRole("checkbox", { name: "Select possible entry 1" }),
    );
    expect(
      screen.queryByRole("region", { name: "Selected relationship preview" }),
    ).toBeNull();
    fireEvent.click(
      screen.getByRole("button", { name: "Preview selected relationships" }),
    );
    region = await screen.findByRole("region", {
      name: "Selected relationship preview",
    });
    expect(within(region).getByRole("checkbox")).toHaveProperty(
      "checked",
      false,
    );
    fireEvent(window, new PageTransitionEvent("pageshow", { persisted: true }));
    expect(
      screen.queryByRole("region", { name: "Selected relationship preview" }),
    ).toBeNull();
    fireEvent.click(
      screen.getByRole("checkbox", { name: "Select possible entry 1" }),
    );
    expect(
      screen.queryByRole("region", { name: "Selected relationship preview" }),
    ).toBeNull();
    expect(sessionStorage.getItem(storageKey)).toBeNull();
  });

  it.each([
    "expired",
    "future",
    "foreign",
    "unknown_field",
    "raw_value",
    "oversized",
    "bad_json",
  ])(
    "rejects %s stored metadata without touching another namespace",
    async (kind) => {
      const fixture = await workspaceFixture(true);
      let value = storageRecord(fixture.preview.command);
      if (kind === "expired")
        value = storageRecord(
          fixture.preview.command,
          Date.now() - 30 * 60 * 1000,
        );
      if (kind === "future")
        value = storageRecord(fixture.preview.command, Date.now() + 60_000);
      if (kind === "foreign")
        value = storageRecord({
          ...fixture.preview.command,
          binding: { ...fixture.preview.command.binding, ownerId: id(99) },
        });
      if (kind === "unknown_field")
        value = JSON.stringify({
          ...JSON.parse(value),
          relationshipAcknowledgement: "accepted",
        });
      if (kind === "raw_value")
        value = JSON.stringify({
          ...JSON.parse(value),
          command: {
            ...fixture.preview.command,
            value: "Synthetic forbidden source copy",
          },
        });
      if (kind === "oversized") value = "x".repeat(20_001);
      if (kind === "bad_json") value = "{";
      sessionStorage.setItem(storageKey, value);
      sessionStorage.setItem("other-owner:unrelated", "keep");
      await wirePage({ canCommit: true });
      await waitFor(() =>
        expect(sessionStorage.getItem(storageKey)).toBeNull(),
      );
      expect(
        screen.queryByRole("region", { name: "Pending original import" }),
      ).toBeNull();
      expect(sessionStorage.getItem("other-owner:unrelated")).toBe("keep");
      expect(screen.queryByText("Synthetic forbidden source copy")).toBeNull();
    },
  );

  it("restores only metadata, requires fresh named acceptance, and preserves original locale after capture movement", async () => {
    const fixture = await workspaceFixture(true);
    const command = {
      ...fixture.preview.command,
      binding: { ...fixture.preview.command.binding, effectiveLocale: "fr-FR" },
    };
    sessionStorage.setItem(storageKey, storageRecord(command));
    const page = await wirePage({ canCommit: true });
    const pending = await screen.findByRole("region", {
      name: "Pending original import",
    });
    expect(within(pending).getByRole("checkbox")).toHaveProperty(
      "checked",
      false,
    );
    expect(page.commands).toHaveLength(0);
    expect(JSON.parse(sessionStorage.getItem(storageKey)!).command).toEqual(
      command,
    );
    page.fixture.captured.reviewRevision = 3;
    page.fixture.captured.reviewHeadId = id(71);
    await act(async () => {
      page.router.revalidate();
    });
    await waitFor(() => expect(page.router.state.revalidation).toBe("idle"));
    expect(JSON.parse(sessionStorage.getItem(storageKey)!).command).toEqual(
      command,
    );
    expect(
      within(
        screen.getByRole("region", { name: "Pending original import" }),
      ).getByRole("checkbox"),
    ).toHaveProperty("checked", false);
    page.fixture.source.attemptId = id(70);
    page.fixture.readData.attempt.id = id(70);
    page.fixture.readData.attempt.envelope = {
      ...page.fixture.envelope,
      id: id(70),
    };
    await act(async () => {
      page.router.revalidate();
    });
    await waitFor(() => expect(sessionStorage.getItem(storageKey)).toBeNull());
    expect(
      screen.queryByRole("region", { name: "Pending original import" }),
    ).toBeNull();
  });

  it("resets preview and acknowledgement when actual captured revisions move", async () => {
    const page = await wirePage({ canCommit: true });
    const region = await previewSelection();
    accept(region);
    page.fixture.captured.reviewRevision = 3;
    page.fixture.captured.reviewHeadId = id(71);
    await act(async () => {
      page.router.revalidate();
    });
    await waitFor(() =>
      expect(
        screen.queryByRole("region", { name: "Selected relationship preview" }),
      ).toBeNull(),
    );
    expect(
      screen.getByRole("checkbox", { name: "Select possible entry 1" }),
    ).toHaveProperty("checked", false);
    expect(page.commands).toHaveLength(0);
  });

  it("displays every derived group while bounding each selection to32", async () => {
    const text =
      "Experience\n" +
      Array.from(
        { length: 33 },
        (_, index) =>
          `Employer: Synthetic Org ${index}\nRole: Synthetic Role ${index}\nDates: 2020 - 2022`,
      ).join("\n");
    await wirePage({ text });
    const checkboxes = screen.getAllByRole("checkbox");
    expect(checkboxes).toHaveLength(33);
    for (const checkbox of checkboxes) fireEvent.click(checkbox);
    expect(
      screen.getByRole("button", { name: "Preview selected relationships" }),
    ).toHaveProperty("disabled", true);
    expect(
      screen.getByText("Choose no more than 32 entries for one import."),
    ).toBeTruthy();
    expect(screen.getByText("Synthetic Org 32")).toBeTruthy();
    expect(
      mocks.rpc.mock.calls.filter(
        (call) => call[0] === "import_canonical_grouped_original",
      ),
    ).toHaveLength(0);
  });

  it("reports storage failure honestly and keeps an in-memory retry", async () => {
    await wirePage({ canCommit: true, outcome: "ambiguous" });
    const region = await previewSelection();
    vi.spyOn(Storage.prototype, "setItem").mockImplementation(() => {
      throw Error("synthetic storage denial");
    });
    accept(region);
    fireEvent.click(
      within(region).getByRole("button", {
        name: "Save selected entry relationships",
      }),
    );
    await screen.findByText(
      /This tab could not store or restore the pending import/,
    );
    expect(
      await screen.findByRole("region", { name: "Pending original import" }),
    ).toBeTruthy();
  });
});
