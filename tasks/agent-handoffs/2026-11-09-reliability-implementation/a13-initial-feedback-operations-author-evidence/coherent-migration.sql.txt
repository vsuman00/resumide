create function private.ifo_fail(p_code text) returns void
language plpgsql volatile set search_path='' as $ifo_0$
begin
  case p_code
    when 'invalid' then raise exception 'INITIAL_FEEDBACK_INVALID' using errcode='22023';
    when 'not_found' then raise exception 'INITIAL_FEEDBACK_NOT_FOUND' using errcode='42501';
    when 'stale' then raise exception 'INITIAL_FEEDBACK_STALE' using errcode='P0C01';
    else raise exception 'INITIAL_FEEDBACK_UNAVAILABLE' using errcode='P0C03';
  end case;
end;
$ifo_0$;

create function private.ifo_keys(p_value jsonb,p_expected text[]) returns boolean
language plpgsql immutable set search_path='' as $ifo_1$
begin
  if jsonb_typeof(p_value) is distinct from 'object' then return false;end if;
  return (select array_agg(key order by key collate "C") from jsonb_object_keys(p_value) key)
    is not distinct from (select array_agg(key order by key collate "C") from unnest(p_expected) key);
end;
$ifo_1$;

create function private.ifo_sha(p_value text) returns text
language plpgsql immutable set search_path='' as $ifo_2$
begin
  return encode(sha256(convert_to(p_value,'UTF8')),'hex');
end;
$ifo_2$;

create function private.ifo_atom(p_value text) returns text
language plpgsql immutable set search_path='' as $ifo_3$
begin
  if p_value is null then return 'N;';end if;
  return 'T'||octet_length(convert_to(p_value,'UTF8'))::text||':'||p_value||';';
end;
$ifo_3$;

create function private.ifo_stamp(p_value timestamptz) returns text
language plpgsql stable set search_path='' as $ifo_4$
begin
  if p_value is null then return null;end if;
  return to_char(p_value at time zone 'UTC','YYYY-MM-DD"T"HH24:MI:SS.US"Z"');
end;
$ifo_4$;

create function private.ifo_usage_valid(p_usage jsonb) returns boolean
language plpgsql immutable set search_path='' as $ifo_5$
declare key text;
begin
  if p_usage is null or p_usage='null'::jsonb then return true;end if;
  if private.ifo_keys(p_usage,array['inputTokens','outputTokens']) is not true then return false;end if;
  foreach key in array array['inputTokens','outputTokens'] loop
    if jsonb_typeof(p_usage->key) is distinct from 'string'
      or (p_usage->>key) collate "C" !~ '^(0|[1-9][0-9]{0,15})$' then return false;end if;
    if (p_usage->>key)::numeric>9007199254740991 then return false;end if;
  end loop;
  return true;
end;
$ifo_5$;

create function private.ifo_policy_frame(p_policy jsonb) returns text
language plpgsql immutable set search_path='' as $ifo_6$
declare answer text:=E'INITIAL-FEEDBACK-POLICY/1\n';key text;
begin
  foreach key in array array['version','provider','model','promptVersion','maxOutputTokens','attemptLimit','budgetPolicyVersion',
    'supportVersion','supportSha256','planVersion','planSha256','writingSupportPolicyVersion'] loop
    answer:=answer||private.ifo_atom(p_policy->>key);
  end loop;
  foreach key in array array['schemaVersion','profileVersion','checkerSha256','readerSha256','finishSha256','contextSha256','exportSha256','metadataSha256'] loop
    answer:=answer||private.ifo_atom(p_policy->'scoringInputProfile'->>key);
  end loop;
  foreach key in array array['schemaVersion','profileVersion','policyVersion','policySha256','checkerSha256','rendererSha256','metadataSha256'] loop
    answer:=answer||private.ifo_atom(p_policy->'planProfile'->>key);
  end loop;
  return answer;
end;
$ifo_6$;

 created_at timestamptz not null default clock_timestamp(),
 foreign key(policy_version,policy_sha256) references private.initial_feedback_policies(version,sha256) on delete restrict,
 check(jsonb_typeof(binding)='object' and octet_length(convert_to(binding::text,'UTF8'))<=32768),
 check(binding_sha256 ~ '^[a-f0-9]{64}$' and binding_sha256=private.ifo_sha(binding::text))

 created_at timestamptz not null default clock_timestamp(),unique(operation_id,ordinal),
 check(private.ifo_usage_valid(usage)),check(usage is null or usage<>'null'::jsonb),
 check(error_code is null or error_code in ('provider_failed','timeout','cancelled','invalid_output','unsupported_output','storage_failed','unknown')),
 check((state='reserved' and claim_lease is null and claimed_at is null and usage is null and error_code is null and budget_attempt_id is null)
   or (state<>'reserved' and private.legacy_scoring_lease(claim_lease) is true and claimed_at is not null)),
 check((state in ('dispatched','held_unknown') and usage is null) or (state in ('known','failed') and usage is not null) or state='reserved'),
 check((state='known' and error_code is null) or state<>'known'),check((state='failed' and error_code is not null) or state<>'failed')
);

create function private.ifo_selected_policy(p_org uuid,p_selector jsonb) returns jsonb
language plpgsql volatile set search_path='' as $ifo_8$

    or jsonb_typeof(p_selector->'sha256') is distinct from 'string'
    or (p_selector->>'sha256') collate "C" !~ '^[a-f0-9]{64}$' then perform private.ifo_fail('invalid');end if;

  if not found or private.ifo_policy_valid(pol) is not true then perform private.ifo_fail('unavailable');end if;
  return pol;
end;

create function private.ifo_original(p_analysis public.analyses) returns jsonb
language plpgsql volatile set search_path='' as $ifo_9$

    and organization_id=p_analysis.organization_id for key share skip locked;

  select * into d from public.analysis_deterministic_results where analysis_id=p_analysis.id and owner_id=p_analysis.owner_id

    or d.result_checksum is distinct from s.result_checksum then perform private.ifo_fail('unavailable');end if;

  if ctx->>'schemaVersion' is distinct from 'legacy-scoring-input/1.0.0'
    or ctx->>'analysisId' is distinct from s.analysis_id::text or ctx->>'ownerId' is distinct from s.owner_id::text
    or ctx->>'organizationId' is distinct from s.organization_id::text
    or ctx->'source'->>'resumeId' is distinct from s.resume_id::text or ctx->'source'->>'resumeVersionId' is distinct from s.resume_version_id::text
    or ctx->'binding'->>'analysisId' is distinct from s.analysis_id::text or ctx->'binding'->>'ownerId' is distinct from s.owner_id::text
    or ctx->'binding'->>'organizationId' is distinct from s.organization_id::text
    or ctx->'binding'->>'resumeId' is distinct from s.resume_id::text or ctx->'binding'->>'resumeVersionId' is distinct from s.resume_version_id::text
    or ctx->'binding'->>'inputBindingSha256' is distinct from s.input_binding_sha256
    or ctx->'binding'->>'jobId' is distinct from s.job_id::text

    or ctx->'binding'->>'textChecksum' is distinct from ctx->'extraction'->>'textChecksum'

    or ctx->'binding'->>'extractionSha256' is distinct from ctx->'extraction'->>'contentSha256'
    or ctx->'extraction'->>'contentSha256' is distinct from private.ifo_sha(((ctx->'extraction')-'contentSha256')::text)
    or ctx->'binding'->>'jobSha256' is distinct from ctx->'job'->>'contentSha256'
    or octet_length(convert_to(ctx::text,'UTF8'))>4194304 then perform private.ifo_fail('unavailable');end if;
  if s.job_id is null then
    if ctx->'job' is distinct from jsonb_build_object('kind','none') then perform private.ifo_fail('unavailable');end if;
  elsif ctx->'job'->>'kind' is distinct from 'owned_job' or ctx->'job'->>'id' is distinct from s.job_id::text
    or ctx->'job'->>'contentSha256' is distinct from private.ifo_sha(((ctx->'job')-'contentSha256')::text) then perform private.ifo_fail('unavailable');end if;

  if private.ifo_sha(core::text) is distinct from s.input_binding_sha256 then perform private.ifo_fail('unavailable');end if;
  -- Original admission only. No mutable JD/extraction read before exact replay.
  return jsonb_build_object('scoringContext',ctx,'deterministicResult',private.legacy_scoring_result(d));

create function private.ifo_immutable() returns trigger
language plpgsql volatile set search_path='' as $ifo_13$
begin
  if tg_op='DELETE' then perform private.ifo_fail('stale');end if;
  if tg_table_name='initial_feedback_policies' then
    if tg_op='UPDATE' or private.ifo_policy_valid(new.descriptor) is not true then perform private.ifo_fail('stale');end if;
  elsif tg_table_name='initial_feedback_operations' then
    if tg_op='UPDATE' then perform private.ifo_fail('stale');end if;
    if new.binding->>'analysisId' is distinct from new.analysis_id::text or new.binding->>'ownerId' is distinct from new.owner_id::text
      or new.binding->>'organizationId' is distinct from new.organization_id::text or new.binding->>'resumeId' is distinct from new.resume_id::text
      or new.binding->>'resumeVersionId' is distinct from new.resume_version_id::text
      or new.binding->'policy'->>'version' is distinct from new.policy_version or new.binding->'policy'->>'sha256' is distinct from new.policy_sha256
      or (new.binding->'policy'->>'attemptLimit')::integer is distinct from new.attempt_limit
      or not exists(select 1 from private.initial_feedback_policies where version=new.policy_version and sha256=new.policy_sha256
        and descriptor=new.binding->'policy') then perform private.ifo_fail('invalid');end if;
  else perform private.ifo_fail('invalid');end if;
  return new;
end;

create function private.ifo_attempt_guard() returns trigger
language plpgsql volatile set search_path='' as $ifo_14$
declare op private.initial_feedback_operations%rowtype;previous private.initial_feedback_attempts%rowtype;
begin
  if tg_op='DELETE' then perform private.ifo_fail('stale');end if;
  select * into op from private.initial_feedback_operations where operation_id=new.operation_id for update;
  if not found or op.state<>'open' or op.publication_id is not null then perform private.ifo_fail('stale');end if;
  if tg_op='INSERT' then
    if new.state<>'reserved' or new.ordinal>op.attempt_limit then perform private.ifo_fail('stale');end if;
    select * into previous from private.initial_feedback_attempts where operation_id=op.operation_id order by ordinal desc limit 1 for update;
    if found then
      if new.ordinal<>previous.ordinal+1 or previous.state<>'failed' or previous.usage is null
        or previous.error_code not in ('provider_failed','timeout') then perform private.ifo_fail('stale');end if;
    elsif new.ordinal<>1 then perform private.ifo_fail('stale');end if;
  else
    if (new.attempt_id,new.operation_id,new.ordinal,new.created_at) is distinct from (old.attempt_id,old.operation_id,old.ordinal,old.created_at)
      or (old.claimed_at is not null and (new.claimed_at,new.claim_lease,new.budget_attempt_id) is distinct from (old.claimed_at,old.claim_lease,old.budget_attempt_id))
      or (old.usage is not null and new.usage is distinct from old.usage)
      or (old.error_code is not null and new.error_code is distinct from old.error_code) then perform private.ifo_fail('stale');end if;
    if old.state='reserved' then
      if new.state<>'dispatched' or new.claimed_at is null or new.usage is not null or new.error_code is not null then perform private.ifo_fail('stale');end if;
    elsif new.state not in ('known','held_unknown','failed') then perform private.ifo_fail('stale');end if;
  end if;
  return new;
end;

create function private.ifo_context(p_analysis public.analyses,p_original jsonb) returns jsonb

create function public.prepare_initial_feedback(p_analysis_id uuid,p_lease jsonb,p_policy jsonb,p_retry_after_attempt_id uuid) returns jsonb
language plpgsql volatile security definer set search_path='' as $ifo_16$
declare a public.analyses%rowtype;op private.initial_feedback_operations%rowtype;att private.initial_feedback_attempts%rowtype;
  original jsonb;ctx jsonb;pol jsonb;b jsonb;next_ordinal integer:=1;
begin
  if private.legacy_scoring_lease(p_lease) is not true or p_analysis_id is null
    or private.ifo_keys(p_policy,array['version','sha256']) is not true
    or jsonb_typeof(p_policy->'version') is distinct from 'string' or jsonb_typeof(p_policy->'sha256') is distinct from 'string'
    or p_policy->>'version' is distinct from 'initial-feedback-publication/1.0.0' or (p_policy->>'sha256') collate "C" !~ '^[a-f0-9]{64}$'
    then perform private.ifo_fail('invalid');end if;
  if private.initial_feedback_operation_ready() is not true then perform private.ifo_fail('unavailable');end if;
  a:=private.legacy_scoring_access(p_analysis_id,null);original:=private.ifo_original(a);
  select * into op from private.initial_feedback_operations where analysis_id=a.id for update;
  if found then
    if op.owner_id<>a.owner_id or op.organization_id<>a.organization_id
      or op.binding is distinct from private.ifo_binding(original,op.binding->'policy') then perform private.ifo_fail('not_found');end if;
    if private.legacy_scoring_static_job(a.id,p_lease) is not true and not exists(select 1 from private.initial_feedback_attempts
      where operation_id=op.operation_id and claim_lease=p_lease) then perform private.ifo_fail('not_found');end if;
    select * into att from private.initial_feedback_attempts where operation_id=op.operation_id order by ordinal desc limit 1 for update;
    if not found then perform private.ifo_fail('unavailable');end if;
    -- Null selector and a replayed explicit retry return status BEFORE fresh
    -- policy/lease/current-JD/consent/rate/budget checks. No dispatch context.

      return jsonb_build_object('schemaVersion','initial-feedback-prepare/1.0.0','status','pending','operation',private.ifo_operation_json(op),
        'attempt',private.ifo_attempt_json(att),'context',null,'dispatchAllowed',false,'replayed',true);
    end if;
    if att.attempt_id<>p_retry_after_attempt_id or att.state<>'failed' or att.usage is null or att.error_code not in ('provider_failed','timeout')

    next_ordinal:=att.ordinal+1;

    -- remains usable for AI-off/historical/resource deterministic-only recovery.

    if not found then perform private.ifo_fail('not_found');end if;

      returning * into op;
  end if;
  insert into private.initial_feedback_attempts(operation_id,ordinal) values(op.operation_id,next_ordinal) returning * into att;
  if not private.lock_analysis_stage_lease(a.id,p_lease->>'workerId',(p_lease->>'attempt')::integer,(p_lease->>'leaseFence')::uuid)
    or private.try_candidate_privacy_write(a.owner_id) is not true then perform private.ifo_fail('stale');end if;
  return jsonb_build_object('schemaVersion','initial-feedback-prepare/1.0.0','status','prepared','operation',private.ifo_operation_json(op),
    'attempt',private.ifo_attempt_json(att),'context',ctx,'dispatchAllowed',false,'replayed',false);
end;

create function public.claim_initial_feedback_dispatch(p_analysis_id uuid,p_lease jsonb,p_operation_id uuid,p_attempt_id uuid,p_binding_sha256 text,p_budget jsonb) returns jsonb
language plpgsql volatile security definer set search_path='' as $ifo_17$
declare a public.analyses%rowtype;op private.initial_feedback_operations%rowtype;att private.initial_feedback_attempts%rowtype;
  original jsonb;ctx jsonb;pol jsonb;budget jsonb:=null;req jsonb;
begin
  if p_analysis_id is null or p_operation_id is null or p_attempt_id is null or p_binding_sha256 is null
    or p_binding_sha256 collate "C" !~ '^[a-f0-9]{64}$' or private.legacy_scoring_lease(p_lease) is not true then perform private.ifo_fail('invalid');end if;
  if private.initial_feedback_operation_ready() is not true then perform private.ifo_fail('unavailable');end if;
  a:=private.legacy_scoring_access(p_analysis_id,null);original:=private.ifo_original(a);
  select * into op from private.initial_feedback_operations where operation_id=p_operation_id and analysis_id=a.id
    and owner_id=a.owner_id and organization_id=a.organization_id for update;
  if not found or op.binding_sha256<>p_binding_sha256
    or op.binding is distinct from private.ifo_binding(original,op.binding->'policy') then perform private.ifo_fail('not_found');end if;
  select * into att from private.initial_feedback_attempts where attempt_id=p_attempt_id and operation_id=op.operation_id for update;
  if not found then perform private.ifo_fail('not_found');end if;
  if private.legacy_scoring_static_job(a.id,p_lease) is not true and att.claim_lease is distinct from p_lease then perform private.ifo_fail('not_found');end if;
  if op.state<>'open' or op.publication_id is not null or att.state<>'reserved' then
    return jsonb_build_object('schemaVersion','initial-feedback-claim/1.0.0','operation',private.ifo_operation_json(op),'attempt',private.ifo_attempt_json(att),
      'dispatchAllowed',false,'replayed',true,'budget',null);
  end if;
  if a.status<>'qualitative_review' or not private.lock_analysis_stage_lease(a.id,p_lease->>'workerId',(p_lease->>'attempt')::integer,(p_lease->>'leaseFence')::uuid)
    then perform private.ifo_fail('stale');end if;
  pol:=private.ifo_selected_policy(a.organization_id,jsonb_build_object('version',op.policy_version,'sha256',op.policy_sha256));
  if pol is distinct from op.binding->'policy' then perform private.ifo_fail('stale');end if;
  ctx:=private.ifo_context(a,original);
  if ctx->>'generationEligibility'<>'eligible' or private.ifo_binding(ctx,pol) is distinct from op.binding then perform private.ifo_fail('stale');end if;
  -- The actual frozen source parser checks its source-domain/resource limits
  -- BEFORE any provider permission; it supplies no candidate qualification.
  perform private.ifp_document(ctx->'scoringContext'->'extraction'->>'text');
  perform 1 from public.consents where user_id=a.owner_id and purpose='qualitative_ai' and policy_version='candidate-ai-v1' for share skip locked;
  if not found then perform private.ifo_fail('not_found');end if;
  if pol->'budgetPolicyVersion'='null'::jsonb then
    if p_budget is not null and p_budget<>'null'::jsonb then perform private.ifo_fail('invalid');end if;
  else
    if private.ifo_keys(p_budget,array['p_user_id','p_organization_id','p_request']) is not true
      or p_budget->>'p_user_id' is distinct from a.owner_id::text or p_budget->>'p_organization_id' is distinct from a.organization_id::text
      then perform private.ifo_fail('invalid');end if;
    req:=p_budget->'p_request';
    if private.ai_budget_request_valid(req,'claim_dispatch') is not true or req->>'operationId' is distinct from op.operation_id::text
      or req->>'attemptId' is distinct from att.attempt_id::text or (req->>'attemptOrdinal')::numeric is distinct from att.ordinal::numeric
      or req->>'surface' is distinct from 'qualitative' or req->>'model' is distinct from pol->>'model'
      or req->>'promptPolicyVersion' is distinct from pol->>'promptVersion' or req->>'policyVersion' is distinct from pol->>'budgetPolicyVersion'
      or req->>'payloadSha256' is distinct from op.binding_sha256
      or (req->>'reservedOutputTokens')::numeric is distinct from (pol->>'maxOutputTokens')::numeric then perform private.ifo_fail('invalid');end if;
    -- Never wait for the inverse org -> membership order of generic R03 while
    -- already holding analysis/queue/source/membership locks. Reentrant after.
    perform 1 from public.organizations where id=a.organization_id for update skip locked;
    if not found then perform private.ifo_fail('stale');end if;
    perform 1 from public.memberships where organization_id=a.organization_id and user_id=a.owner_id
      and role in ('owner','admin','member') for update skip locked;
    if not found then perform private.ifo_fail('stale');end if;
    budget:=public.ai_budget_claim_dispatch(a.owner_id,a.organization_id,req);
    if budget->>'status'='denied' then
      return jsonb_build_object('schemaVersion','initial-feedback-claim/1.0.0','operation',private.ifo_operation_json(op),'attempt',private.ifo_attempt_json(att),
        'dispatchAllowed',false,'replayed',false,'budget',budget);
    end if;
    -- An existing generic winner cannot be adopted as this still-reserved
    -- feedback attempt. Never mark/release that winner or dispatch a new one.
    if budget->>'status' is distinct from 'ok' or budget->'receipt'->'dispatchAllowed' is distinct from 'true'::jsonb then perform private.ifo_fail('stale');end if;
    if budget->'receipt'->>'attemptId' is distinct from att.attempt_id::text or budget->'receipt'->>'operationId' is distinct from op.operation_id::text
      or budget->'receipt'->'replayed' is distinct from 'false'::jsonb then perform private.ifo_fail('stale');end if;
  end if;
  if not private.lock_analysis_stage_lease(a.id,p_lease->>'workerId',(p_lease->>'attempt')::integer,(p_lease->>'leaseFence')::uuid)
    or private.try_candidate_privacy_write(a.owner_id) is not true then perform private.ifo_fail('stale');end if;
  update private.initial_feedback_attempts set state='dispatched',claim_lease=p_lease,claimed_at=clock_timestamp(),
    budget_attempt_id=case when budget is null then null else att.attempt_id end
    where attempt_id=att.attempt_id and state='reserved' returning * into att;
  if not found then perform private.ifo_fail('stale');end if;
  return jsonb_build_object('schemaVersion','initial-feedback-claim/1.0.0','operation',private.ifo_operation_json(op),'attempt',private.ifo_attempt_json(att),
    'dispatchAllowed',true,'replayed',false,'budget',budget);
end;

create function public.settle_initial_feedback_attempt(p_operation_id uuid,p_attempt_id uuid,p_binding_sha256 text,p_usage jsonb,p_error_code text) returns jsonb
language plpgsql volatile security definer set search_path='' as $ifo_18$
declare op private.initial_feedback_operations%rowtype;att private.initial_feedback_attempts%rowtype;
  actual jsonb:=nullif(p_usage,'null'::jsonb);replayed boolean;
begin
  if p_operation_id is null or p_attempt_id is null or p_binding_sha256 is null or p_binding_sha256 collate "C" !~ '^[a-f0-9]{64}$'
    or private.ifo_usage_valid(actual) is not true or (p_error_code is not null and p_error_code not in
      ('provider_failed','timeout','cancelled','invalid_output','unsupported_output','storage_failed','unknown')) then perform private.ifo_fail('invalid');end if;
  if private.initial_feedback_operation_ready() is not true then perform private.ifo_fail('unavailable');end if;
  -- No current candidate/auth/source/lease/policy-selection read. Static exact
  -- metadata remains settleable after withdrawal; this NEVER calls R03 settle.
  select * into op from private.initial_feedback_operations where operation_id=p_operation_id for update;
  if not found or op.binding_sha256<>p_binding_sha256 then perform private.ifo_fail('not_found');end if;
  select * into att from private.initial_feedback_attempts where attempt_id=p_attempt_id and operation_id=op.operation_id for update;
  if not found or att.claimed_at is null or att.claim_lease is null or att.state='reserved' then perform private.ifo_fail('stale');end if;
  if (att.usage is not null and actual is not null and att.usage is distinct from actual)
    or (att.error_code is not null and p_error_code is not null and att.error_code<>p_error_code) then perform private.ifo_fail('stale');end if;
  replayed:=(actual is null or att.usage is not distinct from actual) and (p_error_code is null or att.error_code is not distinct from p_error_code)
    and att.state<>'dispatched';
  if not replayed then
    update private.initial_feedback_attempts set usage=coalesce(usage,actual),error_code=coalesce(error_code,p_error_code),
      state=case when coalesce(usage,actual) is null then 'held_unknown' when coalesce(error_code,p_error_code) is not null then 'failed' else 'known' end
      where attempt_id=att.attempt_id returning * into att;
  end if;
  return jsonb_build_object('schemaVersion','initial-feedback-settlement/1.0.0','operationId',op.operation_id,'attemptId',att.attempt_id,
    'bindingSha256',op.binding_sha256,'state',att.state,'usage',att.usage,'errorCode',att.error_code,'replayed',replayed);
end;
