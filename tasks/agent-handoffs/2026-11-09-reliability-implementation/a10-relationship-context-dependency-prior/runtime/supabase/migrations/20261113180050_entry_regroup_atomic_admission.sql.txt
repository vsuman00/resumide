create function private.entry_regroup_base_data(p_scope jsonb,p_base jsonb) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_base_data$
declare v public.canonical_resume_versions%rowtype; checked public.canonical_resume_versions%rowtype; evidence public.document_evidence_attempts%rowtype;
  k text; fact jsonb; section jsonb; entry jsonb; bullet jsonb; change jsonb; citation jsonb; observation jsonb; facts jsonb; ids text[]; targets text[];
  adapter jsonb; idx jsonb; quote_index jsonb:='{}'; signature text; depth integer:=0; visited uuid[]:='{}'; previous_revision integer;
  total_facts bigint:=0; total_refs bigint:=0; total_text bigint:=0; doc_text bigint;
  layers jsonb[]:='{}'; origin_json jsonb; prior_json jsonb; layer integer; matches bigint;
begin
  if private.grouped_import_keys(p_scope,array['ownerId','organizationId','resumeId','analysisId','sourceId','attemptId','inputSha256']) is not true
    or private.grouped_import_keys(p_base,array['versionId','revision','contentSha256','locale']) is not true then perform private.correction_fail('invalid'); end if;
  foreach k in array array['ownerId','organizationId','resumeId','analysisId','sourceId','attemptId'] loop
    if private.grouped_import_uuid(p_scope->k) is not true then perform private.correction_fail('invalid'); end if;
  end loop;
  if private.grouped_import_uuid(p_base->'versionId') is not true or private.evidence_integer(p_base->'revision',1,2147483647) is not true
    or jsonb_typeof(p_scope->'inputSha256') is distinct from 'string' or (p_scope->>'inputSha256') collate "C" !~ '^[a-f0-9]{64}$'
    or jsonb_typeof(p_base->'contentSha256') is distinct from 'string' or (p_base->>'contentSha256') collate "C" !~ '^[a-f0-9]{64}$'
    or jsonb_typeof(p_base->'locale') is distinct from 'string' or (p_base->>'locale') collate "C" !~ '^[a-z]{2,3}(?:-[A-Z]{2})?$' then perform private.correction_fail('invalid'); end if;
  select * into v from public.canonical_resume_versions where id=(p_base->>'versionId')::uuid;
  if not found or v.revision<>(p_base->>'revision')::numeric::integer or v.locale is distinct from p_base->>'locale'
    or encode(extensions.digest(convert_to(v.content::text,'UTF8'),'sha256'),'hex') is distinct from p_base->>'contentSha256' then perform private.correction_fail('invalid'); end if;
  select * into evidence from public.document_evidence_attempts where id=(p_scope->>'attemptId')::uuid
    and (owner_id,organization_id,analysis_id,resume_version_id,input_sha256)=
      ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,(p_scope->>'analysisId')::uuid,(p_scope->>'sourceId')::uuid,p_scope->>'inputSha256');
  if not found then perform private.correction_fail('invalid'); end if;
  adapter:=p_scope||jsonb_build_object('evidenceSchemaVersion','evidence/2.0.0','fieldExtractorVersion','evidence-fields/1.0.1',
    'targetModelVersion','evidence-targets/1.0.0','associationModelVersion','evidence-entry-associations/1.0.0','producerVersion','canonical-engine/1.0.3',
    'admissionPolicyVersion','grouped-original-admission/1.0.0','structuralTextProfileVersion','a02-structural-text/1.0.0',
    'correctionProjectionModelVersion','canonical-corrections/1.0.1','correctionAdmissionPolicyVersion','candidate-correction-admission/1.0.0','effectiveLocale',v.locale);
  idx:=private.grouped_import_source_index(evidence.envelope,adapter);
  for observation in select value from jsonb_each(idx->'observations') loop
    select jsonb_agg(jsonb_build_array(c.value->>'attemptId',c.value->>'spanId',(c.value->>'start')::numeric::integer,(c.value->>'end')::numeric::integer)
      order by convert_to(c.value->>'attemptId','UTF8'),convert_to(c.value->>'spanId','UTF8'),(c.value->>'start')::numeric::integer,(c.value->>'end')::numeric::integer)::text
      into signature from jsonb_array_elements(observation->'citations') c(value);
    signature:=jsonb_build_array(observation->>'field',signature::jsonb)::text;
    quote_index:=quote_index||jsonb_build_object(signature,coalesce(quote_index->signature,'[]'::jsonb)||jsonb_build_array(observation->'value'));
  end loop;
  checked:=v;
  loop
    if checked.id=any(visited) or (checked.owner_id,checked.organization_id,checked.resume_id,checked.source_resume_version_id,checked.source_evidence_version_id)
      is distinct from ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,(p_scope->>'resumeId')::uuid,(p_scope->>'sourceId')::uuid,(p_scope->>'attemptId')::uuid)
      or checked.content->>'producerVersion' not in ('canonical-engine/1.0.0','canonical-engine/1.0.1','canonical-engine/1.0.2','canonical-engine/1.0.3','canonical-engine/1.0.4')
      or private.canonical_shape_valid(case when checked.content->>'producerVersion' in ('canonical-engine/1.0.3','canonical-engine/1.0.4')
        then jsonb_set(checked.content,'{producerVersion}','"canonical-engine/1.0.1"'::jsonb) else checked.content end) is not true
      or octet_length(checked.content::text)>1048576 or private.grouped_import_utf16_length(checked.content->>'retentionPolicyId')>100
      or checked.content->'parentVersionId' is distinct from coalesce(to_jsonb(checked.parent_version_id),'null'::jsonb)
      or (checked.content->>'versionId',checked.content->>'ownerId',checked.content->>'organizationId',checked.content->>'resumeId',checked.content->>'analysisId',
          checked.content->>'sourceEvidenceVersionId',checked.content->>'inputSha256',checked.content->>'locale',checked.content->>'state')
        is distinct from (checked.id::text,p_scope->>'ownerId',p_scope->>'organizationId',p_scope->>'resumeId',p_scope->>'analysisId',p_scope->>'attemptId',p_scope->>'inputSha256',checked.locale,checked.state) then perform private.correction_fail('invalid'); end if;
    visited:=array_append(visited,checked.id);
    if checked.parent_version_id=any(visited) then perform private.correction_fail('invalid'); end if;
    total_facts:=total_facts+jsonb_array_length(checked.content->'facts');
    if total_facts>20000 then perform private.correction_fail('invalid'); end if;
    -- Existing metadata proof checks an actual .3 origin; shape adaptation alone grants nothing.
    if (select count(*) from public.resume_facts where canonical_version_id=checked.id)<>jsonb_array_length(checked.content->'facts')
      or exists(select 1 from jsonb_array_elements(checked.content->'facts') j(value)
        left join public.resume_facts f on f.canonical_version_id=checked.id and f.id=(j.value->>'id')::uuid
        where f.id is null or (f.owner_id,f.organization_id) is distinct from (checked.owner_id,checked.organization_id)
          or jsonb_build_object('id',f.id,'type',f.type,'value',f.value,'state',f.state,'sources',f.sources,'confirmation',f.confirmation) is distinct from j.value) then perform private.correction_fail('invalid'); end if;
    facts:='{}'; ids:=array[checked.resume_id::text,checked.id::text]; doc_text:=0;
    for fact in select value from jsonb_array_elements(checked.content->'facts') loop
      ids:=array_append(ids,fact->>'id'); facts:=facts||jsonb_build_object(fact->>'id',fact);
      doc_text:=doc_text+private.grouped_import_utf16_length(fact->>'value'); total_refs:=total_refs+jsonb_array_length(fact->'sources');
      if private.grouped_import_utf16_length(fact->>'value')>2000 or fact->>'value' is distinct from private.grouped_import_trim(fact->>'value')
        or (fact->>'state'='source_supported' and jsonb_array_length(fact->'sources')=0)
        or (fact->>'state'='candidate_confirmed') is distinct from (fact->'confirmation'<>'null'::jsonb)
        or (fact->'confirmation'<>'null'::jsonb and ((fact->'confirmation'->>'actorId',fact->'confirmation'->>'value') is distinct from (checked.owner_id::text,fact->>'value')
          or private.grouped_import_utf16_length(fact->'confirmation'->>'value')>2000)) then perform private.correction_fail('invalid'); end if;
      for citation in select value from jsonb_array_elements(fact->'sources') loop
        if citation->>'attemptId' is distinct from checked.source_evidence_version_id::text
          or private.grouped_import_utf16_length(citation->>'spanId')>100
          or private.evidence_integer(citation->'start',0,200000) is not true or private.evidence_integer(citation->'end',1,200000) is not true
          or (citation->>'end')::numeric::integer<=(citation->>'start')::numeric::integer then perform private.correction_fail('invalid'); end if;
      end loop;
      if fact->>'type' in ('name','email','phone','employer','role','dates','education','certification') and jsonb_array_length(fact->'sources')>0 then
        select jsonb_agg(jsonb_build_array(c.value->>'attemptId',c.value->>'spanId',(c.value->>'start')::numeric::integer,(c.value->>'end')::numeric::integer)
          order by convert_to(c.value->>'attemptId','UTF8'),convert_to(c.value->>'spanId','UTF8'),(c.value->>'start')::numeric::integer,(c.value->>'end')::numeric::integer)::text
          into signature from jsonb_array_elements(fact->'sources') c(value);
        signature:=jsonb_build_array(fact->>'type',signature::jsonb)::text;
        if quote_index ? signature and not exists(select 1 from jsonb_array_elements(quote_index->signature) q(value) where q.value=fact->'value') then perform private.correction_fail('invalid'); end if;
      end if;
    end loop;
    perform private.grouped_import_check_refs(checked.content->'contactFactIds',facts); total_refs:=total_refs+jsonb_array_length(checked.content->'contactFactIds');
    if exists(select 1 from jsonb_array_elements_text(checked.content->'contactFactIds') r(id) where facts->r.id->>'type' not in ('name','email','phone','link','location')) then perform private.correction_fail('invalid'); end if;
    for section in select value from jsonb_array_elements(checked.content->'sections') loop
      ids:=array_append(ids,section->>'id'); perform private.grouped_import_check_refs(section->'factIds',facts); total_refs:=total_refs+jsonb_array_length(section->'factIds');
      doc_text:=doc_text+private.grouped_import_utf16_length(section->>'title');
      if private.grouped_import_utf16_length(private.grouped_import_trim(section->>'title')) not between 1 and 100 then perform private.correction_fail('invalid'); end if;
      for entry in select value from jsonb_array_elements(section->'entries') loop
        ids:=array_append(ids,entry->>'id'); perform private.grouped_import_check_refs(entry->'factIds',facts); total_refs:=total_refs+jsonb_array_length(entry->'factIds');
        for bullet in select value from jsonb_array_elements(entry->'bullets') loop
          ids:=array_append(ids,bullet->>'id'); perform private.grouped_import_check_refs(bullet->'factIds',facts); total_refs:=total_refs+jsonb_array_length(bullet->'factIds');
          doc_text:=doc_text+private.grouped_import_utf16_length(bullet->>'text');
          if private.grouped_import_utf16_length(bullet->>'text')>2000 or bullet->>'text' is distinct from private.grouped_import_trim(bullet->>'text') then perform private.correction_fail('invalid'); end if;
        end loop;
      end loop;
    end loop;
    targets:=ids; total_text:=total_text+doc_text;
    for change in select value from jsonb_array_elements(checked.content->'changes') loop
      ids:=array_append(ids,change->>'id');
      if change->>'actorId' is distinct from checked.owner_id::text or not (change->>'targetId'=any(targets)) then perform private.correction_fail('invalid'); end if;
      if change->'priorValue'<>'null'::jsonb then
        total_text:=total_text+private.grouped_import_utf16_length(change->>'priorValue');
        if private.grouped_import_utf16_length(change->>'priorValue')>2000 or change->>'priorValue' is distinct from private.grouped_import_trim(change->>'priorValue') then perform private.correction_fail('invalid'); end if;
      end if;
    end loop;
    if cardinality(ids)<>(select count(distinct x) from unnest(ids) x) or doc_text>200000 or total_text>2000000 or total_refs>20000 then perform private.correction_fail('invalid'); end if;
    layers:=array_append(layers,facts);
    -- A bounded suffix may stop before its root, exactly as the Node projection permits.
    if checked.parent_version_id is null or depth=64 then exit; end if;
    previous_revision:=checked.revision;
    select * into checked from public.canonical_resume_versions where id=checked.parent_version_id;
    if not found or checked.revision<>previous_revision-1 then perform private.correction_fail('invalid'); end if;
    depth:=depth+1;
  end loop;
  -- Match the existing source-choice reader's additional reference budget, including unselected choices.
  for fact in select value from jsonb_array_elements(v.content->'facts') loop
    if fact->>'type' not in ('name','email','phone','employer','role','dates','education','certification') then continue; end if;
    origin_json:=fact;
    if jsonb_array_length(fact->'sources')=0 then
      origin_json:=null;
      for layer in 2..array_length(layers,1) loop
        prior_json:=layers[layer]->(fact->>'id');
        if prior_json is null or prior_json->>'type' is distinct from fact->>'type' then exit; end if;
        if jsonb_array_length(prior_json->'sources')>0 then origin_json:=prior_json; exit; end if;
      end loop;
    end if;
    if origin_json is null then continue; end if;
    select jsonb_agg(jsonb_build_array(c.value->>'attemptId',c.value->>'spanId',(c.value->>'start')::numeric::integer,(c.value->>'end')::numeric::integer)
      order by convert_to(c.value->>'attemptId','UTF8'),convert_to(c.value->>'spanId','UTF8'),(c.value->>'start')::numeric::integer,(c.value->>'end')::numeric::integer)::text
      into signature from jsonb_array_elements(origin_json->'sources') c(value);
    signature:=jsonb_build_array(origin_json->>'type',signature::jsonb)::text;
    select count(*) into matches from jsonb_array_elements(coalesce(quote_index->signature,'[]'::jsonb)) q(value) where q.value=origin_json->'value';
    total_refs:=total_refs+matches*(1+jsonb_array_length(private.correction_location(v.content,(fact->>'id')::uuid)));
    if total_refs>20000 then perform private.correction_fail('invalid'); end if;
  end loop;
  return jsonb_build_object('versionId',v.id,'revision',v.revision,'contentSha256',p_base->'contentSha256','locale',v.locale,'content',v.content,'envelope',evidence.envelope);
end;
$erg_entry_regroup_base_data$;

create function private.entry_regroup_resolve_loaded(p_scope jsonb,p_base jsonb,p_selections jsonb) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_resolve_loaded$
declare base jsonb; adapter jsonb; projected jsonb; requested jsonb:='[]'; canonical jsonb; selection jsonb; member jsonb; citation jsonb; src jsonb; assertion jsonb;
  output jsonb:='[]'; members jsonb; actual_member jsonb; actual_group jsonb; selected_ids jsonb:='{}'; seen_assertions jsonb:='{}'; used_entries jsonb:='{}';
  fact public.resume_facts%rowtype; origin_fact public.resume_facts%rowtype; version public.canonical_resume_versions%rowtype; previous_revision integer;
  location jsonb; places jsonb; source_refs jsonb; original_version uuid; through text; visited uuid[]; depth integer; k text;
  section jsonb; entry jsonb; bullet jsonb; ref text; fact_index jsonb; group_facts jsonb; affected jsonb; anchor text; section_id text;
  affected_entry jsonb; direct_refs jsonb; bullets jsonb; plans jsonb:='{}'; sections jsonb:='[]'; new_entries jsonb; retired jsonb:='[]'; index_value integer; insertion integer;
begin
  if private.grouped_import_keys(p_base,array['versionId','revision','contentSha256','locale','content','envelope']) is not true
    or jsonb_typeof(p_selections) is distinct from 'array' then perform private.correction_fail('invalid'); end if;
  if jsonb_array_length(p_selections) not between 1 and 32 then perform private.correction_fail('invalid'); end if;
  -- Only checked internal callers possess this loaded base. No proof/normalizer reentry.
  base:=p_base;
  adapter:=p_scope||jsonb_build_object('evidenceSchemaVersion','evidence/2.0.0','fieldExtractorVersion','evidence-fields/1.0.1',
    'targetModelVersion','evidence-targets/1.0.0','associationModelVersion','evidence-entry-associations/1.0.0','producerVersion','canonical-engine/1.0.3',
    'admissionPolicyVersion','grouped-original-admission/1.0.0','structuralTextProfileVersion','a02-structural-text/1.0.0',
    'correctionProjectionModelVersion','canonical-corrections/1.0.1','correctionAdmissionPolicyVersion','candidate-correction-admission/1.0.0','effectiveLocale',base->'locale');
  projected:=private.grouped_import_project_candidates(base->'envelope',adapter);
  select jsonb_object_agg(value->>'id',value) into fact_index from jsonb_array_elements(base->'content'->'facts');
  for selection in select value from jsonb_array_elements(p_selections) loop
    if private.grouped_import_keys(selection,array['pageId','method','sectionId','anchorEntryId','insertionIndex','affectedEntryIds','members']) is not true
      or jsonb_typeof(selection->'members') is distinct from 'array' then perform private.correction_fail('invalid'); end if;
    if jsonb_array_length(selection->'members') not between 2 and 3 then perform private.correction_fail('invalid'); end if;
    members:='[]';
    for member in select value from jsonb_array_elements(selection->'members') loop
      if private.grouped_import_keys(member,array['assertionId','field','factId','citations','mappedThrough','originVersionId','location']) is not true
        or private.grouped_import_keys(member->'location',array['kind','sectionId','entryId','bulletId']) is not true
        or jsonb_typeof(member->'citations') is distinct from 'array' then perform private.correction_fail('invalid'); end if;
      if jsonb_array_length(member->'citations')<>1 or private.grouped_import_uuid(member->'factId') is not true then perform private.correction_fail('invalid'); end if;
      citation:=member->'citations'->0;
      if private.grouped_import_keys(citation,array['attemptId','spanId','pageId','method','start','end','pageStart','pageEnd']) is not true then perform private.correction_fail('invalid'); end if;
      members:=members||jsonb_build_array(jsonb_build_object('assertionId',member->'assertionId','field',member->'field','citations',member->'citations'));
    end loop;
    requested:=requested||jsonb_build_array(jsonb_build_object('pageId',selection->'pageId','method',selection->'method','members',members));
  end loop;
  select coalesce(jsonb_agg(value order by ordinality),'[]'::jsonb) into canonical from jsonb_array_elements(projected) with ordinality
    where value in (select value from jsonb_array_elements(requested));
  if canonical is distinct from requested then perform private.correction_fail('invalid'); end if;
  for selection in select value from jsonb_array_elements(p_selections) loop
    members:='[]'; group_facts:='{}'; section_id:=null; anchor:=null;
    for member in select value from jsonb_array_elements(selection->'members') loop
      if selected_ids ? (member->>'factId') or seen_assertions ? (member->>'assertionId') then perform private.correction_fail('invalid'); end if;
      selected_ids:=selected_ids||jsonb_build_object(member->>'factId',true); seen_assertions:=seen_assertions||jsonb_build_object(member->>'assertionId',true);
      select * into fact from public.resume_facts where canonical_version_id=(base->>'versionId')::uuid and id=(member->>'factId')::uuid;
      if not found or (fact.owner_id,fact.organization_id,fact.type) is distinct from ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,member->>'field') then perform private.correction_fail('invalid'); end if;
      places:=private.correction_location(base->'content',fact.id);
      if jsonb_array_length(places)<>1 then perform private.correction_fail('invalid'); end if;
      location:=places->0;
      if location->>'kind' not in ('entry','bullet') or location->'sectionId'='null'::jsonb or location->'entryId'='null'::jsonb then perform private.correction_fail('invalid'); end if;
      if section_id is null then section_id:=location->>'sectionId'; elsif section_id<>location->>'sectionId' then perform private.correction_fail('invalid'); end if;
      if member->>'field'='employer' then anchor:=location->>'entryId'; end if;
      select value into section from jsonb_array_elements(base->'content'->'sections') where value->>'id'=section_id;
      if not found or section->>'type'<>'experience' then perform private.correction_fail('invalid'); end if;
      origin_fact:=fact; original_version:=null; through:='current_citations';
      if jsonb_array_length(origin_fact.sources)=0 then
        through:='owned_ancestor_chain'; depth:=0; visited:='{}';
        select * into version from public.canonical_resume_versions where id=(base->>'versionId')::uuid;
        loop
          if version.parent_version_id is null or depth>=64 or version.id=any(visited) then perform private.correction_fail('invalid'); end if;
          visited:=array_append(visited,version.id); previous_revision:=version.revision;
          select * into version from public.canonical_resume_versions where id=version.parent_version_id
            and (owner_id,organization_id,resume_id,source_resume_version_id,source_evidence_version_id)=
              ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,(p_scope->>'resumeId')::uuid,(p_scope->>'sourceId')::uuid,(p_scope->>'attemptId')::uuid)
            and content->>'analysisId'=p_scope->>'analysisId' and content->>'inputSha256'=p_scope->>'inputSha256';
          if not found or version.revision<>previous_revision-1
            or version.content->>'versionId' is distinct from version.id::text
            or version.content->'parentVersionId' is distinct from coalesce(to_jsonb(version.parent_version_id),'null'::jsonb) then perform private.correction_fail('invalid'); end if;
          select * into origin_fact from public.resume_facts where canonical_version_id=version.id and id=fact.id;
          if not found or (origin_fact.owner_id,origin_fact.organization_id,origin_fact.type) is distinct from (fact.owner_id,fact.organization_id,fact.type) then perform private.correction_fail('invalid'); end if;
          if not exists(select 1 from jsonb_array_elements(version.content->'facts') j(value) where j.value=jsonb_build_object('id',origin_fact.id,
            'type',origin_fact.type,'value',origin_fact.value,'state',origin_fact.state,'sources',origin_fact.sources,'confirmation',origin_fact.confirmation)) then perform private.correction_fail('invalid'); end if;
          depth:=depth+1;
          if jsonb_array_length(origin_fact.sources)>0 then original_version:=version.id; exit; end if;
        end loop;
      end if;
      select value into assertion from jsonb_array_elements(base->'envelope'->'assertions') where value->>'id'=member->>'assertionId';
      -- Selectable groups have exactly one citation, so no collation-dependent multireference ordering is introduced.
      select jsonb_agg(jsonb_build_object('attemptId',p_scope->'attemptId','spanId',value->'spanId','start',value->'start','end',value->'end') order by ordinality)
        into source_refs from jsonb_array_elements(assertion->'evidence') with ordinality;
      if origin_fact.value is distinct from assertion->>'value' or origin_fact.type is distinct from assertion->>'field'
        or origin_fact.sources is distinct from source_refs then perform private.correction_fail('invalid'); end if;
      actual_member:=jsonb_build_object('assertionId',member->'assertionId','field',member->'field','factId',fact.id,'citations',member->'citations',
        'mappedThrough',through,'originVersionId',original_version,'location',location);
      if actual_member is distinct from member then perform private.correction_fail('invalid'); end if;
      members:=members||jsonb_build_array(actual_member); group_facts:=group_facts||jsonb_build_object(fact.id::text,true);
    end loop;
    if anchor is null then perform private.correction_fail('invalid'); end if;
    select coalesce(jsonb_agg(e.value->'id' order by e.ordinality),'[]'::jsonb),min(e.ordinality)::integer-1 into affected,insertion
      from jsonb_array_elements(section->'entries') with ordinality e(value,ordinality)
      where e.value->>'id' in (select value->'location'->>'entryId' from jsonb_array_elements(members));
    direct_refs:='[]'; bullets:='[]';
    for entry in select e.value from jsonb_array_elements(section->'entries') with ordinality e(value,ordinality)
      where e.value->'id' in (select value from jsonb_array_elements(affected)) order by e.ordinality loop
      if used_entries ? (entry->>'id') then perform private.correction_fail('invalid'); end if;
      used_entries:=used_entries||jsonb_build_object(entry->>'id',true);
      for ref in select jsonb_array_elements_text(entry->'factIds') union all
        select jsonb_array_elements_text(b.value->'factIds') from jsonb_array_elements(entry->'bullets') b(value) loop
        if fact_index->ref->>'type' in ('employer','role','dates') and not (group_facts ? ref) then perform private.correction_fail('invalid'); end if;
      end loop;
      direct_refs:=direct_refs||(entry->'factIds'); bullets:=bullets||(entry->'bullets');
      if entry->>'id'<>anchor then retired:=retired||jsonb_build_array(entry->'id'); end if;
    end loop;
    if jsonb_array_length(direct_refs)>100 or jsonb_array_length(bullets)>100 then perform private.correction_fail('invalid'); end if;
    perform private.grouped_import_check_refs(direct_refs,fact_index);
    actual_group:=jsonb_build_object('pageId',selection->'pageId','method',selection->'method','sectionId',section_id,'anchorEntryId',anchor,
      'insertionIndex',insertion,'affectedEntryIds',affected,'members',members);
    if actual_group is distinct from selection then perform private.correction_fail('invalid'); end if;
    output:=output||jsonb_build_array(actual_group);
    plans:=plans||jsonb_build_object(section_id||':'||insertion,jsonb_build_object('id',anchor,'factIds',direct_refs,'bullets',bullets));
  end loop;
  for section in select value from jsonb_array_elements(base->'content'->'sections') loop
    new_entries:='[]'; index_value:=0;
    for entry in select value from jsonb_array_elements(section->'entries') loop
      affected_entry:=plans->((section->>'id')||':'||index_value);
      if affected_entry is not null then new_entries:=new_entries||jsonb_build_array(affected_entry);
      elsif not (used_entries ? (entry->>'id')) then new_entries:=new_entries||jsonb_build_array(entry); end if;
      index_value:=index_value+1;
    end loop;
    sections:=sections||jsonb_build_array(jsonb_set(section,'{entries}',new_entries));
  end loop;
  return jsonb_build_object('selections',output,'afterSections',sections,'retiredEntryIds',retired);
end;
$erg_entry_regroup_resolve_loaded$;

create function private.entry_regroup_normalize_loaded(p_scope jsonb,p_command jsonb,p_loaded jsonb) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_normalize_loaded$
declare normal jsonb; base jsonb; expected jsonb; binding jsonb; resolved jsonb; prefix text; revision_value integer; head uuid; n bigint; bad boolean; k text;
begin
  if private.grouped_import_keys(p_command,array['normalForm','previewHash','idempotencyKey']) is not true
    or private.grouped_import_uuid(p_command->'idempotencyKey') is not true
    or jsonb_typeof(p_command->'previewHash') is distinct from 'string' or (p_command->>'previewHash') collate "C" !~ '^[a-f0-9]{64}$' then perform private.correction_fail('invalid'); end if;
  normal:=p_command->'normalForm';
  if private.grouped_import_keys(normal,array['schemaVersion','operation','binding','base','expected','acknowledgementPolicy','acknowledgement','unselectedDisposition','selections']) is not true
    or normal->>'schemaVersion' is distinct from 'canonical-entry-regroup-command/1.0.0'
    or normal->>'operation' is distinct from 'approve_whole_entry_regroup'
    or normal->>'acknowledgementPolicy' is distinct from 'whole_entry_merge_not_fact_confirmation'
    or normal->'acknowledgement' is distinct from 'true'::jsonb
    or normal->>'unselectedDisposition' is distinct from 'retain_eligible_original_origins'
    or octet_length(normal::text)>262144 then perform private.correction_fail('invalid'); end if;
  base:=p_loaded;
  if private.evidence_integer(normal->'base'->'revision',1,2147483646) is not true then perform private.correction_fail('invalid'); end if;
  if (base-array['content','envelope']) is distinct from normal->'base' then perform private.correction_fail('invalid'); end if;
  binding:=p_scope||jsonb_build_object('evidenceSchemaVersion','evidence/2.0.0','fieldExtractorVersion','evidence-fields/1.0.1',
    'targetModelVersion','evidence-targets/1.0.0','associationModelVersion','evidence-entry-associations/1.0.0','producerVersion','canonical-engine/1.0.4',
    'admissionPolicyVersion','whole-entry-regroup-admission/1.0.0','structuralTextProfileVersion','a02-structural-text/1.0.0',
    'correctionProjectionModelVersion','canonical-corrections/1.0.1','correctionAdmissionPolicyVersion','candidate-correction-admission/1.0.0','regroupModelVersion','canonical-entry-regroup/1.0.0');
  if normal->'binding' is distinct from binding then perform private.correction_fail('invalid'); end if;
  for k in select jsonb_object_keys(binding) loop
    if jsonb_typeof(binding->k) is distinct from 'string' or private.grouped_import_utf16_length(binding->>k)>200 then perform private.correction_fail('invalid'); end if;
  end loop;
  expected:=normal->'expected';
  if private.grouped_import_keys(expected,array['reviewRevision','reviewHeadId','assignmentRevision','assignmentHeadId','associationRevision','associationHeadId','regroupRevision','regroupHeadId','canonicalHeadId']) is not true
    or expected->'canonicalHeadId' is distinct from base->'versionId' then perform private.correction_fail('invalid'); end if;
  -- Captures are immutable GLOBAL-analysis prefixes, not today's state and not attempt-filtered history.
  foreach prefix in array array['review','assignment','association','regroup'] loop
    if private.evidence_integer(expected->(prefix||'Revision'),0,2147483647) is not true then perform private.correction_fail('invalid'); end if;
    revision_value:=(expected->>(prefix||'Revision'))::numeric::integer;
    if prefix='review' and revision_value>1000 then perform private.correction_fail('invalid'); end if;
    if revision_value=0 then
      if expected->(prefix||'HeadId') is distinct from 'null'::jsonb then perform private.correction_fail('invalid'); end if;
    elsif private.grouped_import_uuid(expected->(prefix||'HeadId')) is not true then perform private.correction_fail('invalid'); end if;
    head:=null;
    if prefix='review' then
      select count(*) into n from public.evidence_corrections where analysis_id=(p_scope->>'analysisId')::uuid and revision<=revision_value;
      select id into head from public.evidence_corrections where analysis_id=(p_scope->>'analysisId')::uuid and revision=revision_value;
      select exists(select 1 from public.evidence_corrections e left join public.document_evidence_attempts a on a.id=e.attempt_id
        where e.analysis_id=(p_scope->>'analysisId')::uuid and e.revision<=revision_value and
        ((e.owner_id,e.organization_id,e.actor_id) is distinct from ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,(p_scope->>'ownerId')::uuid)
          or a.id is null or (a.owner_id,a.organization_id,a.analysis_id,a.resume_version_id,a.input_sha256) is distinct from
            ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,(p_scope->>'analysisId')::uuid,(p_scope->>'sourceId')::uuid,p_scope->>'inputSha256')
          or e.expected_revision<>e.revision-1 or e.field not in ('name','email','phone','employer','role','dates','education','certification')
          or e.operation not in ('confirm','correct') or private.grouped_import_utf16_length(e.value) not between 1 and 2000
          or e.value is distinct from private.grouped_import_trim(e.value)
          or (e.operation='confirm') is distinct from (e.candidate_id is not null)
          or (e.operation='confirm' and private.grouped_import_utf16_length(e.candidate_id) not between 1 and 100)
          or (e.operation='confirm' and e.attempt_id=(p_scope->>'attemptId')::uuid and not exists(
            select 1 from jsonb_array_elements(base->'envelope'->'assertions') a(value)
            where a.value->>'id'=e.candidate_id and a.value->>'field'=e.field and a.value->>'value'=e.value)))) into bad;
    elsif prefix='assignment' then
      select count(*) into n from public.candidate_correction_assignments where analysis_id=(p_scope->>'analysisId')::uuid and assignment_revision<=revision_value;
      select id into head from public.candidate_correction_assignments where analysis_id=(p_scope->>'analysisId')::uuid and assignment_revision=revision_value;
      select exists(select 1 from public.candidate_correction_assignments where analysis_id=(p_scope->>'analysisId')::uuid and assignment_revision<=revision_value
        and (owner_id,organization_id,resume_id,source_resume_version_id,input_sha256) is distinct from
          ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,(p_scope->>'resumeId')::uuid,(p_scope->>'sourceId')::uuid,p_scope->>'inputSha256')) into bad;
    elsif prefix='association' then
      select count(*) into n from public.candidate_entry_association_batches where analysis_id=(p_scope->>'analysisId')::uuid and association_revision<=revision_value;
      select id into head from public.candidate_entry_association_batches where analysis_id=(p_scope->>'analysisId')::uuid and association_revision=revision_value;
      select exists(select 1 from public.candidate_entry_association_batches where analysis_id=(p_scope->>'analysisId')::uuid and association_revision<=revision_value
        and (owner_id,organization_id,resume_id,source_resume_version_id,input_sha256) is distinct from
          ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,(p_scope->>'resumeId')::uuid,(p_scope->>'sourceId')::uuid,p_scope->>'inputSha256')) into bad;
    else
      select count(*) into n from public.candidate_entry_regroup_batches where analysis_id=(p_scope->>'analysisId')::uuid and regroup_revision<=revision_value;
      select id into head from public.candidate_entry_regroup_batches where analysis_id=(p_scope->>'analysisId')::uuid and regroup_revision=revision_value;
      select exists(select 1 from public.candidate_entry_regroup_batches where analysis_id=(p_scope->>'analysisId')::uuid and regroup_revision<=revision_value
        and (owner_id,organization_id,resume_id,source_resume_version_id,evidence_attempt_id,input_sha256) is distinct from
          ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,(p_scope->>'resumeId')::uuid,(p_scope->>'sourceId')::uuid,(p_scope->>'attemptId')::uuid,p_scope->>'inputSha256')) into bad;
    end if;
    if n<>revision_value or coalesce(to_jsonb(head),'null'::jsonb) is distinct from expected->(prefix||'HeadId') or bad then perform private.correction_fail('invalid'); end if;
  end loop;
  resolved:=private.entry_regroup_resolve_loaded(p_scope,base,normal->'selections');
  if resolved->'selections' is distinct from normal->'selections'
    or private.grouped_import_command_sha256(private.entry_regroup_atoms(normal)) is distinct from p_command->>'previewHash' then perform private.correction_fail('invalid'); end if;
  return normal;
end;

create function private.entry_regroup_inspect(p_cache jsonb,p_kind text,p_id text,p_value jsonb) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_inspect$
declare c jsonb:=private.entry_regroup_work(p_cache,1); cache_key text:=p_kind||':'||p_id; f jsonb; s jsonb; e jsonb; b jsonb; x jsonb;
  fact_count bigint:=0; refs bigint:=0; text_units bigint:=0; work bigint:=1; bytes bigint;
begin
  if p_value is null or p_id is null then perform private.correction_fail('unavailable'); end if;
  if c->'seen' ? cache_key then return c; end if;
  bytes:=octet_length(convert_to(p_value::text,'UTF8'));
  if (c->>'bytes')::bigint+bytes>33554432 or (p_kind='document' and bytes>1048576) or (p_kind='attempt' and bytes>16777216) or (p_kind='command' and bytes>262144) then perform private.correction_fail('unavailable'); end if;
  if p_kind='document' then
    if jsonb_typeof(p_value->'facts') is distinct from 'array' or jsonb_typeof(p_value->'sections') is distinct from 'array' then perform private.correction_fail('unavailable'); end if;
    c:=jsonb_set(c,'{documents}',to_jsonb((c->>'documents')::bigint+1));
    for f in select value from jsonb_array_elements(p_value->'facts') loop
      fact_count:=fact_count+1; refs:=refs+jsonb_array_length(f->'sources'); work:=work+1+jsonb_array_length(f->'sources');
      text_units:=text_units+private.grouped_import_utf16_length(f->>'value');
      if f->'confirmation'<>'null'::jsonb then text_units:=text_units+private.grouped_import_utf16_length(f->'confirmation'->>'value'); end if;
    end loop;
    refs:=refs+jsonb_array_length(p_value->'contactFactIds');
    for s in select value from jsonb_array_elements(p_value->'sections') loop
      work:=work+1; refs:=refs+jsonb_array_length(s->'factIds'); text_units:=text_units+private.grouped_import_utf16_length(s->>'title');
      for e in select value from jsonb_array_elements(s->'entries') loop
        work:=work+1; refs:=refs+jsonb_array_length(e->'factIds');
        for b in select value from jsonb_array_elements(e->'bullets') loop
          work:=work+1; refs:=refs+jsonb_array_length(b->'factIds'); text_units:=text_units+private.grouped_import_utf16_length(b->>'text');
        end loop;
      end loop;
    end loop;
    for x in select value from jsonb_array_elements(p_value->'changes') loop work:=work+1; text_units:=text_units+coalesce(private.grouped_import_utf16_length(x->>'priorValue'),0); end loop;
    work:=work+refs;
  elsif p_kind='attempt' then
    c:=jsonb_set(c,'{attempts}',to_jsonb((c->>'attempts')::bigint+1));
    for x in select value from jsonb_array_elements(p_value->'assertions') loop work:=work+1; refs:=refs+jsonb_array_length(x->'evidence'); text_units:=text_units+private.grouped_import_utf16_length(x->>'value'); end loop;
    for x in select value from jsonb_array_elements(p_value->'spans') loop work:=work+1; text_units:=text_units+private.grouped_import_utf16_length(x->>'text'); end loop;
    for x in select value from jsonb_array_elements(p_value->'nativeSource'->'pages') loop work:=work+1; text_units:=text_units+private.grouped_import_utf16_length(x->>'text'); end loop;
    work:=work+refs;
  elsif p_kind='command' then
    for s in select value from jsonb_array_elements(p_value->'normalForm'->'selections') loop
      work:=work+1;
      for x in select value from jsonb_array_elements(s->'members') loop work:=work+1; refs:=refs+jsonb_array_length(x->'citations'); end loop;
      refs:=refs+jsonb_array_length(s->'affectedEntryIds');
    end loop;
    work:=work+refs;
  elsif p_kind='review' then text_units:=private.grouped_import_utf16_length(p_value->>'value');
  end if;
  c:=jsonb_set(c,'{bytes}',to_jsonb((c->>'bytes')::bigint+bytes)); c:=jsonb_set(c,'{facts}',to_jsonb((c->>'facts')::bigint+fact_count));
  c:=jsonb_set(c,'{refs}',to_jsonb((c->>'refs')::bigint+refs)); c:=jsonb_set(c,'{text}',to_jsonb((c->>'text')::bigint+text_units));
  if (c->>'documents')::bigint>2048 or (c->>'attempts')::bigint>64 or (c->>'bytes')::bigint>33554432 or (c->>'facts')::bigint>200000
    or (c->>'refs')::bigint>400000 or (c->>'text')::bigint>8000000 then perform private.correction_fail('unavailable'); end if;
  c:=jsonb_set(c,'{seen}',(c->'seen')||jsonb_build_object(cache_key,true)); return private.entry_regroup_work(c,work);
end;
$erg_entry_regroup_inspect$;

create function private.entry_regroup_load_base(p_scope jsonb,p_key jsonb,p_cache jsonb) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_load_base$
declare c jsonb:=private.entry_regroup_work(p_cache,1); loaded jsonb; v public.canonical_resume_versions%rowtype; e public.document_evidence_attempts%rowtype;
  id_value uuid; seen uuid[]:='{}'; depth integer:=0; doc_refs bigint; doc_work bigint; key_value text:=p_key->>'versionId';
begin
  if c->'bases' ? key_value then
    loaded:=c->'bases'->key_value;
    if (loaded-array['content','envelope']) is distinct from p_key then perform private.correction_fail('invalid'); end if;
    return jsonb_build_object('data',loaded,'cache',c);
  end if;
  -- Account each actual raw-loader suffix visit; JSON/count summaries are charged once by identity.
  id_value:=(p_key->>'versionId')::uuid;
  loop
    select * into v from public.canonical_resume_versions where id=id_value;
    if not found or id_value=any(seen) then perform private.correction_fail('invalid'); end if;
    c:=private.entry_regroup_inspect(c,'document',v.id::text,v.content); seen:=array_append(seen,id_value);
    select jsonb_array_length(v.content->'facts')+coalesce(sum(jsonb_array_length(f.value->'sources')),0) into doc_work from jsonb_array_elements(v.content->'facts') f(value);
    -- Raw validation revisits facts, references, entries/bullets and source observations. Count these visits, not just unique JSON.
    doc_work:=doc_work+jsonb_array_length(v.content->'contactFactIds');
    select doc_work+coalesce(sum(1+jsonb_array_length(s.value->'factIds')),0) into doc_work from jsonb_array_elements(v.content->'sections') s(value);
    select doc_work+coalesce(sum(1+jsonb_array_length(x.value->'factIds')),0) into doc_work from jsonb_array_elements(v.content->'sections') s(value) cross join lateral jsonb_array_elements(s.value->'entries') x(value);
    select doc_work+coalesce(sum(1+jsonb_array_length(b.value->'factIds')),0) into doc_work from jsonb_array_elements(v.content->'sections') s(value) cross join lateral jsonb_array_elements(s.value->'entries') x(value) cross join lateral jsonb_array_elements(x.value->'bullets') b(value);
    c:=private.entry_regroup_work(c,doc_work+1);
    if v.parent_version_id is null or depth=64 then exit; end if;
    id_value:=v.parent_version_id; depth:=depth+1;
  end loop;
  select * into e from public.document_evidence_attempts where id=(p_scope->>'attemptId')::uuid;
  if not found then perform private.correction_fail('invalid'); end if;
  c:=private.entry_regroup_inspect(c,'attempt',e.id::text,e.envelope);
  c:=private.entry_regroup_work(c,1+jsonb_array_length(e.envelope->'assertions')+jsonb_array_length(e.envelope->'spans'));
  loaded:=private.entry_regroup_base_data(p_scope,p_key);
  c:=jsonb_set(c,'{bases}',(c->'bases')||jsonb_build_object(key_value,loaded));
  return jsonb_build_object('data',loaded,'cache',c);
end;
$erg_entry_regroup_load_base$;

create function private.entry_regroup_entry_signature(p_content jsonb,p_section uuid,p_entry uuid) returns jsonb
language plpgsql immutable security invoker set search_path='' as $erg_entry_regroup_entry_signature$
declare s jsonb; e jsonb; f jsonb; facts jsonb:='[]'; ref text; n integer;
begin
  select count(*),jsonb_agg(value)->0 into n,s from jsonb_array_elements(p_content->'sections') where value->>'id'=p_section::text;
  if n<>1 or s->>'type'<>'experience' then return null; end if;
  select count(*),jsonb_agg(value)->0 into n,e from jsonb_array_elements(s->'entries') where value->>'id'=p_entry::text;
  if n<>1 then return null; end if;
  for ref in select jsonb_array_elements_text(e->'factIds') union all
    select jsonb_array_elements_text(value->'factIds') from jsonb_array_elements(e->'bullets') loop
    select count(*),jsonb_agg(value)->0 into n,f from jsonb_array_elements(p_content->'facts') where value->>'id'=ref;
    if n<>1 then return null; end if;
    facts:=facts||jsonb_build_array(jsonb_build_object('id',f->'id','type',f->'type','value',f->'value','sources',f->'sources'));
  end loop;
  return jsonb_build_object('sectionId',p_section,'entry',e,'facts',facts);
end;

create function private.entry_regroup_links(p_version uuid) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_links$
declare result jsonb;
begin
  if exists(select 1 from public.canonical_entry_regroup_snapshots where canonical_version_id=p_version) then
    select coalesce(jsonb_agg(jsonb_build_object('sectionId',section_id,'entryId',entry_id,'originFamily',origin_family,
      'associationGroupId',association_group_id,'associationBatchId',association_batch_id,'regroupGroupId',regroup_group_id,'regroupBatchId',regroup_batch_id,
      'originalApprovalVersionId',original_approval_version_id) order by composition_order),'[]'::jsonb) into result
      from public.canonical_entry_regroup_links where canonical_version_id=p_version;
  else
    select coalesce(jsonb_agg(jsonb_build_object('sectionId',g.section_id,'entryId',g.entry_id,'originFamily','association',
      'associationGroupId',g.id,'associationBatchId',g.batch_id,'regroupGroupId',null,'regroupBatchId',null,'originalApprovalVersionId',g.result_version_id)
      order by s.ordinality,e.ordinality),'[]'::jsonb) into result
      from public.canonical_entry_association_links l join public.candidate_entry_association_groups g on g.id=l.group_id
      join public.canonical_resume_versions v on v.id=p_version
      cross join lateral jsonb_array_elements(v.content->'sections') with ordinality s(value,ordinality)
      cross join lateral jsonb_array_elements(s.value->'entries') with ordinality e(value,ordinality)
      where l.canonical_version_id=p_version and s.value->>'id'=g.section_id::text and e.value->>'id'=g.entry_id::text;
  end if;
  return result;
end;
$erg_entry_regroup_links$;

create function private.entry_regroup_group_active(p_version uuid,p_link jsonb) returns boolean
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_group_active$
declare v public.canonical_resume_versions%rowtype; origin public.canonical_resume_versions%rowtype; old_signature jsonb; current_signature jsonb;
begin
  select * into v from public.canonical_resume_versions where id=p_version;
  select * into origin from public.canonical_resume_versions where id=(p_link->>'originalApprovalVersionId')::uuid;
  if v.id is null or origin.id is null or (v.owner_id,v.organization_id,v.resume_id,v.source_resume_version_id,v.source_evidence_version_id)
    is distinct from (origin.owner_id,origin.organization_id,origin.resume_id,origin.source_resume_version_id,origin.source_evidence_version_id) then perform private.correction_fail('unavailable'); end if;
  old_signature:=private.entry_regroup_entry_signature(origin.content,(p_link->>'sectionId')::uuid,(p_link->>'entryId')::uuid);
  current_signature:=private.entry_regroup_entry_signature(v.content,(p_link->>'sectionId')::uuid,(p_link->>'entryId')::uuid);
  if old_signature is null then perform private.correction_fail('unavailable'); end if;
  return current_signature is not null and current_signature=old_signature;
end;
$erg_entry_regroup_group_active$;

create function private.entry_regroup_expected_links(p_base uuid,p_result uuid,p_batch uuid) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_expected_links$
declare links jsonb:='[]'; link jsonb; g public.candidate_entry_regroup_groups%rowtype; result jsonb; doc jsonb;
begin
  for link in select value from jsonb_array_elements(private.entry_regroup_links(p_base)) loop
    if not exists(select 1 from public.candidate_entry_regroup_groups x where x.batch_id=p_batch and x.section_id=(link->>'sectionId')::uuid
      and (link->>'entryId')::uuid=any(x.affected_entry_ids)) and private.entry_regroup_group_active(p_result,link) then links:=links||jsonb_build_array(link); end if;
  end loop;
  for g in select * from public.candidate_entry_regroup_groups where batch_id=p_batch order by group_order loop
    links:=links||jsonb_build_array(jsonb_build_object('sectionId',g.section_id,'entryId',g.anchor_entry_id,'originFamily','regroup','associationGroupId',null,
      'associationBatchId',null,'regroupGroupId',g.id,'regroupBatchId',g.batch_id,'originalApprovalVersionId',g.result_version_id));
  end loop;
  select content into strict doc from public.canonical_resume_versions where id=p_result;
  select coalesce(jsonb_agg(l.value order by s.ordinality,e.ordinality),'[]'::jsonb) into result
    from jsonb_array_elements(links) l(value)
    cross join jsonb_array_elements(doc->'sections') with ordinality s(value,ordinality)
    cross join lateral jsonb_array_elements(s.value->'entries') with ordinality e(value,ordinality)
    where s.value->'id'=l.value->'sectionId' and e.value->'id'=l.value->'entryId';
  if jsonb_array_length(result)<>jsonb_array_length(links) or jsonb_array_length(result)>1000
    or (select count(distinct (value->>'sectionId',value->>'entryId')) from jsonb_array_elements(result))<>jsonb_array_length(result) then perform private.correction_fail('unavailable'); end if;
  return result;
end;
$erg_entry_regroup_expected_links$;

create function private.entry_regroup_check_header(p_version uuid,p_cache jsonb) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_check_header$
declare c jsonb:=private.entry_regroup_work(p_cache,1); v public.canonical_resume_versions%rowtype; s public.canonical_entry_regroup_snapshots%rowtype;
  parent public.canonical_resume_versions%rowtype; link jsonb; expected jsonb:='[]'; actual jsonb; origin uuid; g public.candidate_entry_regroup_groups%rowtype;
  a public.candidate_entry_association_groups%rowtype; cache_key text; e public.document_evidence_attempts%rowtype; target uuid;
begin
  select * into strict v from public.canonical_resume_versions where id=p_version;
  c:=private.entry_regroup_inspect(c,'document',v.id::text,v.content);
  select * into strict e from public.document_evidence_attempts where id=v.source_evidence_version_id;
  if (v.owner_id,v.organization_id,v.source_resume_version_id,v.content->>'analysisId',v.content->>'inputSha256')
    is distinct from (e.owner_id,e.organization_id,e.resume_version_id,e.analysis_id::text,e.input_sha256) then perform private.correction_fail('unavailable'); end if;
  select * into s from public.canonical_entry_regroup_snapshots where canonical_version_id=v.id;
  if not found then
    if v.content->>'producerVersion'='canonical-engine/1.0.4' or exists(select 1 from public.canonical_entry_regroup_snapshots where canonical_version_id=v.parent_version_id)
      then perform private.correction_fail('unavailable'); end if;
    perform private.entry_association_validate_head(v.id); return c;
  end if;
  c:=private.entry_regroup_inspect(c,'header',v.id::text,to_jsonb(s));
  if (s.owner_id,s.organization_id,s.resume_id,s.analysis_id,s.source_resume_version_id,s.evidence_attempt_id,s.input_sha256)
    is distinct from (v.owner_id,v.organization_id,v.resume_id,e.analysis_id,v.source_resume_version_id,e.id,e.input_sha256) then perform private.correction_fail('unavailable'); end if;
  actual:=private.entry_regroup_links(v.id);
  if s.creation_kind='inherited' then
    if s.inherited_from_version_id is distinct from v.parent_version_id or v.command->>'operation'='rollback' then perform private.correction_fail('unavailable'); end if;
    for link in select value from jsonb_array_elements(private.entry_regroup_links(v.parent_version_id)) loop
      if private.entry_regroup_group_active(v.id,link) then expected:=expected||jsonb_build_array(link); end if;
    end loop;
    if actual is distinct from expected then perform private.correction_fail('unavailable'); end if;
  elsif s.creation_kind='restore' then
    target:=s.restored_from_version_id;
    select * into strict parent from public.canonical_resume_versions where id=target;
    if v.command->>'operation' is distinct from 'rollback' or v.command->>'targetVersionId' is distinct from target::text or parent.revision>=v.revision
      or (parent.owner_id,parent.organization_id,parent.resume_id,parent.source_resume_version_id,parent.source_evidence_version_id)
        is distinct from (v.owner_id,v.organization_id,v.resume_id,v.source_resume_version_id,v.source_evidence_version_id)
      or (v.content-array['versionId','parentVersionId','createdAt','state','changes']) is distinct from
        (parent.content-array['versionId','parentVersionId','createdAt','state','changes'])
      or actual is distinct from private.entry_regroup_links(target) then perform private.correction_fail('unavailable'); end if;
  elsif s.creation_kind='regroup' then
    if not (c->'origins' ? ('regroup:'||s.creating_regroup_batch_id::text)) then perform private.correction_fail('unavailable'); end if;
  else
    if v.content->>'producerVersion'='canonical-engine/1.0.4' then perform private.correction_fail('unavailable'); end if;
    perform private.entry_association_validate_head(v.id);
    if exists(select 1 from public.canonical_entry_regroup_links where canonical_version_id=v.id) then perform private.correction_fail('unavailable'); end if;
  end if;
  for link in select value from jsonb_array_elements(actual) loop
    c:=private.entry_regroup_work(c,1);
    if link->>'originFamily'='regroup' then
      select * into strict g from public.candidate_entry_regroup_groups where id=(link->>'regroupGroupId')::uuid;
      if not (c->'origins' ? ('regroup:'||g.batch_id::text)) or (g.batch_id,g.section_id,g.anchor_entry_id,g.result_version_id)
        is distinct from ((link->>'regroupBatchId')::uuid,(link->>'sectionId')::uuid,(link->>'entryId')::uuid,(link->>'originalApprovalVersionId')::uuid)
        then perform private.correction_fail('unavailable'); end if;
    else
      select * into strict a from public.candidate_entry_association_groups where id=(link->>'associationGroupId')::uuid;
      if (a.batch_id,a.section_id,a.entry_id,a.result_version_id) is distinct from
        ((link->>'associationBatchId')::uuid,(link->>'sectionId')::uuid,(link->>'entryId')::uuid,(link->>'originalApprovalVersionId')::uuid) then perform private.correction_fail('unavailable'); end if;
      cache_key:='association:'||a.batch_id::text;
      if not (c->'origins' ? cache_key) then
        perform private.entry_association_validate_head(a.result_version_id);
        c:=jsonb_set(c,'{origins}',(c->'origins')||jsonb_build_object(cache_key,true));
      end if;
    end if;
    if private.entry_regroup_group_active(v.id,link) is not true then perform private.correction_fail('unavailable'); end if;
  end loop;
  return c;
end;
$erg_entry_regroup_check_header$;

create function private.entry_regroup_origin_proof(p_scope jsonb,p_through integer,p_cache jsonb) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_origin_proof$
declare c jsonb:=p_cache; b public.candidate_entry_regroup_batches%rowtype; v public.canonical_resume_versions%rowtype;
  g public.candidate_entry_regroup_groups%rowtype; m public.candidate_entry_regroup_members%rowtype; prior uuid; n integer:=0; k text;
  loaded jsonb; plan jsonb; selection jsonb; member jsonb; data jsonb; actual jsonb; doc jsonb; bytes bytea; digest text; group_index integer; member_index integer;
  current_link jsonb; origin public.canonical_resume_versions%rowtype; e public.document_evidence_attempts%rowtype;
begin
  if p_through not between 0 and 1000 then perform private.correction_fail('unavailable'); end if;
  for b in select * from public.candidate_entry_regroup_batches where analysis_id=(p_scope->>'analysisId')::uuid and regroup_revision<=p_through order by regroup_revision loop
    c:=private.entry_regroup_work(c,1); n:=n+1;
    if b.regroup_revision<>n or b.previous_batch_id is distinct from prior or (b.owner_id,b.organization_id,b.resume_id,b.analysis_id,b.source_resume_version_id,b.evidence_attempt_id,b.input_sha256)
      is distinct from ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,(p_scope->>'resumeId')::uuid,(p_scope->>'analysisId')::uuid,(p_scope->>'sourceId')::uuid,(p_scope->>'attemptId')::uuid,p_scope->>'inputSha256')
      then perform private.correction_fail('unavailable'); end if;
    if c->'origins' ? ('regroup:'||b.id::text) then prior:=b.id; continue; end if;
    c:=private.entry_regroup_inspect(c,'command',b.idempotency_key::text,b.command);
    loaded:=private.entry_regroup_load_base(p_scope,b.command->'normalForm'->'base',c); c:=loaded->'cache'; data:=loaded->'data';
    c:=private.entry_regroup_check_header(b.base_version_id,c);
    if (data->>'versionId')::uuid<>b.base_version_id or (data->>'revision')::integer<>b.base_revision or data->>'contentSha256'<>b.base_content_sha256
      or b.command->'normalForm'->'expected'->>'regroupRevision'<>(n-1)::text
      or b.command->'normalForm'->'expected'->'regroupHeadId' is distinct from coalesce(to_jsonb(prior),'null'::jsonb)
      or b.command->>'idempotencyKey'<>b.idempotency_key::text then perform private.correction_fail('unavailable'); end if;
    actual:=private.entry_regroup_normalize_loaded(p_scope,b.command,data);
    bytes:=private.grouped_import_command_bytes(private.entry_regroup_atoms(actual)); digest:=private.grouped_import_command_sha256(private.entry_regroup_atoms(actual));
    if b.command_bytes is distinct from bytes or b.command_sha256 is distinct from digest or b.codec_version<>'a02-command-bytes/1.0.0' then perform private.correction_fail('unavailable'); end if;
    plan:=private.entry_regroup_resolve_loaded(p_scope,data,actual->'selections');
    select * into strict v from public.canonical_resume_versions where id=b.result_version_id;
    c:=private.entry_regroup_inspect(c,'document',v.id::text,v.content);
    if (v.owner_id,v.organization_id,v.resume_id,v.source_resume_version_id,v.source_evidence_version_id,v.parent_version_id,v.revision,v.idempotency_key)
      is distinct from (b.owner_id,b.organization_id,b.resume_id,b.source_resume_version_id,b.evidence_attempt_id,b.base_version_id,b.result_revision,b.storage_key)
      or b.result_revision<>b.base_revision+1 or v.command is distinct from jsonb_build_object('schemaVersion','entry-regroup-storage/1.0.0','operation','entry_regroup',
        'batchId',b.id,'semanticSha256',b.command_sha256,'idempotencyKey',b.storage_key)
      or encode(extensions.digest(convert_to(v.content::text,'UTF8'),'sha256'),'hex') is distinct from b.result_content_sha256 then perform private.correction_fail('unavailable'); end if;
    doc:=private.entry_regroup_child_body(data->'content',plan->'afterSections',v.id,(v.content->'changes'->0->>'id')::uuid,v.content->>'createdAt');
    if doc is distinct from v.content or private.canonical_shape_valid(doc) is not true then perform private.correction_fail('unavailable'); end if;
    if (select count(*) from public.resume_facts where canonical_version_id=v.id)<>jsonb_array_length(doc->'facts') or exists(
      select 1 from jsonb_array_elements(doc->'facts') j(value) left join public.resume_facts f on f.canonical_version_id=v.id and f.id=(j.value->>'id')::uuid
      where f.id is null or (f.owner_id,f.organization_id) is distinct from (v.owner_id,v.organization_id) or
        jsonb_build_object('id',f.id,'type',f.type,'value',f.value,'state',f.state,'sources',f.sources,'confirmation',f.confirmation) is distinct from j.value) then perform private.correction_fail('unavailable'); end if;
    if (select count(*) from public.candidate_entry_regroup_groups where batch_id=b.id)<>jsonb_array_length(actual->'selections') then perform private.correction_fail('unavailable'); end if;
    group_index:=0;
    for g in select * from public.candidate_entry_regroup_groups where batch_id=b.id order by group_order loop
      c:=private.entry_regroup_work(c,1); selection:=actual->'selections'->group_index; group_index:=group_index+1;
      if (g.page_id,g.method) is distinct from (selection->>'pageId',selection->>'method') or g.group_order<>group_index or (g.base_version_id,g.result_version_id,g.section_id,g.anchor_entry_id,g.insertion_index)
        is distinct from (b.base_version_id,b.result_version_id,(selection->>'sectionId')::uuid,(selection->>'anchorEntryId')::uuid,(selection->>'insertionIndex')::integer)
        or to_jsonb(g.affected_entry_ids) is distinct from selection->'affectedEntryIds'
        or g.source_selection is distinct from jsonb_build_object('pageId',selection->'pageId','method',selection->'method','members',
          (select jsonb_agg(value-array['factId','mappedThrough','originVersionId','location'] order by ordinality) from jsonb_array_elements(selection->'members') with ordinality)) then perform private.correction_fail('unavailable'); end if;
      if (select count(*) from public.candidate_entry_regroup_members where group_id=g.id)<>jsonb_array_length(selection->'members') then perform private.correction_fail('unavailable'); end if;
      member_index:=0;
      for m in select * from public.candidate_entry_regroup_members where group_id=g.id order by source_order loop
        c:=private.entry_regroup_work(c,1); member:=selection->'members'->member_index; member_index:=member_index+1;
        if m.source_order<>member_index or (m.assertion_id,m.field,m.fact_id,m.original_entry_id,m.location_kind,m.original_bullet_id,m.mapped_through,m.origin_version_id)
          is distinct from (member->>'assertionId',member->>'field',(member->>'factId')::uuid,(member->'location'->>'entryId')::uuid,
            member->'location'->>'kind',(member->'location'->>'bulletId')::uuid,member->>'mappedThrough',(member->>'originVersionId')::uuid)
          or m.citations is distinct from member->'citations' then perform private.correction_fail('unavailable'); end if;
      end loop;
    end loop;
    -- Every inherited regroup origin is earlier and already verified; no recursive proof call exists.
    for current_link in select value from jsonb_array_elements(private.entry_regroup_links(b.base_version_id)) loop
      if current_link->>'originFamily'='regroup' then
        select * into strict origin from public.canonical_resume_versions where id=(current_link->>'originalApprovalVersionId')::uuid;
        if origin.revision>=v.revision or not (c->'origins' ? ('regroup:'||(current_link->>'regroupBatchId'))) then perform private.correction_fail('unavailable'); end if;
      end if;
    end loop;
    if private.entry_regroup_links(v.id) is distinct from private.entry_regroup_expected_links(b.base_version_id,v.id,b.id) then perform private.correction_fail('unavailable'); end if;
    c:=jsonb_set(c,'{origins}',(c->'origins')||jsonb_build_object('regroup:'||b.id::text,true));
    c:=private.entry_regroup_check_header(v.id,c); prior:=b.id;
  end loop;
  if n<>p_through then perform private.correction_fail('unavailable'); end if;
  return c;
end;
$erg_entry_regroup_origin_proof$;

create function private.entry_regroup_scope(p_user_id uuid,p_analysis_id uuid,p_original_attempt uuid default null) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_scope$
declare r public.resumes%rowtype; a public.analyses%rowtype; v public.resume_versions%rowtype; e public.document_evidence_attempts%rowtype; initial_owner uuid;
begin
  select owner_id into initial_owner from public.analyses where id=p_analysis_id;
  if p_user_id is null or initial_owner is distinct from p_user_id then perform private.correction_fail('not_found'); end if;
  perform private.lock_candidate_owners(array[p_user_id]);
  if not private.try_candidate_privacy_write(p_user_id) then perform private.correction_fail('not_found'); end if;
  select root.* into r from public.resumes root join public.resume_versions src on src.resume_id=root.id join public.analyses x on x.resume_version_id=src.id
    where x.id=p_analysis_id and x.owner_id=p_user_id and src.owner_id=p_user_id and root.owner_id=p_user_id
      and root.organization_id=src.organization_id and src.organization_id=x.organization_id for update of root;
  if not found or r.status<>'active' then perform private.correction_fail('not_found'); end if;
  select * into a from public.analyses where id=p_analysis_id and owner_id=p_user_id and organization_id=r.organization_id for update;
  if not found then perform private.correction_fail('not_found'); end if;
  perform 1 from public.memberships where organization_id=r.organization_id and user_id=p_user_id and role in ('owner','admin','member') for share;
  if not found then perform private.correction_fail('not_found'); end if;
  select * into v from public.resume_versions where id=a.resume_version_id and resume_id=r.id and owner_id=p_user_id and organization_id=r.organization_id for share;
  if not found then perform private.correction_fail('not_found'); end if;
  if p_original_attempt is null then
    select * into e from public.document_evidence_attempts where analysis_id=a.id order by attempt_number desc limit 1 for share;
  else
    select * into e from public.document_evidence_attempts where id=p_original_attempt and analysis_id=a.id for share;
  end if;
  if e.id is null or (e.owner_id,e.organization_id,e.resume_version_id,e.input_sha256) is distinct from (p_user_id,r.organization_id,v.id,v.checksum)
    or (e.envelope->>'id',e.envelope->>'ownerId',e.envelope->>'analysisId',e.envelope->>'sourceId',e.envelope->>'inputSha256')
      is distinct from (e.id::text,p_user_id::text,a.id::text,v.id::text,v.checksum) then perform private.correction_fail('unavailable'); end if;
  -- No current head, global history, current prefix cap or head proof before prior lookup.
  return jsonb_build_object('ownerId',p_user_id,'organizationId',r.organization_id,'resumeId',r.id,'analysisId',a.id,'sourceId',v.id,'attemptId',e.id,'inputSha256',v.checksum);
end;
$erg_entry_regroup_scope$;

create function private.entry_regroup_prove_data(p_scope jsonb,p_base jsonb,p_through integer,p_cache jsonb) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_prove_data$
declare c jsonb:=private.entry_regroup_origin_proof(p_scope,p_through,p_cache); loaded jsonb; v public.canonical_resume_versions%rowtype; id_value uuid; depth integer:=0;
begin
  loaded:=private.entry_regroup_load_base(p_scope,p_base,c); c:=loaded->'cache'; id_value:=(p_base->>'versionId')::uuid;
  loop
    c:=private.entry_regroup_check_header(id_value,c);
    select * into strict v from public.canonical_resume_versions where id=id_value;
    if v.parent_version_id is null or depth=64 then exit; end if;
    id_value:=v.parent_version_id; depth:=depth+1;
  end loop;
  return jsonb_build_object('data',loaded->'data','cache',c);
end;
$erg_entry_regroup_prove_data$;

create function private.entry_regroup_validate_head(p_version uuid) returns void
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_validate_head$
declare v public.canonical_resume_versions%rowtype; e public.document_evidence_attempts%rowtype; scope jsonb; through integer; checked jsonb;
begin
  select * into strict v from public.canonical_resume_versions where id=p_version;
  if not exists(select 1 from public.canonical_entry_regroup_snapshots where canonical_version_id=v.id) then
    if v.content->>'producerVersion'='canonical-engine/1.0.4' or exists(select 1 from public.canonical_entry_regroup_snapshots where canonical_version_id=v.parent_version_id)
      then perform private.correction_fail('unavailable'); end if;
    perform private.entry_association_validate_head(v.id); return;
  end if;
  select * into strict e from public.document_evidence_attempts where id=v.source_evidence_version_id;
  scope:=jsonb_build_object('ownerId',v.owner_id,'organizationId',v.organization_id,'resumeId',v.resume_id,'analysisId',e.analysis_id,
    'sourceId',v.source_resume_version_id,'attemptId',e.id,'inputSha256',e.input_sha256);
  select coalesce(max(regroup_revision),0) into through from public.candidate_entry_regroup_batches
    where analysis_id=e.analysis_id and result_revision<=v.revision;
  checked:=private.entry_regroup_prove_data(scope,jsonb_build_object('versionId',v.id,'revision',v.revision,'locale',v.locale,
    'contentSha256',encode(extensions.digest(convert_to(v.content::text,'UTF8'),'sha256'),'hex')),through,private.entry_regroup_cache());
end;
$erg_entry_regroup_validate_head$;

create function private.entry_regroup_current(p_scope jsonb,p_cache jsonb) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_current$
declare c jsonb:=p_cache; expected jsonb; h public.canonical_resume_versions%rowtype; e public.document_evidence_attempts%rowtype;
  review_n bigint; assignment_n bigint; association_n bigint; regroup_n bigint; review_id uuid; assignment_id uuid; association_id uuid; regroup_id uuid;
  b public.candidate_entry_association_batches%rowtype; prior uuid; n integer:=0; checked jsonb;
begin
  select * into strict e from public.document_evidence_attempts where analysis_id=(p_scope->>'analysisId')::uuid order by attempt_number desc limit 1;
  if e.id::text is distinct from p_scope->>'attemptId' then perform private.correction_fail('conflict'); end if;
  select count(*),coalesce(max(revision),0) into review_n,assignment_n from public.evidence_corrections where analysis_id=e.analysis_id;
  if review_n<>assignment_n or review_n>1000 then perform private.correction_fail('unavailable'); end if;
  select id into review_id from public.evidence_corrections where analysis_id=e.analysis_id and revision=review_n;
  select count(*),coalesce(max(assignment_revision),0) into assignment_n,association_n from public.candidate_correction_assignments where analysis_id=e.analysis_id;
  if assignment_n<>association_n or assignment_n>1000 then perform private.correction_fail('unavailable'); end if;
  select id into assignment_id from public.candidate_correction_assignments where analysis_id=e.analysis_id and assignment_revision=assignment_n;
  select count(*),coalesce(max(association_revision),0) into association_n,regroup_n from public.candidate_entry_association_batches where analysis_id=e.analysis_id;
  if association_n<>regroup_n or association_n>1000 then perform private.correction_fail('unavailable'); end if;
  for b in select * from public.candidate_entry_association_batches where analysis_id=e.analysis_id order by association_revision loop
    c:=private.entry_regroup_work(c,1); n:=n+1;
    if b.association_revision<>n or b.previous_batch_id is distinct from prior
      or (b.owner_id,b.organization_id,b.resume_id,b.source_resume_version_id,b.input_sha256) is distinct from
        ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,(p_scope->>'resumeId')::uuid,(p_scope->>'sourceId')::uuid,p_scope->>'inputSha256') then perform private.correction_fail('unavailable'); end if;
    if not (c->'origins' ? ('association:'||b.id::text)) then
      perform private.entry_association_validate_head(b.result_version_id);
      c:=jsonb_set(c,'{origins}',(c->'origins')||jsonb_build_object('association:'||b.id::text,true));
    end if;
    prior:=b.id;
  end loop;
  association_id:=prior;
  select count(*),coalesce(max(regroup_revision),0) into regroup_n,n from public.candidate_entry_regroup_batches where analysis_id=e.analysis_id;
  if regroup_n<>n or regroup_n>1000 then perform private.correction_fail('unavailable'); end if;
  select id into regroup_id from public.candidate_entry_regroup_batches where analysis_id=e.analysis_id and regroup_revision=regroup_n;
  c:=private.entry_regroup_origin_proof(p_scope,regroup_n::integer,c);
  select v.* into h from public.canonical_resume_heads x join public.canonical_resume_versions v on v.id=x.version_id
    where x.resume_id=(p_scope->>'resumeId')::uuid for update of x;
  if h.id is null then perform private.correction_fail('conflict'); end if;
  checked:=private.entry_regroup_prove_data(p_scope,jsonb_build_object('versionId',h.id,'revision',h.revision,'locale',h.locale,
    'contentSha256',encode(extensions.digest(convert_to(h.content::text,'UTF8'),'sha256'),'hex')),regroup_n::integer,c); c:=checked->'cache';
  expected:=jsonb_build_object('reviewRevision',review_n,'reviewHeadId',review_id,'assignmentRevision',assignment_n,'assignmentHeadId',assignment_id,
    'associationRevision',association_n,'associationHeadId',association_id,'regroupRevision',regroup_n,'regroupHeadId',regroup_id,'canonicalHeadId',h.id);
  return jsonb_build_object('captured',expected,'head',checked->'data','cache',c);
end;
$erg_entry_regroup_current$;

create function private.entry_regroup_review_prefix(p_scope jsonb,p_revision integer,p_head uuid,p_cache jsonb) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_review_prefix$
declare c jsonb:=p_cache; events jsonb:='[]'; chunk jsonb; event jsonb; witness jsonb; received integer:=0; pages integer:=0; last_id uuid;
  attempt public.document_evidence_attempts%rowtype; limit_revision integer;
begin
  if p_revision not between 0 and 1000 or ((p_revision=0) is distinct from (p_head is null)) then perform private.correction_fail('invalid'); end if;
  while received<p_revision loop
    -- Genuine bounded reads: immutable GLOBAL prefix, not attempt-filtered rows and not a fabricated page count.
    select coalesce(jsonb_agg(jsonb_build_object('id',id,'attempt_id',attempt_id,'analysis_id',analysis_id,'owner_id',owner_id,'actor_id',actor_id,
      'revision',revision,'expected_revision',expected_revision,'field',field,'operation',operation,'value',value,'candidate_id',candidate_id,
      'created_at',to_char(created_at at time zone 'UTC','YYYY-MM-DD"T"HH24:MI:SS.US"Z"')) order by revision),'[]'::jsonb) into chunk
      from (select * from public.evidence_corrections where analysis_id=(p_scope->>'analysisId')::uuid and revision>received and revision<=p_revision order by revision limit 100) page;
    if jsonb_array_length(chunk)=0 then perform private.correction_fail('unavailable'); end if;
    pages:=pages+1; if pages>10 then perform private.correction_fail('unavailable'); end if;
    for event in select value from jsonb_array_elements(chunk) loop
      c:=private.entry_regroup_inspect(c,'review',event->>'id',event); received:=received+1;
      if (event->>'revision')::integer<>received or (event->>'expected_revision')::integer<>received-1
        or (event->>'owner_id',event->>'actor_id',event->>'analysis_id') is distinct from (p_scope->>'ownerId',p_scope->>'ownerId',p_scope->>'analysisId')
        or event->>'field' not in ('name','email','phone','employer','role','dates','education','certification')
        or event->>'operation' not in ('confirm','correct') or private.grouped_import_utf16_length(event->>'value') not between 1 and 2000
        or event->>'value' is distinct from private.grouped_import_trim(event->>'value')
        or ((event->>'operation'='confirm') is distinct from (event->'candidate_id'<>'null'::jsonb)) then perform private.correction_fail('unavailable'); end if;
      select * into strict attempt from public.document_evidence_attempts where id=(event->>'attempt_id')::uuid;
      if (attempt.owner_id,attempt.organization_id,attempt.analysis_id,attempt.resume_version_id,attempt.input_sha256)
        is distinct from ((p_scope->>'ownerId')::uuid,(p_scope->>'organizationId')::uuid,(p_scope->>'analysisId')::uuid,(p_scope->>'sourceId')::uuid,p_scope->>'inputSha256') then perform private.correction_fail('unavailable'); end if;
      c:=private.entry_regroup_inspect(c,'attempt',attempt.id::text,attempt.envelope);
      if event->>'operation'='confirm' and (private.grouped_import_utf16_length(event->>'candidate_id') not between 1 and 100 or
        (attempt.id::text=p_scope->>'attemptId' and not exists(select 1 from jsonb_array_elements(attempt.envelope->'assertions') x(value)
          where x.value->>'id'=event->>'candidate_id' and x.value->>'field'=event->>'field' and x.value->>'value'=event->>'value'))) then perform private.correction_fail('unavailable'); end if;
      last_id:=(event->>'id')::uuid;
    end loop;
    events:=events||chunk;
  end loop;
  if received<>p_revision or last_id is distinct from p_head then perform private.correction_fail('unavailable'); end if;
  select * into strict attempt from public.document_evidence_attempts where id=(p_scope->>'attemptId')::uuid;
  witness:=jsonb_build_object('schemaVersion','evidence-review-history/1.0.0','state','complete_captured_prefix',
    'binding',(p_scope-array['organizationId','resumeId'])||jsonb_build_object('evidenceSchemaVersion','evidence/2.0.0','fieldExtractorVersion',attempt.envelope->'fieldExtractorVersion','targetModelVersion','evidence-targets/1.0.0'),
    'highWaterRevision',p_revision,'highWaterEventId',p_head,'receivedCount',received,'pageCount',pages);
  return jsonb_build_object('review',jsonb_build_object('revision',p_revision,'headId',p_head,'events',events,'historyAttestation',witness),'cache',c);
end;
$erg_entry_regroup_review_prefix$;

create function private.entry_regroup_write_snapshot(p_version uuid,p_command jsonb) returns void
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_write_snapshot$
declare v public.canonical_resume_versions%rowtype; e public.document_evidence_attempts%rowtype; batch public.candidate_entry_regroup_batches%rowtype;
  kind text; inherited uuid; restored uuid; links jsonb:='[]'; link jsonb; i integer:=0;
begin
  select * into strict v from public.canonical_resume_versions where id=p_version;
  if v.command is distinct from p_command or exists(select 1 from public.canonical_entry_regroup_snapshots where canonical_version_id=v.id) then perform private.correction_fail('unavailable'); end if;
  select * into batch from public.candidate_entry_regroup_batches where result_version_id=v.id;
  if batch.id is not null then
    kind:='regroup'; links:=private.entry_regroup_expected_links(batch.base_version_id,v.id,batch.id);
  elsif p_command->>'operation'='rollback' and (exists(select 1 from public.canonical_entry_regroup_snapshots where canonical_version_id=v.parent_version_id)
    or exists(select 1 from public.canonical_entry_regroup_snapshots where canonical_version_id=(p_command->>'targetVersionId')::uuid)) then
    kind:='restore'; restored:=(p_command->>'targetVersionId')::uuid; links:=private.entry_regroup_links(restored);
  elsif exists(select 1 from public.canonical_entry_regroup_snapshots where canonical_version_id=v.parent_version_id) then
    kind:='inherited'; inherited:=v.parent_version_id;
    for link in select value from jsonb_array_elements(private.entry_regroup_links(inherited)) loop
      if private.entry_regroup_group_active(v.id,link) then links:=links||jsonb_build_array(link); end if;
    end loop;
  else
    if v.content->>'producerVersion'='canonical-engine/1.0.4' then perform private.correction_fail('unavailable'); end if;
    perform private.entry_association_write_snapshot(v.id,p_command); return;
  end if;
  select * into strict e from public.document_evidence_attempts where id=v.source_evidence_version_id;
  insert into public.canonical_entry_regroup_snapshots(canonical_version_id,owner_id,organization_id,resume_id,analysis_id,source_resume_version_id,evidence_attempt_id,input_sha256,
    composition_version,creation_kind,creating_regroup_batch_id,inherited_from_version_id,restored_from_version_id)
    values(v.id,v.owner_id,v.organization_id,v.resume_id,e.analysis_id,v.source_resume_version_id,e.id,e.input_sha256,
      'entry-regroup-composition/1.0.0',kind,batch.id,inherited,restored);
  for link in select value from jsonb_array_elements(links) loop
    i:=i+1;
    if private.entry_regroup_group_active(v.id,link) is not true then perform private.correction_fail('unavailable'); end if;
    insert into public.canonical_entry_regroup_links(canonical_version_id,owner_id,organization_id,resume_id,analysis_id,source_resume_version_id,evidence_attempt_id,input_sha256,
      composition_order,section_id,entry_id,origin_family,association_group_id,association_batch_id,regroup_group_id,regroup_batch_id,original_approval_version_id)
      values(v.id,v.owner_id,v.organization_id,v.resume_id,e.analysis_id,v.source_resume_version_id,e.id,e.input_sha256,i,(link->>'sectionId')::uuid,(link->>'entryId')::uuid,
        link->>'originFamily',(link->>'associationGroupId')::uuid,(link->>'associationBatchId')::uuid,(link->>'regroupGroupId')::uuid,(link->>'regroupBatchId')::uuid,(link->>'originalApprovalVersionId')::uuid);
  end loop;
end;
$erg_entry_regroup_write_snapshot$;

create function private.entry_regroup_save_version(p_user_id uuid,p_source_id uuid,p_body jsonb,p_command jsonb,p_request_id text) returns jsonb
language plpgsql volatile security invoker set search_path='' as $erg_entry_regroup_save_version$
declare b public.candidate_entry_regroup_batches%rowtype; base public.canonical_resume_versions%rowtype; e public.document_evidence_attempts%rowtype; event jsonb; bytes bytea; digest text;
begin
  if private.grouped_import_keys(p_command,array['schemaVersion','operation','batchId','semanticSha256','idempotencyKey']) is not true
    or p_command->>'schemaVersion'<>'entry-regroup-storage/1.0.0' or p_command->>'operation'<>'entry_regroup'
    or private.grouped_import_uuid(p_command->'batchId') is not true or private.grouped_import_uuid(p_command->'idempotencyKey') is not true
    or p_request_id is null or char_length(p_request_id) not between 1 and 100 then perform private.correction_fail('invalid'); end if;
  select * into strict b from public.candidate_entry_regroup_batches where id=(p_command->>'batchId')::uuid;
  -- Protected pending rows are the authority. Committed inner commands cannot bypass outer semantic replay.
  if exists(select 1 from public.canonical_resume_versions where id=b.result_version_id) then raise exception 'REGROUP_REQUIRES_OUTER_REPLAY' using errcode='22023'; end if;
  select * into strict base from public.canonical_resume_versions where id=b.base_version_id;
  select * into strict e from public.document_evidence_attempts where id=b.evidence_attempt_id;
  bytes:=private.grouped_import_command_bytes(private.entry_regroup_atoms(b.command->'normalForm')); digest:=private.grouped_import_command_sha256(private.entry_regroup_atoms(b.command->'normalForm'));
  if p_command is distinct from jsonb_build_object('schemaVersion','entry-regroup-storage/1.0.0','operation','entry_regroup','batchId',b.id,'semanticSha256',b.command_sha256,'idempotencyKey',b.storage_key)
    or b.command_bytes is distinct from bytes or b.command_sha256 is distinct from digest or b.command->>'previewHash' is distinct from digest
    or b.codec_version<>'a02-command-bytes/1.0.0' or b.storage_key=b.idempotency_key
    or (b.owner_id,b.organization_id,b.resume_id,b.source_resume_version_id,b.evidence_attempt_id,b.base_revision)
      is distinct from (p_user_id,base.organization_id,base.resume_id,p_source_id,base.source_evidence_version_id,base.revision)
    or (e.owner_id,e.organization_id,e.analysis_id,e.resume_version_id,e.input_sha256) is distinct from (b.owner_id,b.organization_id,b.analysis_id,b.source_resume_version_id,b.input_sha256)
    or b.base_content_sha256 is distinct from encode(extensions.digest(convert_to(base.content::text,'UTF8'),'sha256'),'hex')
    or not exists(select 1 from public.canonical_resume_heads where resume_id=b.resume_id and version_id=b.base_version_id and revision=b.base_revision)
    or b.result_content_sha256 is distinct from encode(extensions.digest(convert_to(p_body::text,'UTF8'),'sha256'),'hex')
    or private.canonical_shape_valid(p_body) is not true or octet_length(p_body::text)>1048576
    or p_body->>'producerVersion'<>'canonical-engine/1.0.4' or p_body->>'versionId'<>b.result_version_id::text
    or p_body->>'parentVersionId'<>b.base_version_id::text or b.result_revision<>b.base_revision+1
    or p_body->'facts' is distinct from base.content->'facts' or p_body->'contactFactIds' is distinct from base.content->'contactFactIds'
    or (p_body-array['versionId','parentVersionId','createdAt','producerVersion','state','sections','changes']) is distinct from
      (base.content-array['versionId','parentVersionId','createdAt','producerVersion','state','sections','changes']) then perform private.correction_fail('invalid'); end if;
  event:=p_body->'changes'->0;
  if jsonb_array_length(p_body->'changes')<>1 or event->>'actorId'<>p_user_id::text or event->>'action'<>'edit'
    or event->>'targetId'<>b.resume_id::text or event->'priorValue'<>'null'::jsonb or event->'createdAt' is distinct from p_body->'createdAt' then perform private.correction_fail('invalid'); end if;
  insert into public.canonical_resume_versions(id,resume_id,source_resume_version_id,organization_id,owner_id,locale,state,content,created_at,parent_version_id,
    source_evidence_version_id,revision,idempotency_key,command,request_id)
    values(b.result_version_id,b.resume_id,b.source_resume_version_id,b.organization_id,b.owner_id,p_body->>'locale',p_body->>'state',p_body,(p_body->>'createdAt')::timestamptz,
      b.base_version_id,b.evidence_attempt_id,b.result_revision,b.storage_key,p_command,p_request_id);
  insert into public.resume_facts(canonical_version_id,id,organization_id,owner_id,type,value,state,sources,confirmation)
    select b.result_version_id,(value->>'id')::uuid,b.organization_id,b.owner_id,value->>'type',value->>'value',value->>'state',value->'sources',nullif(value->'confirmation','null'::jsonb) from jsonb_array_elements(p_body->'facts');
  insert into public.change_events(canonical_version_id,id,organization_id,owner_id,actor_id,operation,event,created_at)
    values(b.result_version_id,(event->>'id')::uuid,b.organization_id,b.owner_id,b.owner_id,'edit',event,(p_body->>'createdAt')::timestamptz);
  update public.canonical_resume_heads set version_id=b.result_version_id,revision=b.result_revision where resume_id=b.resume_id and version_id=b.base_version_id and revision=b.base_revision;
  if not found then perform private.correction_fail('conflict'); end if;
  perform private.write_correction_snapshot(b.result_version_id,p_command);
  insert into public.audit_events(organization_id,actor_id,action,target_type,target_id,outcome,request_id)
    values(b.organization_id,b.owner_id,'canonical.entry_regroup','canonical_resume_version',b.result_version_id,'success',p_request_id);
  return jsonb_build_object('status','applied','versionId',b.result_version_id,'revision',b.result_revision,'idempotent',false);
end;
$erg_entry_regroup_save_version$;

create function public.apply_canonical_entry_regroup(p_user_id uuid,p_analysis_id uuid,p_command jsonb,p_request_id text) returns jsonb
language plpgsql volatile security definer set search_path='' as $erg_apply_canonical_entry_regroup$
declare scope jsonb; prior public.candidate_entry_regroup_batches%rowtype; normal jsonb; c jsonb:=private.entry_regroup_cache(); loaded jsonb; data jsonb; current_view jsonb;
  bytes bytea; digest text; plan jsonb; document jsonb; storage jsonb; result jsonb; expected jsonb; selection jsonb; member jsonb; g uuid; batch uuid; version uuid; change uuid; inner_key uuid;
  created text; group_index integer:=0; member_index integer; through integer; checked jsonb;
begin
  if private.grouped_import_keys(p_command,array['normalForm','previewHash','idempotencyKey']) is not true or private.grouped_import_uuid(p_command->'idempotencyKey') is not true
    or private.grouped_import_uuid(p_command->'normalForm'->'binding'->'attemptId') is not true or p_request_id is null or char_length(p_request_id) not between 1 and 100 then perform private.correction_fail('invalid'); end if;
  scope:=private.entry_regroup_scope(p_user_id,p_analysis_id,(p_command->'normalForm'->'binding'->>'attemptId')::uuid);
  if private.canonical_correction_write_ready() is not true then perform private.correction_fail('unavailable'); end if;
  select * into prior from public.candidate_entry_regroup_batches where resume_id=(scope->>'resumeId')::uuid and idempotency_key=(p_command->>'idempotencyKey')::uuid;
  if found then
    if prior.command is distinct from p_command or (prior.owner_id,prior.organization_id,prior.analysis_id,prior.source_resume_version_id,prior.evidence_attempt_id,prior.input_sha256)
      is distinct from (p_user_id,(scope->>'organizationId')::uuid,p_analysis_id,(scope->>'sourceId')::uuid,(scope->>'attemptId')::uuid,scope->>'inputSha256') then perform private.correction_fail('conflict'); end if;
    -- Only ORIGINAL prefix proof; neither current head nor later history participates.
    c:=private.entry_regroup_origin_proof(scope,prior.regroup_revision,c);
    return private.entry_regroup_receipt(prior.id,true);
  end if;
  normal:=p_command->'normalForm';
  if private.evidence_integer(normal->'expected'->'regroupRevision',0,1000) is not true then perform private.correction_fail('invalid'); end if;
  through:=(normal->'expected'->>'regroupRevision')::numeric::integer;
  loaded:=private.entry_regroup_prove_data(scope,normal->'base',through,c); c:=loaded->'cache'; data:=loaded->'data';
  c:=private.entry_regroup_inspect(c,'command',p_command->>'idempotencyKey',p_command);
  normal:=private.entry_regroup_normalize_loaded(scope,p_command,data); bytes:=private.grouped_import_command_bytes(private.entry_regroup_atoms(normal)); digest:=private.grouped_import_command_sha256(private.entry_regroup_atoms(normal));
  current_view:=private.entry_regroup_current(scope,c); c:=current_view->'cache'; expected:=normal->'expected';
  if current_view->'captured' is distinct from expected or (current_view->'head'-array['content','envelope']) is distinct from normal->'base' then perform private.correction_fail('conflict'); end if;
  if through>=1000 then perform private.correction_fail('unavailable'); end if;
  plan:=private.entry_regroup_resolve_loaded(scope,data,normal->'selections');
  -- Technical allocation occurs only after prior miss and every fresh check.
  batch:=gen_random_uuid(); version:=gen_random_uuid(); change:=gen_random_uuid(); inner_key:=gen_random_uuid();
  created:=to_char(clock_timestamp() at time zone 'UTC','YYYY-MM-DD"T"HH24:MI:SS.US"Z"');
  document:=private.entry_regroup_child_body(data->'content',plan->'afterSections',version,change,created);
  c:=private.entry_regroup_inspect(c,'document',version::text,document);
  storage:=jsonb_build_object('schemaVersion','entry-regroup-storage/1.0.0','operation','entry_regroup','batchId',batch,'semanticSha256',digest,'idempotencyKey',inner_key);
  insert into public.candidate_entry_regroup_batches(id,owner_id,organization_id,resume_id,analysis_id,source_resume_version_id,evidence_attempt_id,input_sha256,
    regroup_revision,previous_batch_id,idempotency_key,command,command_bytes,command_sha256,codec_version,base_version_id,base_revision,base_content_sha256,
    result_version_id,result_revision,result_content_sha256,storage_key)
    values(batch,p_user_id,(scope->>'organizationId')::uuid,(scope->>'resumeId')::uuid,p_analysis_id,(scope->>'sourceId')::uuid,(scope->>'attemptId')::uuid,scope->>'inputSha256',
      through+1,(expected->>'regroupHeadId')::uuid,(p_command->>'idempotencyKey')::uuid,p_command,bytes,digest,'a02-command-bytes/1.0.0',
      (data->>'versionId')::uuid,(data->>'revision')::integer,data->>'contentSha256',version,(data->>'revision')::integer+1,
      encode(extensions.digest(convert_to(document::text,'UTF8'),'sha256'),'hex'),inner_key);
  for selection in select value from jsonb_array_elements(normal->'selections') loop
    c:=private.entry_regroup_work(c,1); group_index:=group_index+1; g:=gen_random_uuid();
    insert into public.candidate_entry_regroup_groups(id,batch_id,owner_id,organization_id,resume_id,analysis_id,source_resume_version_id,evidence_attempt_id,input_sha256,
      base_version_id,result_version_id,group_order,page_id,method,section_id,anchor_entry_id,insertion_index,affected_entry_ids,source_selection)
      values(g,batch,p_user_id,(scope->>'organizationId')::uuid,(scope->>'resumeId')::uuid,p_analysis_id,(scope->>'sourceId')::uuid,(scope->>'attemptId')::uuid,scope->>'inputSha256',
        (data->>'versionId')::uuid,version,group_index,selection->>'pageId',selection->>'method',(selection->>'sectionId')::uuid,(selection->>'anchorEntryId')::uuid,
        (selection->>'insertionIndex')::integer,array(select jsonb_array_elements_text(selection->'affectedEntryIds'))::uuid[],
        jsonb_build_object('pageId',selection->'pageId','method',selection->'method','members',(select jsonb_agg(value-array['factId','mappedThrough','originVersionId','location'] order by ordinality) from jsonb_array_elements(selection->'members') with ordinality)));
    member_index:=0;
    for member in select value from jsonb_array_elements(selection->'members') loop
      c:=private.entry_regroup_work(c,1); member_index:=member_index+1;
      insert into public.candidate_entry_regroup_members(id,group_id,batch_id,owner_id,organization_id,resume_id,analysis_id,source_resume_version_id,evidence_attempt_id,input_sha256,
        base_version_id,result_version_id,source_order,assertion_id,field,fact_id,citations,original_entry_id,location_kind,original_bullet_id,mapped_through,origin_version_id,origin_fact_id)
        values(gen_random_uuid(),g,batch,p_user_id,(scope->>'organizationId')::uuid,(scope->>'resumeId')::uuid,p_analysis_id,(scope->>'sourceId')::uuid,(scope->>'attemptId')::uuid,scope->>'inputSha256',
          (data->>'versionId')::uuid,version,member_index,member->>'assertionId',member->>'field',(member->>'factId')::uuid,member->'citations',(member->'location'->>'entryId')::uuid,
          member->'location'->>'kind',(member->'location'->>'bulletId')::uuid,member->>'mappedThrough',(member->>'originVersionId')::uuid,
          case when member->>'mappedThrough'='owned_ancestor_chain' then (member->>'factId')::uuid else null end);
    end loop;
  end loop;
  result:=public.save_canonical_version(p_user_id,(scope->>'sourceId')::uuid,document,storage,p_request_id);
