type Result = { data: unknown; error: unknown };
type Read = {
  table: string;
  columns: string;
  filters: Record<string, unknown>;
  single: boolean;
};
const ok = (data: unknown): Result => ({ data, error: null });
function ownedStore(
  context: RequirementRelationshipContext,
  override?: (read: Read, sequence: Read[]) => Result | undefined,
) {
  const c = context.canonical,
    s = context.source,
    r = context.requirements;
  const docs = [c, ...context.ancestors];
  const reads: Read[] = [];
  const events: Record<string, unknown>[] = [];
  const head = {
    version_id: context.heads.canonicalHeadId,
    revision: context.heads.canonicalHeadRevision,
  };
  const root = {
    id: s.resumeId,
    owner_id: s.ownerId,
    organization_id: s.organizationId,
    status: "active",
  };
  const analysis = {
    id: s.analysisId,
    owner_id: s.ownerId,
    organization_id: s.organizationId,
    resume_version_id: s.sourceId,
    job_id: r.content.jobId,
  };
  const source = {
    id: s.sourceId,
    resume_id: s.resumeId,
    owner_id: s.ownerId,
    organization_id: s.organizationId,
    active_resume: root,
    page_count: context.sourceMetadata.pageCount,
  };
  const job = {
    id: r.content.jobId,
    title: "Software Engineer",
    description: r.content.sourceText,
    owner_id: s.ownerId,
    organization_id: s.organizationId,
  };
  const canonicalRows = docs.map((d) => ({
    id: d.versionId,
    resume_id: s.resumeId,
    owner_id: s.ownerId,
    organization_id: s.organizationId,
    source_resume_version_id: s.sourceId,
    parent_version_id: d.content.parentVersionId,
    revision: d.revision,
    content: d.content,
    created_at: d.content.createdAt,
  }));
  const requirementRows = [
    {
      id: r.versionId,
      analysis_id: s.analysisId,
      job_id: r.content.jobId,
      owner_id: s.ownerId,
      organization_id: s.organizationId,
      parent_version_id: r.content.parentVersionId,
      revision: r.revision,
      content: r.content,
      created_at: r.content.createdAt,
    },
  ];
  const rpc = vi.fn<(name: string, args: unknown) => Promise<Result>>(
    async (name) => {
      if (name !== "read_requirement_relationship_context")
        throw Error("UNEXPECTED_WRITE_RPC");
      return ok(context);
    },
  );
  function valueAt(row: Record<string, unknown>, path: string): unknown {
    let value: unknown = row;
    for (const key of path.split(".")) {
      if (typeof value !== "object" || value === null) return undefined;
      value = (value as Record<string, unknown>)[key];
    }
    return value;
  }
  const from = vi.fn((table: string) => {
    const read: Read = { table, columns: "", filters: {}, single: false };
    const query = {
      select(columns: string) {
        read.columns = columns;
        return query;
      },
      eq(key: string, value: unknown) {
        read.filters[key] = value;
        return query;
      },
      neq() {
        return query;
      },
      order() {
        return query;
      },
      limit() {
        return query;
      },
      range() {
        return query;
      },
      maybeSingle() {
        read.single = true;
        return query;
      },
      then(yes: (result: Result) => unknown, no: (error: unknown) => unknown) {
        reads.push(read);
        const replacement = override?.(read, reads);
        if (replacement) return Promise.resolve(replacement).then(yes, no);
        let rows: Record<string, unknown>[];
        if (table === "analyses") rows = [analysis];
        else if (table === "resume_versions") rows = [source];
        else if (table === "memberships")
          rows = [
            {
              organization_id: s.organizationId,
              user_id: s.ownerId,
              role: "owner",
            },
          ];
        else if (table === "data_requests") rows = [];
        else if (table === "authoring_events") rows = events;
        else if (table === "resumes") rows = [root];
        else if (table === "jobs") rows = [job];
        else if (table === "canonical_resume_heads")
          rows = [
            {
              resume_id: s.resumeId,
              owner_id: s.ownerId,
              organization_id: s.organizationId,
              version_id: head.version_id,
              revision: head.revision,
            },
          ];
        else if (table === "canonical_resume_versions") rows = canonicalRows;
        else if (table === "job_requirement_versions") rows = requirementRows;
        else if (table === "document_evidence_attempts")
          rows = [
            {
              id: context.attempt.id,
              analysis_id: s.analysisId,
              owner_id: s.ownerId,
              organization_id: s.organizationId,
              envelope: context.attempt.envelope,
              resume_version_id: s.sourceId,
              input_sha256: s.inputSha256,
              attempt_number: context.attempt.attemptNumber,
            },
          ];
        else throw Error("UNEXPECTED_TABLE_READ");
        rows = rows.filter((row) =>
          Object.entries(read.filters).every(
            ([key, value]) => valueAt(row, key) === value,
          ),
        );
        if (
          table === "canonical_resume_versions" ||
          table === "job_requirement_versions"
        )
          rows = [...rows].sort(
            (a, b) => Number(b.revision) - Number(a.revision),
          );
        return Promise.resolve(ok(read.single ? (rows[0] ?? null) : rows)).then(
          yes,
          no,
        );
      },
    };
    return query;
  });
  return {
    client: { from, rpc } as unknown as Parameters<
      typeof executeTypedAuthoringHttp
    >[0],
    from,
    rpc,
    reads,
    canonicalRows,
    requirementRows,
    source,
    analysis,
    job,
    events,
    head,
  };
}

function request(
  fields: Record<string, string> | URLSearchParams | FormData,
  options: { method?: string; origin?: string } = {},
) {
  return new Request(
    "https://example.test/analysis/" + id(4) + "/author/typed",
    {
      method: options.method ?? "POST",
      headers: { Origin: options.origin ?? "https://example.test" },
      body:
        fields instanceof URLSearchParams || fields instanceof FormData
          ? fields
          : new URLSearchParams(fields),
    },
  );
}
function privateResponse(response: Response) {
  expect(response.headers.get("Cache-Control")).toBe("private, no-store");
  expect(response.headers.get("X-Robots-Tag")).toBe("noindex, nofollow");
}
async function setup() {
  const f = await createRequirementContextFixture({ confirm: true });
  const selection = {
    canonicalVersionId: f.context.canonical.versionId,
    requirementVersionId: f.context.requirements.versionId,
    clauseId: f.context.requirements.content.relations.clauses[0].id,
    targetFactId: f.targetFactId,
  };
  const patch = buildTypedAuthoringPatch(f.context, selection, "Created");
  const accept = createTypedAuthoringCommand(patch, "accept", id(700), true),
    reject = createTypedAuthoringCommand(patch, "reject", id(701), false);
  const edited = reviseCanonical(f.context.canonical.content, {
    operation: "edit",
    factId: f.targetFactId,
    value: patch.after,
  });
  const confirmed = reviseCanonical(edited, {
    operation: "confirm",
    factId: f.targetFactId,
  });
  const receiptFor = (
    command: TypedAuthoringCommand,
  ): TypedAuthoringReceipt => ({
    schemaVersion: "typed-authoring-receipt/1.0.0",
    status: "applied",
    action: command.operation,
    idempotencyKey: command.idempotencyKey,
    commandSha256: typedAuthoringCommandSha256(command),
    baseVersionId: f.context.canonical.versionId,
    baseRevision: f.context.canonical.revision,
    requirementVersionId: f.context.requirements.versionId,
    requirementRevision: f.context.requirements.revision,
    versionId:
      command.operation === "accept"
        ? confirmed.versionId
        : f.context.canonical.versionId,
    revision:
      f.context.canonical.revision + (command.operation === "accept" ? 2 : 0),
    eventId: id(command.operation === "accept" ? 702 : 703),
    idempotent: false,
  });
  const store = ownedStore(f.context);
  store.rpc.mockImplementation(async (name, args) => {
    if (name === "read_requirement_relationship_context") return ok(f.context);
    if (name === "read_typed_authoring_readiness")
      return ok({
        ...TYPED_AUTHORING_READINESS_PROFILE,
        schemaReady: true,
        writerExecutable: true,
      });
    if (name === "apply_typed_authoring_action")
      return ok(
        receiptFor((args as { p_command: TypedAuthoringCommand }).p_command),
      );
    throw Error("UNEXPECTED_WRITE_RPC");
  });
  mocks.user.mockResolvedValue({ id: f.owned.ownerId });
  mocks.admin.mockReturnValue(store.client);
  const args = (req: Request) => ({
    request: req,
    params: { id: f.owned.analysisId },
  });
  const decisions = (command: TypedAuthoringCommand = accept) =>
    request({
      action: "decision",
      command: JSON.stringify(command),
      ...(command.operation === "accept"
        ? { reviewAcknowledgement: "accepted" }
        : {}),
    });
  return {
    ...f,
    store,
    selection,
    patch,
    accept,
    reject,
    edited,
    confirmed,
    receiptFor,
    args,
    decisions,
  };
}

      id: document.versionId,
      resume_id: document.resumeId,
      owner_id: document.ownerId,
      organization_id: document.organizationId,
      source_resume_version_id: f.owned.sourceId,
      parent_version_id: document.parentVersionId,
      revision,
      content: document,
      created_at: document.createdAt,
    });
  f.store.head.version_id = f.confirmed.versionId;
  f.store.head.revision = receipt.revision;

    id: receipt.eventId,
    analysis_id: f.owned.analysisId,
    resume_id: f.owned.resumeId,

    organization_id: f.owned.organizationId,
    base_version_id: receipt.baseVersionId,

    idempotency_key: receipt.idempotencyKey,
    action: "accept",
    command: f.accept,
    result: receipt,
    retention_policy_id: f.confirmed.retentionPolicyId,

    status: "applied",
    action: "undo",

    revision: receipt.revision + 1,
    eventId: id(711),

        receipt: JSON.stringify(r),
        idempotencyKey: id(712),
      }),
  };

it("undoes the actual owned recorded acceptance to its original base via guarded legacy undo", async () => {
  const f = await setup();
  const u = prepareUndo(f);
  const response = await action(
    f.args(u.form({ ...u.receipt, idempotent: true })),
  );
  expect(response.status).toBe(200);
  expect(await response.json()).toEqual({
    kind: "undone",
    receipt: u.undoReceipt,
    idempotencyKey: u.undoKey,
  });
  expect(f.store.rpc).toHaveBeenCalledOnce();
  const [name, args] = f.store.rpc.mock.calls[0];
  expect(name).toBe("apply_authoring_action");
  expect(args).toMatchObject({
    p_user_id: f.owned.ownerId,
    p_analysis_id: f.owned.analysisId,
    p_request_id: u.undoKey,
    p_command: {
      action: "undo",
      expectedVersionId: u.receipt.versionId,
      idempotencyKey: u.undoKey,
      targetVersionId: u.receipt.baseVersionId,
    },
  });
  expect(args).toHaveProperty("p_body.parentVersionId", u.receipt.versionId);
  expect(args).toHaveProperty(
    "p_body.facts",
    f.context.canonical.content.facts,
  );
  expect(args).toHaveProperty("p_confirmation_body", null);
  expect(
    f.store.reads.some(
      (read) =>
        read.table === "authoring_events" &&
        read.filters.id === u.receipt.eventId &&
        read.filters.owner_id === f.owned.ownerId,
    ),
  ).toBe(true);
  expect(fetch).not.toHaveBeenCalled();
});
it.each([
  "missing_event",
  "tampered_receipt",
  "foreign_event",
  "stale_head",
  "withdrawn",
] as const)(
  "does not undo %s or publish an uncertain success",
  async (kind) => {
    const f = await setup();
    const u = prepareUndo(f);
    let receipt = u.receipt;
    if (kind === "missing_event") f.store.events.length = 0;
    if (kind === "tampered_receipt")
      receipt = { ...receipt, commandSha256: "f".repeat(64) };
    if (kind === "foreign_event") f.store.events[0].owner_id = id(740);
    if (kind === "stale_head") {
      f.store.head.version_id = id(741);
      f.store.head.revision++;
    }
    if (kind === "withdrawn") f.store.source.active_resume.status = "deleted";
    const response = await action(f.args(u.form(receipt)));
    expect(response.status).toBe(
      kind === "stale_head" ? 409 : kind === "tampered_receipt" ? 422 : 404,
    );
    expect((await response.json()).kind).toBe("error");
    expect(f.store.rpc).not.toHaveBeenCalled();
    privateResponse(response);
  },
);
