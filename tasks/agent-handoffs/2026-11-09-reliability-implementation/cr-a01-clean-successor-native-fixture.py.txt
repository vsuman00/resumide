"""Replay the exact packaged SQL on a disposable, socket-only PostgreSQL 17 fixture."""

import json
import pathlib
import re
import subprocess
import sys
import tempfile

root = pathlib.Path.cwd()
pg = pathlib.Path('/opt/homebrew/opt/postgresql@17/bin')
logs = pathlib.Path(sys.argv[1])
with tempfile.TemporaryDirectory(prefix='resumide-a01-pg-') as run_dir:
    run = pathlib.Path(run_dir)
    data = run / 'data'
    env = {'HOME': str(run), 'PATH': '/usr/bin:/bin', 'LC_ALL': 'C'}

    def call(tool, *args, **kwargs):
        return subprocess.run([str(pg / tool), *map(str, args)], timeout=25, env=env, text=True, **kwargs)

    def sql(**kwargs):
        return call('psql', '-X', '-v', 'ON_ERROR_STOP=1', '-h', run, '-U', 'postgres', '-d', 'postgres', '-Atq', **kwargs)

    def logged(name, source):
        with (logs / name).open('w') as report:
            return sql(input=source, stdout=report, stderr=report)

    call('initdb', '-D', data, '--no-sync', '--encoding=UTF8', '--auth=trust', '-U', 'postgres', check=True, stdout=subprocess.DEVNULL)
    try:
        call('pg_ctl', '-w', '-D', data, '-l', run / 'server.log', '-o', f"-c listen_addresses='' -c unix_socket_directories={run} -c unix_socket_permissions=0700 -c statement_timeout=12000", 'start', check=True, stdout=subprocess.DEVNULL)
        bootstrap = '''
        create role anon;
        create role authenticated;
        create role service_role bypassrls;
        create schema auth;
        create schema storage;
        create schema extensions;
        create extension pgcrypto with schema extensions;
        alter database postgres set search_path to public, extensions;
        create table auth.users(id uuid primary key, email text);
        create function auth.uid() returns uuid language sql stable as $$select nullif(current_setting('request.jwt.claim.sub',true),'')::uuid$$;
        create table storage.buckets(id text primary key,name text not null,public boolean,file_size_limit bigint,allowed_mime_types text[]);
        create table storage.objects(id uuid primary key);
        create function extensions.jsonb_matches_schema(json,jsonb) returns boolean language plpgsql immutable as $$begin raise exception 'FIXTURE_JSON_SCHEMA_VALIDATION_NOT_AVAILABLE'; end;$$;
        '''
        assert logged('bootstrap.log', bootstrap).returncode == 0
        assert logged('preflight.log', (root / 'scripts/verify-empty-supabase-target.sql').read_text()).returncode == 0
        assert sql(input='show server_encoding;', capture_output=True, check=True).stdout.strip() == 'UTF8'
        successor = run / 'successor'
        subprocess.run(
            [sys.executable, str(root / 'scripts/materialize-managed-schema-successor.py'), str(successor)],
            timeout=20, check=True,
        )
        for order, path in enumerate(sorted(successor.glob('*.sql')), 1):
            source = path.read_text()
            if path.name.startswith('20261012054243_'):
                line = 'create extension if not exists pg_jsonschema with schema extensions;'
                assert source.count(line) == 1
                source = source.replace(line, '-- The compatibility scaffold supplies an explicit fail-closed stand-in.')
            name = f'{order:02}-{path.stem}.log'
            result = logged(name, source)
            if result.returncode:
                print(f'STOP at operation {order}: {path.name}', flush=True)
                print((logs / name).read_text()[-1500:], flush=True)
                raise SystemExit(result.returncode)
            if order == 55:
                result = logged('correction-ready.log', 'select private.canonical_correction_write_ready();')
                assert result.returncode == 0 and (logs / 'correction-ready.log').read_text().strip() == 't'
            if order == 60:
                binding = """do $test$
                declare result jsonb;
                begin
                  result:=private.canonical_rewrite_binding(
                    '{"source":{"id":"synthetic"},"canonical":{"id":"c","locale":"en","content":"private"},"requirements":{"id":"r","content":"private"},"heads":{}}',
                    '{"id":"d"}','{"id":"p"}');
                  assert result#>'{authoring,canonical}'='{"id":"c"}'::jsonb;
                  assert result#>'{authoring,requirements}'='{"id":"r"}'::jsonb;
                  assert result#>'{authoring,source}'='{"id":"synthetic"}'::jsonb;
                  assert result->'draft'='{"id":"d"}'::jsonb;
                  assert result->'policy'='{"id":"p"}'::jsonb;
                end
                $test$; select 'Rewrite binding JSON fields passed';"""
                result = logged('rewrite-binding.log', binding)
                print((logs / 'rewrite-binding.log').read_text(), flush=True)
                if result.returncode:
                    raise SystemExit(result.returncode)
            if path.name.startswith('20261111010000_'):
                probe = '''
                do $check$ begin
                  assert jsonb_array_length(private.jd_relation_literals(repeat('Git ',300),'git','alias'))=300;
                  assert jsonb_array_length(private.jd_relation_literals(repeat('Git ',1000),'git','ln'))=1000;
                  begin
                    perform private.jd_relation_literals(repeat('Git ',301),'git','alias');
                    raise exception 'Expected alias bound';
                  exception when sqlstate '22023' then
                    assert sqlerrm='JD literal work limit';
                  end;
                  begin
                    perform private.jd_relation_literals(repeat('Git ',1001),'git','ln');
                    raise exception 'Expected literal bound';
                  exception when sqlstate '22023' then
                    assert sqlerrm='JD literal work limit';
                  end;
                end $check$;
                select '300/301 and 1000/1001 work limits passed';
                '''
                result = logged('literal-limits.log', probe)
                print((logs / 'literal-limits.log').read_text(), flush=True)
                if result.returncode:
                    raise SystemExit(result.returncode)
        postflight = '''
        begin transaction read only;
        do $verify$
        declare signature text; relation text;
        begin
          assert private.document_page_count_ready();
          assert private.privacy_generation_ready();
          assert private.jd_clause_schema_ready();
          assert private.owned_ai_budget_export_ready();
          foreach signature in array array[
            'public.claim_analysis_job_v2(text,integer)',
            'public.persist_analysis_stage_v2(uuid,text,integer,uuid,text,jsonb)',
            'public.complete_deletion_request_with_admission_retirement(uuid,text,integer)'
          ] loop
            assert has_function_privilege('service_role',signature,'EXECUTE'), signature;
            assert not has_function_privilege('anon',signature,'EXECUTE'), signature;
            assert not has_function_privilege('authenticated',signature,'EXECUTE'), signature;
          end loop;
          foreach relation in array array[
            'public.canonical_resume_versions','public.resume_facts',
            'public.canonical_resume_heads','public.change_events','public.authoring_events'
          ] loop
            assert not has_table_privilege('service_role',relation,'DELETE'), relation;
            assert has_table_privilege('service_role',relation,'SELECT'), relation;
          end loop;
          assert not has_function_privilege('service_role','public.apply_jd_clause_order_action(uuid,uuid,jsonb,text)','EXECUTE');
        end $verify$;
        set local role service_role;
        select public.read_document_page_count_readiness();
        rollback;
        '''
        result = logged('postflight.log', postflight)
        print('After migration 91:', (logs / 'postflight.log').read_text().strip(), flush=True)
        if result.returncode:
            raise SystemExit(result.returncode)
        metadata = json.loads((logs / 'postflight.log').read_text())
        profile = (root / 'app/lib/server/document-page-count-readiness-profile.ts').read_text()
        expected = dict(re.findall(r'([A-Za-z][A-Za-z0-9]*):\s*"([^"]+)"', profile))
        assert len(expected) == 5
        assert metadata == {
            **expected, 'schemaReady': True, 'writerExecutable': True, 'fencedWriterExecutable': True,
        }
        print('The exact packet, worker readiness profile and catalogue ACLs passed.')
    finally:
        if (data / 'postmaster.pid').is_file():
            call('pg_ctl', '-w', '-D', data, '-m', 'immediate', 'stop', check=True, stdout=subprocess.DEVNULL)
