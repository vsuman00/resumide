type Slot = {
  id: string;
  updatedAt: string;
  contentSha256: string;
  versionId: string | null;
  revision: number;
  headId: string | null;
  projection: {
    summary: string | null;
    bullets: { original: string; rewrite: string; reasoning: string }[];
  };
};
async function store() {
  const f = await createRequirementContextFixture({ confirm: true });
  const selected = {
    canonicalVersionId: f.context.canonical.versionId,
    requirementVersionId: f.context.requirements.versionId,
  };
  const patch = buildTypedAuthoringPatch(
    f.context,
    {
      ...selected,
      clauseId: f.context.requirements.content.relations.clauses[0].id,
      targetFactId: f.targetFactId,
    },
    "Created",
  );
  const fact = f.context.canonical.content.facts.find(
    (v) => v.id === f.targetFactId,
  )!;
  let slot: Slot | null = null,
    publication: CanonicalRewritePublication | null = null,
    operation: CanonicalRewriteOperation | null = null,
    inserts = 0;
  const control = {
    initializerExecutable: true,
    schemaReady: true,
    writerExecutable: true,
    privacy: false,
    historical: false,
    staleAtWrite: false,
    ambiguousInsert: false,
    corruptCreated: false,
    profilePatch: {} as Record<string, unknown>,
    contextPatch: null as unknown,
  };
  const events: string[] = [];
  const context = () => {
    const value = structuredClone(f.context);
    if (control.historical) {
      value.heads.canonicalHeadId = id(991);
      value.heads.canonicalHeadRevision++;
    }
    return value;
  };
  const nativeContext = () => {
    if (!slot) throw Error("MISSING_FAKE_DATABASE_SLOT");
    const { projection, ...draft } = slot;
    return {
      schemaVersion: "canonical-rewrite-context/1.0.0",
      status: "available",
      relationshipContext: context(),
      draft,
      bindingSha256: "2".repeat(64),
      manifest: [
        {
          sourceId: 0,
          factId: fact.id,
          valueSha256: patch.target.currentValueSha256,
          location: {
            sectionId: patch.target.sectionId,
            entryId: patch.target.entryId,
            bulletId: patch.target.bulletId,
          },
          confirmation: fact.confirmation,
          basis: patch.support.basis,
          currentCitations: patch.sourceCitations,
          clauseIds: [patch.selection.clauseId],
          authoringVersion: patch.policies.authoringVersion,
        },
      ],
      saved: publication
        ? { state: "current", publication, legacyDraft: null }
        : {
            state: "legacy_unbound",
            publication: null,
            legacyDraft: projection,
          },
      readiness: {
        ...rewriteProfile,
        schemaReady: true,
        mutatorsExecutable: control.writerExecutable,
      },
    };
  };
  const state = () => {
    const readiness = {
      ...profile,
      schemaReady: control.schemaReady,
      initializerExecutable: control.initializerExecutable,
      ...control.profilePatch,
    };
    return slot
      ? {
          schemaVersion: "canonical-rewrite-workspace-state/1.0.0",
          status: "available",
          context: nativeContext(),
          readiness,
        }
      : {
          schemaVersion: "canonical-rewrite-workspace-state/1.0.0",
          status: "initialization_required",
          relationshipContext: control.contextPatch ?? context(),
          readiness,
        };
  };
  const seed = (summary: string | null = null) => {
    slot = {
      id: id(901),
      updatedAt: "2026-11-15T00:00:00.123456Z",
      contentSha256: "1".repeat(64),
      versionId: null,
      revision: 0,
      headId: null,
      projection: { summary, bullets: [] },
    };
  };
  const client = {
    rpc: vi.fn(async (name: string, args: Record<string, unknown>) => {
      events.push(name);
      if (
        args.p_user_id !== f.owned.ownerId ||
        args.p_analysis_id !== f.owned.analysisId ||
        control.privacy
      )
        return {
          data: null,
          error: { code: "42501", message: "PRIVATE_UNIT_OWNERSHIP" },
        };
      if (name === "read_canonical_rewrite_workspace_state")
        return { data: state(), error: null };
      if (name === "initialize_canonical_rewrite_workspace") {
        if (control.staleAtWrite || control.historical)
          return {
            data: null,
            error: { code: "P0C01", message: "PRIVATE_UNIT_STALE" },
          };
        if (!control.initializerExecutable)
          return { data: null, error: { code: "P0C03" } };
        const created = !slot;
        if (created) {
          seed();
          inserts++;
        }
        if (control.ambiguousInsert) {
          control.ambiguousInsert = false;
          throw Error("PRIVATE_UNIT_AMBIGUOUS_RESPONSE");
        }
        const result = state();
        if (control.corruptCreated && result.status === "available")
          result.context!.saved = {
            state: "legacy_unbound",
            publication: null,
            legacyDraft: {
              summary: "Unexpected nonempty content",
              bullets: [],
            },
          };
        return {
          data: {
            schemaVersion: "canonical-rewrite-workspace-initialization/1.0.0",
            created,
            state: result,
          },
          error: null,
        };
      }
      if (name === "read_canonical_rewrite_context")
        return slot
          ? { data: nativeContext(), error: null }
          : {
              data: null,
              error: { code: "P0002", message: "PRIVATE_UNIT_NO_SUMMARY" },
            };
      if (name === "prepare_canonical_rewrite") {
        const command = args.p_command as {
          request: NonNullable<typeof operation>["request"];
          policy: NonNullable<typeof operation>["policy"];
        };
        if (operation)
          return {
            data: {
              status: "prepared",
              operation: { ...operation, dispatchAllowed: false },
              context: null,
            },
            error: null,
          };
        operation = {
          schemaVersion: "canonical-rewrite-operation/1.0.0",
          operationId: command.request.operationId,
          ownerId: f.owned.ownerId,
          analysisId: f.owned.analysisId,
          request: command.request,
          policy: command.policy,
          bindingSha256: "2".repeat(64),
          state: "reserved",
          dispatchAllowed: false,
          publication: null,
        };
        return {
          data: { status: "prepared", operation, context: nativeContext() },
          error: null,
        };
      }
      if (name === "claim_canonical_rewrite_dispatch") {
        operation!.state = "dispatched";
        operation!.dispatchAllowed = true;
        return {
          data: { operation: { ...operation! }, budget: null },
          error: null,
        };
      }
      if (name === "settle_canonical_rewrite_attempt")
        return {
          data: {
            schemaVersion: "canonical-rewrite-settlement/1.0.0",
            operationId: operation!.operationId,
            bindingSha256: operation!.bindingSha256,
            usage: args.p_usage,
            state: operation!.state,
          },
          error: null,
        };
      if (name === "finish_canonical_rewrite") {
        const workspace = await loadCanonicalRewriteWorkspace(
          { rpc: async () => ({ data: nativeContext(), error: null }) },
          f.owned.analysisId,
          f.owned.ownerId,
          { ...selected, model: "unit-model" },
        );
        const result = reconstructCanonicalRewrite(
          workspace,
          operation!.operationId,
          args.p_plan,
        );
        publication = {
          result,
          receipt: {
            schemaVersion: "canonical-rewrite-receipt/1.0.0",
            state: "saved_proposals",
            operationId: operation!.operationId,
            bindingSha256: operation!.bindingSha256,
            runId: id(920),
            draftId: slot!.id,
            draftVersionId: id(921),
            revision: 1,
            draftUpdatedAt: "2026-11-15T00:00:01.654321Z",
            resultSha256: "3".repeat(64),
            projectionSha256: "4".repeat(64),
            idempotent: false,
          },
        };
        slot = {
          ...slot!,
          updatedAt: publication.receipt.draftUpdatedAt,
          contentSha256: publication.receipt.projectionSha256,
          versionId: publication.receipt.draftVersionId,
          headId: publication.receipt.draftVersionId,
          revision: 1,
          projection: {
            summary: null,
            bullets: result.bullets.map((b) => ({
              original: b.before,
              rewrite: b.after,
              reasoning: b.rationale,
            })),
          },
        };
        operation!.state = "saved";
        operation!.dispatchAllowed = false;
        operation!.publication = publication;
        return { data: publication, error: null };
      }
      throw Error("UNEXPECTED_BOOTSTRAP_RPC");
    }),
  };
  mocks.user.mockResolvedValue({ id: f.owned.ownerId });
  mocks.server.mockReturnValue({});
  mocks.admin.mockReturnValue(client);
  const request = (
    form: URLSearchParams | FormData = new URLSearchParams(selected),
    signal?: AbortSignal,
  ) =>
    new Request(
      `https://example.test/analysis/${f.owned.analysisId}/rewrite/workspace`,
      {
        method: "POST",
        headers: { Origin: "https://example.test" },
        body: form,
        signal,
      },
    );
  const post = () =>
    action({ request: request(), params: { id: f.owned.analysisId } });
  return {
    ...f,
    selected,
    client,
    state,
    control,
    events,
    seed,
    request,
    post,
    get slot() {
      return slot;
    },
    get inserts() {
      return inserts;
    },
  };
}

describe("actual owned bootstrap with real builders and fake native storage", () => {
  it("shows the old reader's missing-slot boundary without inventing a draft", async () => {
    const f = await store();
    await expect(
      loadCanonicalRewriteWorkspace(
        f.client,
        f.owned.analysisId,
        f.owned.ownerId,
        { ...f.selected, model: "unit-model" },
      ),
    ).rejects.toMatchObject({ reason: "not_found" });
    expect(f.slot).toBeNull();
    expect(f.inserts).toBe(0);
  });
  it("reads an honest missing state with no writer mutation or fake draft tokens", async () => {
    const f = await store(),
      state = await loadCanonicalRewriteWorkspaceState(
        f.client,
        f.owned.analysisId,
        f.owned.ownerId,
        { ...f.selected, model: "unit-model" },
      );
    expect(state).toMatchObject({
      status: "initialization_required",

        source: f.context.source,
        canonical: { versionId: f.selected.canonicalVersionId },

      canInitialize: true,

    expect(f.inserts).toBe(0);

  it("initializes with AI off and returns actual empty tokens without generation capability", async () => {
    const f = await store(),
      response = await f.post(),
      result = await response.json();
    expect(response.status).toBe(200);
    expect(result).toMatchObject({
      kind: "workspace_initialized",
      created: true,
      workspace: {
        binding: {
          draft: {
            id: f.slot!.id,
            updatedAt: f.slot!.updatedAt,
            contentSha256: f.slot!.contentSha256,
            revision: 0,
            versionId: null,
            headId: null,
          },
        },
        saved: {
          state: "legacy_unbound",
          publication: null,
          legacyDraft: { summary: null, bullets: [] },

      },
    });
    expect(f.inserts).toBe(1);
    expect(f.events).toEqual([
      "read_canonical_rewrite_workspace_state",
      "initialize_canonical_rewrite_workspace",

    expect(response.headers.get("Cache-Control")).toBe("private, no-store");
  });

      f.control.writerExecutable = false;

        f.owned.analysisId,
        f.owned.ownerId,

      expect(result).toMatchObject({
        created: false,
        workspace: {
          canGenerate: false,
          binding: { draft: { id: before!.id, updatedAt: before!.updatedAt } },
        },
      });
      expect(f.slot).toEqual(before);
      expect(f.inserts).toBe(0);
      expect(f.events).toEqual(["read_canonical_rewrite_workspace_state"]);
    },
  );
  it("retries an ambiguous insert as an existing current row with no duplicate or automatic generation", async () => {
    const f = await store();
    f.control.ambiguousInsert = true;
    await expect(
      initializeCanonicalRewriteWorkspace(
        f.client,
        f.owned.analysisId,
        f.owned.ownerId,
        f.selected,
        { model: "unit-model" },
      ),
    ).rejects.toMatchObject({ reason: "unavailable", recovery: "retry_exact" });
    const first = structuredClone(f.slot);
    const retry = await initializeCanonicalRewriteWorkspace(
      f.client,
      f.owned.analysisId,
      f.owned.ownerId,
      f.selected,
      { model: "unit-model" },
    );
    expect(retry.created).toBe(false);
    expect(f.slot).toEqual(first);
    expect(f.inserts).toBe(1);
    expect(
      f.events.filter((n) => n === "initialize_canonical_rewrite_workspace"),
    ).toHaveLength(1);
  });
  it("uses real initialization tokens for a separately consented existing generation protocol and exact replay", async () => {
    const f = await store(),
      before = JSON.stringify(f.context.canonical.content),
      init = await initializeCanonicalRewriteWorkspace(
        f.client,
        f.owned.analysisId,
        f.owned.ownerId,
        f.selected,
        { model: "unit-model" },
      );
    const command = createCanonicalRewriteRequest(init.workspace, id(922)),
      authorize = vi.fn(async () => {}),
      execute = vi.fn(async () => ({
        output: {
          summarySourceId: null,
          bullets: [
            {
              sourceId: 0,
              clauseId: init.workspace.clauses[0].clauseId,
              operation: "equivalent_action",
              replacementAction: "Created",
            },
          ],
        },
        usage: { inputTokens: 8, outputTokens: 3 },
      }));
    expect(command.draftId).toBe(f.slot!.id);
    expect(command.expectedDraftUpdatedAt).toBe(f.slot!.updatedAt);
    expect(execute).not.toHaveBeenCalled();
    const options = {
      model: "unit-model",
      expectedProfile: rewriteProfile,
      getTransport: () => ({ execute }),
      timeoutMs: 1000,
      authorize,
    };

      f.client,
      f.owned.analysisId,
      f.owned.ownerId,
      command,
      options,
    );
    expect(saved.kind).toBe("saved");

        f.client,

        f.owned.ownerId,

    expect(execute).toHaveBeenCalledTimes(1);

    expect(JSON.stringify(f.context.canonical.content)).toBe(before);
    const slot = structuredClone(f.slot),

        f.owned.analysisId,
        f.owned.ownerId,
        f.selected,
        { model: "unit-model" },

    expect(existing.workspace.saved.state).toBe("current");
    expect(f.slot).toEqual(slot);
  });
  it.each(["missing", "existing"])(

        f.client,
        f.owned.analysisId,

    ).rejects.toMatchObject({ reason: "unavailable" });
    expect(f.inserts).toBe(0);
    expect(f.events).not.toContain("initialize_canonical_rewrite_workspace");
  });
  it("refuses a forged created=true nonempty projection while preserving exact retry recovery", async () => {
    const f = await store();
    f.control.corruptCreated = true;
    await expect(
      initializeCanonicalRewriteWorkspace(
        f.client,
        f.owned.analysisId,
        f.owned.ownerId,

      (
        await initializeCanonicalRewriteWorkspace(
          f.client,
          f.owned.analysisId,
          f.owned.ownerId,
          f.selected,
          { model: "unit-model" },
        )
      ).created,
    ).toBe(false);
    expect(f.inserts).toBe(1);
  });
});

describe("bounded authenticated initialization resource", () => {
  it.each(["urlencoded", "multipart"])(
    "admits exact two UUID fields in %s",
    async (kind) => {
      const f = await store(),
        form =
          kind === "urlencoded"
            ? new URLSearchParams(f.selected)
            : new FormData();
      if (form instanceof FormData)
        for (const [k, v] of Object.entries(f.selected)) form.set(k, v);
      expect(
        await readCanonicalRewriteWorkspaceSelection(f.request(form)),
      ).toEqual(f.selected);
    },
  );
  it.each(["duplicate", "unknown", "file", "malformed", "uppercase"])(
    "refuses %s fields before admin IO",
    async (kind) => {
      const f = await store();
      let form: URLSearchParams | FormData = new URLSearchParams(f.selected);
      if (kind === "duplicate")
        form.append("canonicalVersionId", f.selected.canonicalVersionId);
      if (kind === "unknown") form.set("draftId", id(701));
      if (kind === "malformed") form.set("canonicalVersionId", "not-an-id");
      if (kind === "uppercase")
        form.set("canonicalVersionId", "A0000000-0000-4000-8000-000000000001");
      if (kind === "file") {
        form = new FormData();
        form.set(
          "canonicalVersionId",
          new Blob([f.selected.canonicalVersionId]),
          "id.txt",
        );
        form.set("requirementVersionId", f.selected.requirementVersionId);
      }
      expect(
        (
          await action({
            request: f.request(form),
            params: { id: f.owned.analysisId },
          })
        ).status,
      ).toBe(422);
      expect(mocks.admin).not.toHaveBeenCalled();
    },
  );
  it("refuses encoded overflow and stalled/cancel-rejecting bodies without waiting for cleanup", async () => {
    const f = await store();

    oversized.set("padding", "x".repeat(4097));
    expect(
      (
        await action({
          request: f.request(oversized),
          params: { id: f.owned.analysisId },
        })
      ).status,
    ).toBe(413);
    expect(mocks.admin).not.toHaveBeenCalled();
    vi.useFakeTimers();
    const cancel = vi.fn(() => Promise.reject(Error("PRIVATE_UNIT_CANCEL")));
    const stream = new ReadableStream<Uint8Array>(
      { pull: () => new Promise(() => {}), cancel },
      { highWaterMark: 0 },
    );
    const request = new Request("https://example.test", {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body: stream,
      duplex: "half",
    } as RequestInit & { duplex: "half" });
    const pending = readCanonicalRewriteWorkspaceSelection(request, 5).catch(

    );

      reason: "unavailable",
      recovery: "retry_exact",
    });
    expect(cancel).toHaveBeenCalledTimes(1);
    expect(stream.locked).toBe(false);
  });
