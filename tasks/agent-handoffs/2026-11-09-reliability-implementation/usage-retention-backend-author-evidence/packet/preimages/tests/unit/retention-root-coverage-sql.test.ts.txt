type Patch = {
  signature: string;
  before: string;
  after: string;
  old: string;
  new: string;
};
function block(s: string, n: string) {
  const m = new RegExp(`\\$${n}\\$([\\s\\S]*?)\\$${n}\\$`).exec(s);
  if (!m) throw Error("MISSING_DELIMITED_SOURCE");
  return m[1];
}
function body(s: string, n: string) {
  const escaped = n.replace(/[.*+?^${}()|[\]\\]/gu, "\\$&");
  const m = new RegExp(
    `create(?: or replace)? function ${escaped}\\([^;]*?\\bas\\s+(\\$[A-Za-z0-9_]*\\$)([\\s\\S]*?)\\1;`,
  ).exec(s);
  if (!m) throw Error("MISSING_SOURCE_BODY");
  return m[2];
}
const originalPatches = JSON.parse(
  block(clock, "disposition_patches"),

function original(n: string) {
  const p = originalPatches.find((p) => p.signature.startsWith(n + "("));
  if (p) return p.new;
  if (n === "private.owned_ai_disposition_ready") return body(clock, n);
  return body(
    read(
      n === "public.fail_retention_item"
        ? "supabase/migrations/20260911110000_phase7_security_privacy_operations.sql"
        : "supabase/migrations/20261115100607_owned_ai_retirement.sql",
    ),
    n,
  );
}
function current(n: string) {
  return (
    patches.find((p) => p.signature.startsWith(n + "("))?.new ?? original(n)
  );
}

type Item = {
  key: string;
  state: "queued" | "failed" | "deleted";
  attempts: number;
  createdAt: number;
  completedAt: number | null;
};
// Independent target protocol oracle. It does NOT interpret or execute SQL.
function protocol(versions: number, deleted = false, existing: Item[] = []) {
  const inventory = Array.from(
      { length: versions },
      (_, n) => `owned-version-${n}`,
    ),
    items = new Map(existing.map((i) => [i.key, { ...i }]));
  let rootDeleted = deleted,
    rootGone = false,
    deadline: number | null = null;
  const covered = () =>
    inventory.every((key) => items.get(key)?.state === "deleted") &&
    [...items.keys()].every((key) => inventory.includes(key));
  return {
    items,
    inventory,
    get rootGone() {
      return rootGone;
    },
    get deadline() {
      return deadline;
    },
    claim(limit: number, now: number) {
      if (rootDeleted && items.size === 0) throw Error("FIRST_CLOCK_REPAIR");
      for (const key of inventory.filter((k) => !items.has(k)).slice(0, limit))
        items.set(key, {
          key,
          state: "queued",
          attempts: 0,
          createdAt: now,
          completedAt: null,
        });
      const work = [...items.values()]
        .filter((i) => i.state !== "deleted")
        .sort(
          (a, b) =>
            a.attempts - b.attempts ||
            a.createdAt - b.createdAt ||
            a.key.localeCompare(b.key),
        )
        .slice(0, limit);
      for (const i of work) {
        i.state = "queued";
        i.attempts++;
      }
      if (work.length) {
        rootDeleted = true;
        deadline ??=
          Math.min(...[...items.values()].map((i) => i.createdAt)) +
          720 * 60 * 60 * 1000;
      }
      return work;
    },
    complete(key: string, now: number) {
      const i = items.get(key);
      if (!i) return false;
      if (i.state === "deleted") return true;
      i.state = "deleted";
      i.completedAt = now;
      if (covered()) rootGone = true;
      return true;
    },
    fail(key: string) {
      const i = items.get(key);
      if (!i || i.state === "deleted") return false;
      i.state = "failed";
      return true;
    },
  };
}
describe("bounded chunk protocol oracle, not SQL/race/physical-erasure proof", () => {
  it("drains more than 1000 versions through p_limit=1 without a history cap or early root deletion", () => {
    const p = protocol(1003);
    let firstDeadline: number | null = null;
    for (let n = 0; n < 1003; n++) {
      const before = p.items.size,
        work = p.claim(1, 100 + n);
      expect(work).toHaveLength(1);
      expect(p.items.size - before).toBeLessThanOrEqual(1);
      firstDeadline ??= p.deadline;
      expect(p.deadline).toBe(firstDeadline);
      expect(p.complete(work[0].key, 2000 + n)).toBe(true);
      expect(p.rootGone).toBe(n === 1002);
    }
    expect([...p.items.values()].every((i) => i.state === "deleted")).toBe(
      true,
    );
  });
  it("keeps missing inventory reachable after every currently enrolled item completes", () => {
    const p = protocol(5),
      first = p.claim(1, 10)[0];
    p.complete(first.key, 11);
    expect(p.rootGone).toBe(false);
    const next = p.claim(2, 20);
    expect(next).toHaveLength(2);
    expect(p.items.size).toBe(3);
    expect(next.every((i) => i.key !== first.key)).toBe(true);
  });
  it("preserves the earliest deleted-item clock during backfill and rejects an unproved first event", () => {
    const prior: Item = {
      key: "owned-version-0",
      state: "deleted",
      attempts: 1,
      createdAt: 7,
      completedAt: 8,
    };
    const p = protocol(3, true, [prior]);
    p.claim(1, 100);
    expect(p.deadline).toBe(7 + 720 * 60 * 60 * 1000);
    expect(p.items.get(prior.key)).toEqual(prior);
    const unknown = protocol(3, true);
    expect(() => unknown.claim(1, 100)).toThrow("FIRST_CLOCK_REPAIR");
    expect(unknown.items.size).toBe(0);
  });
  it("keeps deletion terminal when a stale worker fails after another completes the last item", () => {
    const p = protocol(1),
      work = p.claim(1, 1)[0];
    const staleSnapshot = { ...work };
    expect(staleSnapshot.state).toBe("queued");
    p.complete(work.key, 2);
    expect(p.rootGone).toBe(true);
    expect(p.complete(staleSnapshot.key, 999)).toBe(true);
    expect(p.fail(staleSnapshot.key)).toBe(false);
    expect(p.items.get(work.key)?.completedAt).toBe(2);
  });
});
