"""Verify request-scoped upload cleanup receipts on socket-only PostgreSQL 17."""
import ast
from pathlib import Path
import subprocess
import sys
from tempfile import TemporaryDirectory

migrations = Path("supabase/migrations")

def source(prefix):
    file, = migrations.glob(prefix + "_*.sql")
    return file.read_text()

def section(text, start, end):
    start_at = text.index(start)
    return text[start_at:text.index(end, start_at)]

previous = Path(__file__).with_name("admission-cleanup-recovery-native-fixture.py.txt")
scaffold = next(
    ast.literal_eval(node.value)
    for node in ast.parse(previous.read_text()).body
    if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == "scaffold" for t in node.targets)
)
scaffold += """
alter table public.data_requests add column next_attempt_at timestamptz, add column last_error_code text;
create table public.privacy_request_steps(request_id uuid, status text, attempt_count integer, last_error_code text);
"""

checks = """
do $$
declare
  owner_id uuid:=gen_random_uuid();
  foreign_id uuid:=gen_random_uuid();
  org uuid:=gen_random_uuid();
  other_org uuid:=gen_random_uuid();
  claim_id uuid:=gen_random_uuid();
  operation_id uuid;
  upload_id uuid;
  resume_id uuid;
  version_id uuid;
  object_id uuid;
  path text;
  historical uuid;
  historical_object uuid;
  fresh uuid;
  fresh_object uuid;
  fresh_key text;
  foreign_ref uuid:=gen_random_uuid();
  plan jsonb;
  first_verified_at timestamptz;
begin
  insert into auth.users values(owner_id),(foreign_id);
  insert into public.organizations values(org,owner_id),(other_org,foreign_id);
  insert into public.data_requests(id,user_id,kind,status,lease_owner,lease_expires_at,attempt_count,privacy_generation)
    values(claim_id,owner_id,'deletion','processing','worker',clock_timestamp()+interval '2 minutes',1,1);

  for kind in 1..6 loop
    operation_id:=gen_random_uuid();
    upload_id:=gen_random_uuid();
    resume_id:=gen_random_uuid();
    version_id:=gen_random_uuid();
    object_id:=gen_random_uuid();
    path:='organizations/'||(case when kind=6 then other_org else org end)||
      '/resumes/'||resume_id||'/versions/'||version_id||'.pdf';
    insert into private.analysis_admission_operations(id,organization_id,owner_id,idempotency_key,binding_sha256)
      values(operation_id,case when kind=6 then other_org else org end,case when kind=6 then foreign_id else owner_id end,
        gen_random_uuid(),repeat('a',64));
    insert into private.analysis_admission_uploads(id,operation_id,analysis_id,resume_id,version_id,job_id,storage_key)
      values(upload_id,operation_id,gen_random_uuid(),resume_id,version_id,gen_random_uuid(),path);
    if kind<>3 then
      insert into private.analysis_admission_upload_transport(upload_id,bucket,dispatched_at,acknowledged_at,object_id)
        values(upload_id,'resumes',clock_timestamp(),case when kind<>4 then clock_timestamp() end,case when kind<>4 then object_id end);
    end if;
    if kind in (1,3,4,6) then
      update private.analysis_admission_uploads set state='cleanup_pending' where id=upload_id;
      update private.analysis_admission_uploads set state='cleaned' where id=upload_id;
    elsif kind=5 then
      update private.analysis_admission_uploads set state='adopted' where id=upload_id;
      update private.analysis_admission_operations set adopted_upload_id=upload_id where id=operation_id;
    end if;
    if kind=1 then historical:=upload_id; historical_object:=object_id; end if;
    if kind=2 then fresh:=upload_id; fresh_object:=object_id; fresh_key:=path; end if;
  end loop;

  assert not public.complete_analysis_admission_cleanup(claim_id,'worker',1,historical,historical_object),
    'Unissued historical cleanup was accepted';
  assert not has_table_privilege('service_role','private.analysis_admission_cleanup_receipts','INSERT,UPDATE,DELETE');
  assert not has_table_privilege('authenticated','private.analysis_admission_cleanup_receipts','SELECT');
  assert not has_function_privilege('authenticated','public.complete_analysis_admission_cleanup(uuid,text,integer,uuid,uuid)','execute');

  plan:=public.claim_analysis_admission_cleanup(claim_id,'worker',1);
  assert jsonb_array_length(plan->'uploads')=2;
  assert exists(select 1 from jsonb_array_elements(plan->'uploads') where value->>'uploadId'=historical::text);
  assert exists(select 1 from jsonb_array_elements(plan->'uploads') where value->>'uploadId'=fresh::text);
  assert (select count(*)=2 from private.analysis_admission_cleanup_receipts where request_id=claim_id and verified_at is null);
  assert not public.complete_analysis_admission_cleanup(claim_id,'worker',1,historical,gen_random_uuid());
  assert public.complete_analysis_admission_cleanup(claim_id,'worker',1,historical,historical_object);
  select verified_at into strict first_verified_at
    from private.analysis_admission_cleanup_receipts receipt where receipt.request_id=claim_id and receipt.upload_id=historical;
  assert first_verified_at is not null;

  update public.data_requests set attempt_count=2 where id=claim_id;
  assert not public.complete_analysis_admission_cleanup(claim_id,'worker',1,fresh,fresh_object);
  assert not public.complete_analysis_admission_cleanup(claim_id,'worker',2,fresh,fresh_object);
  assert public.complete_analysis_admission_cleanup(claim_id,'worker',2,historical,historical_object);
  assert (select last_progress_attempt=1 from private.analysis_admission_cleanup_progress where request_id=claim_id);
  assert (select issued_attempt=1 and verified_at=first_verified_at
    from private.analysis_admission_cleanup_receipts receipt where receipt.request_id=claim_id and receipt.upload_id=historical);

  plan:=public.claim_analysis_admission_cleanup(claim_id,'worker',2);
  assert jsonb_array_length(plan->'uploads')=1 and plan#>>'{uploads,0,uploadId}'=fresh::text;
  insert into public.resume_versions values(foreign_ref,foreign_id,other_org,gen_random_uuid(),fresh_key,31,repeat('b',64));
  assert not public.complete_analysis_admission_cleanup(claim_id,'worker',2,fresh,fresh_object);
  assert (select verified_at is null from private.analysis_admission_cleanup_receipts receipt where receipt.request_id=claim_id and receipt.upload_id=fresh);
  delete from public.resume_versions where id=foreign_ref;

  assert public.complete_analysis_admission_cleanup(claim_id,'worker',2,fresh,fresh_object);
  assert public.claim_analysis_admission_cleanup(claim_id,'worker',2)->'uploads'='[]'::jsonb;
  assert not public.defer_analysis_admission_cleanup(claim_id,'worker',2);
  assert (select issued_attempt=2 and verified_at is not null
    from private.analysis_admission_cleanup_receipts receipt where receipt.request_id=claim_id and receipt.upload_id=fresh);
  assert (select last_progress_attempt=2 from private.analysis_admission_cleanup_progress where request_id=claim_id);
  assert (select count(*)=2 from private.analysis_admission_cleanup_receipts);
  assert (select count(*)=6 from private.analysis_admission_operations);
  assert (select status='processing' from public.data_requests where id=claim_id);
  plan:=private.analysis_admission_export_records(owner_id);
  assert jsonb_array_length(plan#>'{analysisAdmissionRetirementCaptures,0,storageAbsenceReceipts}')=2;
  assert private.analysis_admission_export_records(foreign_id)->'analysisAdmissionRetirementCaptures'='[]'::jsonb;
end;
$$;
"""

versions = ["20261121130718", "20261121132235", "20261121203700", "20261121205500"]
if "--baseline" not in sys.argv:
    versions.append("20261121211200")
sql = "\n".join([
    scaffold,
    section(source("20261110080809"),"create table private.analysis_admission_operations","create function private.analysis_admission_denied"),
    section(source("20261109164851"),"create function private.candidate_owner_lock_keys","-- A false try-lock"),
    section(source("20261116075250"),"create function private.privacy_generation_claim","create function public.complete_privacy_request"),
    *(source(version) for version in versions), checks,
])

with TemporaryDirectory(prefix="resumide-admission-receipts-") as tmp:
    env = {"HOME": tmp, "PATH": "/opt/homebrew/bin:/usr/bin:/bin", "LC_ALL": "C"}
    data = str(Path(tmp) / "data")

    def run(argv, **options):
        return subprocess.run(argv, env=env, timeout=25, text=True, check=True, **options)

    assert run(["postgres", "--version"], capture_output=True).stdout.startswith("postgres (PostgreSQL) 17.")
    run(["initdb", "-D", data, "--no-sync", "--auth=trust", "-U", "fixture_owner"], stdout=subprocess.DEVNULL)
    try:
        run(["pg_ctl","-w","-D",data,"-l",str(Path(tmp)/"postgres.log"),
             "-o",f"-c listen_addresses='' -c unix_socket_directories={tmp} -c unix_socket_permissions=0700","start"],
            stdout=subprocess.DEVNULL)
        run(["psql","-X","-v","ON_ERROR_STOP=1","-h",tmp,"-U","fixture_owner","-d","postgres","-Atq"], input=sql)
        print("PASS: historical recheck, live-claim issuance, exact result, stable replay, reference refusal, and owner receipt export.")
    finally:
        if (Path(data) / "postmaster.pid").exists():
            run(["pg_ctl","-w","-D",data,"-m","immediate","stop"], stdout=subprocess.DEVNULL)
