const privateHeaders = () =>
  new Headers({
    "Cache-Control": "private, no-store",
    Pragma: "no-cache",
    "X-Content-Type-Options": "nosniff",
    Vary: "Cookie, Origin",
  });
function secure(headers: Headers) {
  for (const [key, value] of privateHeaders()) headers.set(key, value);
}
function failure(
  headers: Headers,
  code: string,
  message: string,
  status: number,
) {
  secure(headers);
  return Response.json({ error: message, code }, { status, headers });
}
let active = 0;
async function readInput(request: Request): Promise<unknown> {
  const max = BUILD_AI_LIMITS.sourceBytes * 6 + 2048;
  const length = request.headers.get("Content-Length");
  if (length !== null && (!/^\d+$/.test(length) || Number(length) > max))
    throw new BuildAssistError("invalid_request");
  if (!request.body) throw new BuildAssistError("invalid_request");
  const reader = request.body.getReader(),
    deadline = AbortSignal.any([request.signal, AbortSignal.timeout(5000)]);
  let bytes = 0;
  const chunks: Uint8Array[] = [];
  const cancel = () => {
    void reader.cancel().catch(() => undefined);
  };
  deadline.addEventListener("abort", cancel, { once: true });
  try {
    if (deadline.aborted) {
      cancel();
      throw new BuildAssistError("cancelled");
    }
    for (;;) {
      const { done, value } = await reader.read();
      if (deadline.aborted)
        throw new BuildAssistError(
          request.signal.aborted ? "cancelled" : "invalid_request",
        );
      if (done) break;
      bytes += value.byteLength;
      if (bytes > max) {
        cancel();
        throw new BuildAssistError("invalid_request");
      }
      chunks.push(value);
    }
  } finally {
    deadline.removeEventListener("abort", cancel);
    reader.releaseLock();
  }
  try {
    return JSON.parse(
      new TextDecoder("utf-8", { fatal: true }).decode(Buffer.concat(chunks)),
    );
  } catch {
    throw new BuildAssistError("invalid_request");
  }
}
export async function buildSuggestAction(request: Request): Promise<Response> {
  const headers = privateHeaders();
  const requestId = requestIdFrom(request);
  headers.set("X-Request-Id", requestId);
  if (request.method !== "POST") {
    headers.set("Allow", "POST");
    return failure(
      headers,
      "method_not_allowed",
      "Use POST to request a suggestion.",
      405,
    );
  }
  let admitted = false;
  try {
    const config = getServerConfig();
    const user = await getAuthenticatedUser(
      createSupabaseServerClient(request, headers),
    );
    if (!user)
      return failure(
        headers,
        "unauthenticated",
        "Sign in to ask for a suggestion.",
        401,
      );
    if (
      !config.LATEX_CONVERTER_ENABLED ||
      !config.BUILD_AI_ASSIST_ENABLED ||
      !config.AI_ENABLED
    )
      throw new BuildAssistError("disabled");
    const origin = new URL(config.APP_ORIGIN).origin;
    if (
      request.headers.get("Origin") !== origin ||
      new URL(request.url).origin !== origin ||
      (request.headers.get("Sec-Fetch-Site") &&
        request.headers.get("Sec-Fetch-Site") !== "same-origin")
    )
      return failure(headers, "forbidden", "Open Build from this site.", 403);
    if (
      request.headers
        .get("Content-Type")
        ?.split(";")[0]
        .trim()
        .toLowerCase() !== "application/json"
    )
      return failure(
        headers,
        "invalid_request",
        "Send the selected source as JSON.",
        415,
      );
    const prepared = validateBuildAssistRequest(await readInput(request));
    const idempotencyKey = request.headers.get("Idempotency-Key");
    if (
      config.AI_BUDGET_ENABLED &&
      idempotencyKey !== null &&
      !z.uuid().safeParse(idempotencyKey).success
    )
      throw new BuildAssistError("invalid_request");
    if (request.signal.aborted) throw new BuildAssistError("cancelled");
    if (
      config.AI_BUDGET_ENABLED !== true ||
      typeof config.AI_BUDGET_POLICY_VERSION !== "string" ||
      !config.AI_BUDGET_POLICY_VERSION.trim()
    )
      throw new BuildAssistError("provider_unavailable");
    if (active >= BUILD_AI_LIMITS.concurrency)
      throw new BuildAssistError("busy");
    active++;
    admitted = true;
    const admin = createSupabaseAdminClient();
    const membership = await admin
      .from("memberships")
      .select("organization_id")
      .eq("user_id", user.id)
      .eq("role", "owner")
      .order("created_at", { ascending: true })
      .limit(1)
      .maybeSingle();
    if (membership.error || !membership.data)
      return failure(
        headers,
        "provider_unavailable",
        "AI suggestions are unavailable.",
        503,
      );
    const organizationId = membership.data.organization_id;
    for (const [key, limit] of [
      [
        `build-ai:user:${createHash("sha256").update(user.id).digest("hex")}`,
        3,
      ],
      ["build-ai:global:v1", 20],
    ] as const) {
      const result = await admin.rpc("consume_analysis_rate_limit", {
        p_key: key,
        p_limit: limit,
        p_window_seconds: 60,
      });
      if (result.error) throw new BuildAssistError("provider_unavailable");
      if (result.data !== true) {
        headers.set("Retry-After", "60");
        return failure(
          headers,
          "rate_limited",
          "Too many suggestion requests. Try again later.",
          429,
        );
      }
    }
    const consent = await admin.from("consents").upsert(
      {
        organization_id: organizationId,
        user_id: user.id,
        purpose: "qualitative_ai",
        policy_version: BUILD_AI_POLICY_VERSION,
        captured_at: new Date().toISOString(),
      },
      { onConflict: "user_id,purpose,policy_version" },
    );
    if (consent.error) throw new BuildAssistError("provider_unavailable");
    const audit = await admin.from("audit_events").insert({
      organization_id: organizationId,
      actor_id: user.id,
      action: "build_ai.consent_recorded",
      target_type: "consent",
      outcome: "success",
      request_id: config.AI_BUDGET_ENABLED ? requestId : randomUUID(),
    });
    if (audit.error) throw new BuildAssistError("provider_unavailable");
    if (request.signal.aborted) throw new BuildAssistError("cancelled");
    const execute = (onObservedUsage?: (usage: AiBudgetUsage | null) => void) =>
      executeBuildAssist(prepared, {
        apiKey: config.OPENAI_API_KEY,
        baseUrl: config.OPENAI_BASE_URL,
        model: config.OPENAI_MODEL,
        signal: request.signal,
        onObservedUsage,
        onUsage: (usage) =>
          recordProviderUsage({
            provider: "openai",
            model: config.OPENAI_MODEL,
            inputTokens: usage.inputTokens,
            outputTokens: usage.outputTokens,
            inputCostPer1kUsd: config.AI_INPUT_COST_PER_1K_USD,
            outputCostPer1kUsd: config.AI_OUTPUT_COST_PER_1K_USD,
          }),
      });
    const result = await runBudgetedAiAttempt({
      client: createSourceBoundAiBudgetClient(
        admin,
        {
          userId: user.id,
          organizationId,
          source: {
            schemaVersion: "ai-budget-source/2.0.0",
            kind: "build_account",
            requestNonce: idempotencyKey ?? requestId,
          },
        },
        AI_BUDGET_SOURCE_PROFILE,
        { signal: request.signal },
      ),
      binding: (() => {
        const operationId = createSourceBoundAiBudgetOperationId({
          userId: user.id,
          organizationId,
          source: {
            schemaVersion: "ai-budget-source/2.0.0",
            kind: "build_account",
            requestNonce: idempotencyKey ?? requestId,
          },
        });
        return {
          userId: user.id,
          organizationId,
          operationId,
          attemptId: createAiBudgetAttemptId({
            operationId,
            attemptNonce: "http-post/1",
          }),
          surface: "build" as const,
          payloadSha256: createHash("sha256")
            .update(
              JSON.stringify({
                sourceHash: prepared.sourceHash,
                startLine: prepared.startLine,
                endLine: prepared.endLine,
                mode: prepared.mode,
                policyVersion: prepared.policyVersion,
                model: config.OPENAI_MODEL,
                budgetPolicyVersion: config.AI_BUDGET_POLICY_VERSION,
              }),
            )
            .digest("hex"),
          model: config.OPENAI_MODEL,
          promptPolicyVersion: BUILD_AI_POLICY_VERSION,
          policyVersion: config.AI_BUDGET_POLICY_VERSION,
          reservedInputTokens:
            BUILD_AI_LIMITS.reservedTokens - BUILD_AI_LIMITS.outputTokens,
          reservedOutputTokens: BUILD_AI_LIMITS.outputTokens,
        };
      })(),
      signal: request.signal,
      execute: async (recordUsage) => {
        const value = await execute(recordUsage);
        if (value.usage === null) throw new BuildAssistError("output_invalid");
        return { value, usage: value.usage };
      },
    });
    secure(headers);
    return Response.json(result.response, { headers });
  } catch (error) {
    if (error instanceof AiBudgetError) {
      if (
        error.code === "AI_BUDGET_EXHAUSTED" ||
        error.code === "AI_BUDGET_BLOCKED"
      )
        return failure(
          headers,
          "budget_exhausted",
          "The available AI token budget is exhausted.",
          429,
        );
      if (
        error.code === "AI_BUDGET_CONFLICT" ||
        error.code === "AI_BUDGET_STATE"
      )
        return failure(
          headers,
          "request_conflict",
          "This suggestion request cannot be reused. Start a new request.",
          409,
        );
      if (error.code === "AI_BUDGET_DENIED")
        return failure(
          headers,
          "forbidden",
          "Suggestion access changed. Reload Build before trying again.",
          403,
        );
    }
    if (error instanceof QualitativeAiFailure && error.code === "AI_CANCELLED")
      return failure(
        headers,
        "cancelled",
        "The AI request was cancelled.",
        409,
      );
    if (error instanceof BuildAssistError) {
      const status =
        error.code === "disabled"
          ? 404
          : error.code === "consent_required"
            ? 403
            : error.code === "busy" || error.code === "budget_exhausted"
              ? 429
              : error.code === "stale_source" || error.code === "cancelled"
                ? 409
                : error.code === "provider_unavailable"
                  ? 503
                  : error.code === "timeout"
                    ? 504
                    : error.code === "output_invalid"
                      ? 422
                      : 400;
      return failure(headers, error.code, error.message, status);
    }
    return failure(
      headers,
      "provider_unavailable",
      "AI suggestions are unavailable. No edit was applied.",
      503,
    );
  } finally {
    if (admitted) active--;
  }
}
