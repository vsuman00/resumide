function resolveAssociationMember(
  member: EntryAssociationMember,
  targets: ReviewModel["targets"],
): Observation | null {
  const observation = targets.observations.find(
    (item) => item.id === member.observationId,
  );
  const target = targets.targets.find((item) => item.id === member.targetId);
  if (
    !observation ||
    !target ||
    observation.assertionId !== member.assertionId ||
    observation.targetId !== target.id ||
    observation.field !== member.field ||
    target.field !== member.field ||
    target.slotId !== member.assertionId ||
    !target.observationIds.includes(observation.id) ||
    observation.citations.length !== member.citations.length ||
    !member.citations.every((citation, index) => {
      const source = observation.citations[index];
      return (
        source !== undefined &&
        citation.spanId === source.spanId &&
        citation.pageId === source.pageId &&
        citation.method === source.method &&
        citation.start === source.start &&
        citation.end === source.end &&
        citation.pageStart === source.pageStart &&
        citation.pageEnd === source.pageEnd
      );
    })
  )
    return null;
  return observation;
}

function SourceObservation({ observation }: { observation: Observation }) {
  return (
    <div>
      <dt>{observationFieldLabels[observation.field]}</dt>
      <dd>
        {observation.value}
        <ul>
          {observation.citations.map((citation, index) => (
            <li key={`${citation.spanId}-${index}`}>
              <a href={`#source-${encodeURIComponent(citation.spanId)}`}>
                {citation.method === "local_ocr"
                  ? "View cited OCR text"
                  : "View cited PDF text"}
              </a>
            </li>
          ))}
        </ul>
      </dd>
    </div>
  );
}

function EntryCandidate({
  group,
  targets,
}: {
  group: EntryAssociationGroup | ExistingEntryGroup;
  targets: ReviewModel["targets"];
}) {
  const members: EntryAssociationMember[] = [];
  if ("members" in group) members.push(...group.members);
  else {
    for (const [index, observationId] of group.observationIds.entries()) {
      const observation = targets.observations.find(
        (item) => item.id === observationId,
      );
      if (observation)
        members.push({
          observationId,
          assertionId: observation.assertionId,
          targetId: group.targetIds[index],
          field: observation.field,
          citations: observation.citations,
        });
    }
  }
  const observations: Observation[] = [];
  let resolved =
    members.length === group.observationIds.length &&
    members.length === group.targetIds.length;
  for (const [index, member] of members.entries()) {
    const observation = resolveAssociationMember(member, targets);
    if (
      !observation ||
      member.observationId !== group.observationIds[index] ||
      member.targetId !== group.targetIds[index] ||
      !observation.citations.every(
        (citation) =>
          citation.pageId === group.pageId &&
          citation.method === group.method &&
          (!("spanId" in group) || citation.spanId === group.spanId),
      )
    )
      resolved = false;
    else observations.push(observation);
  }
  const label =
    group.kind === "employment"
      ? "Possible employment entry"
      : group.kind === "education"
        ? "Possible education entry"
        : "Possible certification entry";
  return (
    <article
      id={`entry-${group.id}`}
      aria-label={label}
      className="account-card"
    >
      <h4>{label}</h4>
      {resolved ? (
        <>
          <p>
            Unverified entry link ·{" "}
            {group.method === "local_ocr" ? "OCR text" : "PDF text"}
          </p>
          <dl>
            {observations.map((observation) => (
              <SourceObservation
                key={observation.id}
                observation={observation}
              />
            ))}
          </dl>
        </>
      ) : (
        <p>
          This suggested entry could not be matched to the current source.
          Reload the evidence before reviewing its links.
        </p>
      )}
    </article>
  );
}

const privateHeaders = () =>
  new Headers({
    "Cache-Control": "private, no-store",
    "X-Robots-Tag": "noindex, nofollow",
  });
// Document SSR responses require an explicit route header export; data headers
// alone only cover data requests. Preserve session refresh cookies on every path.
export const headers: HeadersFunction = ({
  parentHeaders,
  loaderHeaders,
  actionHeaders,
  errorHeaders,
}) => {
  const result = new Headers(parentHeaders);
  for (const source of [loaderHeaders, actionHeaders, errorHeaders]) {
    if (!source) continue;
    for (const [key, value] of source)
      if (key.toLowerCase() !== "set-cookie") result.set(key, value);
    for (const cookie of source.getSetCookie())
      result.append("Set-Cookie", cookie);
  }
  result.set("Cache-Control", "private, no-store");
  result.set("X-Robots-Tag", "noindex, nofollow");
  return result;
};
export async function loader({ request, params }: Args) {
  const headers = privateHeaders();
  const user = await getAuthenticatedUser(
    createSupabaseServerClient(request, headers),
  );
  if (!user)
    throw redirect(
      `/auth?next=${encodeURIComponent(`/analysis/${params.id ?? ""}/evidence`)}`,
      { headers },
    );
  try {
    const review = await loadEvidenceReview(
      createSupabaseAdminClient(),
      params.id ?? "",
      user.id,
    );
    const entryAssociations = deriveEvidenceEntryAssociations(
      review.evidence,
      review.targets.binding,
    );
    return data(
      {
        ...review,
        entryAssociations,
        scoringDiagnosticsEnabled: getServerConfig().SCORING_V2_ENABLED,
        canonicalEnabled: getServerConfig().CANONICAL_AUTHORING_ENABLED,
        jobReviewEnabled: getServerConfig().JOB_INTELLIGENCE_ENABLED,
        formKeys: Object.fromEntries(
          CRITICAL_FIELDS.map((field) => [
            field,
            {
              correct: randomUUID(),
              confirm: Object.fromEntries(
                review.evidence.assertions
                  .filter((candidate) => candidate.field === field)
                  .map((candidate) => [candidate.id, randomUUID()]),
              ),
            },
          ]),
        ),
      },
      { headers },
    );
  } catch (error) {
    throw data(
      {
        error:
          error instanceof EvidenceReviewError
            ? error.message
            : "Evidence is unavailable. Try again later.",
      },
      {
        status: error instanceof EvidenceReviewError ? error.status : 503,
        headers,
      },
    );
  }
}
export async function action({ request, params }: Args) {
  const headers = privateHeaders();
  try {
    if (request.method !== "POST")
      return data(
        { error: "Submit a review using POST.", success: null },
        { status: 405, headers },
      );
    assertSameOrigin(request, getServerConfig().APP_ORIGIN);
    const user = await getAuthenticatedUser(
      createSupabaseServerClient(request, headers),
    );
    if (!user)
      return data(
        { error: "Sign in to review your evidence.", success: null },
        { status: 401, headers },
      );
    const input = await readEvidenceCorrectionForm(request);
    await correctEvidenceReview(
      createSupabaseAdminClient(),
      params.id ?? "",
      user.id,
      input,
      randomUUID(),
    );
    return data(
      {
        error: null,
        success:
          "Your review was saved separately from the original extraction. This evidence remains unscored.",
      },
      { headers },
    );
  } catch (error) {
    const status =
      error instanceof EvidenceReviewError
        ? error.status
        : error instanceof PublicApiError
          ? error.status
          : 503;
    const message =
      error instanceof EvidenceReviewError || error instanceof PublicApiError
        ? error.message
        : "The review could not be saved. Retry the same submission.";
    return data({ error: message, success: null }, { status, headers });
  }
}

export function ErrorBoundary() {
  const error = useRouteError();
  return (
    <WorkspaceShell
      eyebrow="Document evidence"
      title="Evidence unavailable"
      description="Your document has not been scored."
    >
      <p role="alert">
        {isRouteErrorResponse(error) && typeof error.data?.error === "string"
          ? error.data.error
          : "Evidence could not be loaded safely."}
      </p>
      <Link to="/upload" className="back-button">
        Return to uploads
