const source = "Built services using Python";
function harness() {
  const d = scoreExtractedResume({
    text: source,
    pageCount: 1,
    pageTexts: [source],
    textChecksum: createHash("sha256").update(source).digest("hex"),
    jobDescription: "Python required",
  });
  const feedback = buildPartialFeedback({
    score: d.score,
    ruleTrace: d.ruleTrace,
  });
  feedback.content.tips[0] = {
    type: "good",
    tip: "Created services using Python",
    explanation: "Clarifies wording.",
    quote: source,
  };
  const record = {
    id: uuid(1),
    organization_id: uuid(3),
    status: "completed",
    analysis_results: {
      feedback,
      parse_view: d.parseView,
      rule_trace: d.ruleTrace,
      keyword_coverage: d.keywordCoverage,
      created_at: "2026-11-15T00:00:00Z",
    },
    analysis_deterministic_results: { ruleset_version: d.rulesetVersion },
    writer_drafts: [{ draft: { summary: null, bullets: [] } }],
    resume_versions: { storage_key: "fixture.pdf", organization_id: uuid(3) },
  };
  const state: { metadata: unknown; reply: unknown; error: unknown } = {
    metadata: {
      ...INITIAL_FEEDBACK_PROFILE,
      schemaReady: false,
      readerExecutable: false,
      prepareExecutable: false,
      claimExecutable: false,
      settleExecutable: false,
      finishExecutable: false,
      usageExecutable: false,
      exportExecutable: false,
    },
    reply: null,
    error: null,
  };
  const rpc = vi.fn(async (name: string) =>
    name === "read_initial_feedback_readiness"
      ? { data: state.metadata, error: null }
      : { data: state.reply, error: state.error },
  );
  const from = vi.fn((table: string) => {
    const builder = {
      select: vi.fn(() => builder),
      eq: vi.fn(() => builder),
      maybeSingle: vi.fn(async () => ({ data: record, error: null })),
      limit: vi.fn(async () => ({
        data: table === "document_evidence_attempts" ? [] : null,
        error: null,
      })),
    };
    return builder;
  });
  mocks.user.mockResolvedValue({ id: uuid(2) });
  mocks.admin.mockReturnValue({ from, rpc });
  mocks.audit.mockResolvedValue(undefined);
  return { d, feedback, record, state, rpc, from };
}
async function load() {
  const { loader } = await import("../../app/routes/resume");
  const url = new URL(`http://localhost/resume/${uuid(1)}`);
  const response = await loader({
    params: { id: uuid(1) },
    request: new Request(url),
    url,
    pattern: "/resume/:id",
    context: new RouterContextProvider(),
  });
  if (response instanceof Response) throw response;
  return response;
}

it.each(["current", "historical"] as const)(
  "projects actual owned %s publication with retained compatibility indices and no private payload",
  async (currentness) => {
    const h = harness(),
      f = ownedFixture();
    h.record.analysis_results.feedback = f.feedback;
    h.feedback = f.feedback;
    admit(h, { ...f.read, currentness });
    const response = await load(),
      p = response.data.feedbackProvenance;
    expect(p).toMatchObject({
      status: "available",
      currentness,
      publicationState: "assessed",
    });
    expect(p.tips.filter((t) => t.category === "toneAndStyle")).toEqual([
      {
        category: "toneAndStyle",
        compatibilityIndex: 0,
        classification: "source_claim",
      },
      {
        category: "toneAndStyle",
        compatibilityIndex: 1,
        classification: "editorial_observation",
      },
      {
        category: "toneAndStyle",
        compatibilityIndex: 2,
        classification: "clarification_question",
      },
    ]);
    expect(f.compiled.plan.toneAndStyle.map((t) => t.originalIndex)).toEqual([
      1, 2, 3,
    ]);
    const groups = visibleFeedbackGroups(f.feedback, false, p);
    expect(
      groups.flatMap((g) => g.tips).filter(isFeedbackStrength),
    ).toHaveLength(1);
    expect(
      groups
        .flatMap((g) => g.tips)
        .filter(isFeedbackRepair)
        .some(
          (t) =>
            t.classification === "clarification_question" ||
            t.classification === "compatibility_only",
        ),
    ).toBe(false);
    expect(Object.keys(p).sort()).toEqual([
      "currentness",
      "publicationState",
      "status",
      "tips",
    ]);
    for (const key of [
      "binding",
      "policy",
      "recordedObservation",
      "ownerId",
      "sourceRefs",
      "lease",
    ])
      expect(JSON.stringify(p)).not.toContain(key);
    expect(response.data.result.feedback).toEqual(f.feedback);
    expect(h.rpc.mock.calls.map((c) => c[0])).toEqual([
      "read_initial_feedback_readiness",
      "read_initial_feedback",
    ]);
    const headers = new Headers(response.init?.headers);
    expect(headers.get("Cache-Control")).toContain("private");
    expect(headers.get("X-Robots-Tag")).toContain("noindex");
  },
);
