function boundedExport(raw: unknown) {
  const stack = [{ value: raw, depth: 0 }];
  let bytes = 0,
    visits = 0;
  while (stack.length) {
    const { value, depth } = stack.pop()!;
    if (++visits > 1000000 || depth > 40) throw Error("EXPORT_LIMIT");
    if (typeof value === "string")
      bytes += Buffer.byteLength(JSON.stringify(value), "utf8");
    else if (value === null || typeof value === "boolean") bytes += 5;
    else if (typeof value === "number") {
      if (!Number.isFinite(value)) throw Error("EXPORT_LIMIT");
      bytes += 32;
    } else if (Array.isArray(value)) {
      if (value.length > 12288) throw Error("EXPORT_LIMIT");
      bytes += value.length + 2;
      for (const v of value) stack.push({ value: v, depth: depth + 1 });
    } else if (typeof value === "object" && value !== null) {
      const proto = Object.getPrototypeOf(value),
        ds = Object.getOwnPropertyDescriptors(value);
      if (proto !== Object.prototype && proto !== null)
        throw Error("EXPORT_LIMIT");
      if (Reflect.ownKeys(value).length !== Object.keys(ds).length)
        throw Error("EXPORT_LIMIT");
      for (const [key, d] of Object.entries(ds)) {
        if (d.get || d.set || !d.enumerable) throw Error("EXPORT_LIMIT");
        bytes += Buffer.byteLength(JSON.stringify(key), "utf8") + 2;
        stack.push({ value: d.value, depth: depth + 1 });
      }
    } else throw Error("EXPORT_LIMIT");
    if (bytes > 33554432) throw Error("EXPORT_LIMIT");
  }
}
export const InitialFeedbackExportSchema = z.preprocess(
  (raw, ctx) => {
    try {
      boundedExport(raw);
      return raw;
    } catch {
      ctx.addIssue({ code: "custom", message: "Export limit." });
      return z.NEVER;
    }
  },
  z.strictObject({
    schemaVersion: z.literal("initial-feedback-export/1.0.0"),
    operations: z.array(operationRow).max(4096),
    attempts: z.array(attemptRow).max(12288),
    publications: z.array(publicationRow).max(4096),
  }),
);
export type InitialFeedbackOwnedExport = z.infer<
  typeof InitialFeedbackExportSchema
>;
/** Paired Node result only; the Native four-key export wire is unchanged. */
export const InitialFeedbackPairedOwnedExportSchema = z.preprocess(

  z.strictObject({

type Options = { signal?: AbortSignal };
type Capability = Parameters<
  typeof validateInitialFeedbackReadiness
>[2][number];
function parse<T>(schema: z.ZodType<T>, raw: unknown, submitted = false): T {
  try {
    const p = schema.safeParse(raw);
    if (p.success) return p.data;
  } catch {
    /* No raw decode errors. */
  }
  throw new InitialFeedbackOperationError(
    submitted ? "unavailable" : "invalid",
    submitted ? "unknown" : "not_sent",
  );
}
export function matchesInitialFeedbackPublication(
  publication: InitialFeedbackPublication,
  operation: InitialFeedbackOperation,
  attemptId: string | null,
  completion: InitialFeedbackCompletion,
): boolean {
  const r = publication.receipt;
  if (
    r.operationId !== operation.operationId ||
    r.attemptId !== attemptId ||
    r.bindingSha256 !== operation.bindingSha256 ||
    !isDeepStrictEqual(publication.binding, operation.binding)
  )
    return false;
  if (completion.kind === "assessed")
    return (
      publication.state === "assessed" &&
      isDeepStrictEqual(publication.plan, completion.plan) &&
      isDeepStrictEqual(
        publication.recordedObservation,
        completion.observation,
      ) &&
      publication.generationFailure === null
    );
  return (
    publication.state === "deterministic_only" &&
    publication.reason === completion.reason &&
    isDeepStrictEqual(
      publication.recordedObservation,
      completion.observation,
    ) &&
    isDeepStrictEqual(
      publication.generationFailure,
      completion.generationFailure,
    )
  );
}
/** Explicit real source profile required; no partial readiness/default pin. */
export function createInitialFeedbackPublicationRepository(
  client: InitialFeedbackOperationClient,
  rawScope:
    | InitialFeedbackPublicationScope
    | InitialFeedbackPublicationReadScope
    | InitialFeedbackPublicationOwnerScope,
  expectedProfile: InitialFeedbackProfile,
  options: RepositoryOptions = {},
) {
  const scope = parse(accessScopeSchema, rawScope),
    profile = parse(InitialFeedbackProfileSchema, expectedProfile);
  const retirementProfile =
    options.retirementProfile === undefined
      ? undefined
      : parse(OwnedAiRetirementProfileSchema, options.retirementProfile);
  const timeoutMs = options.timeoutMs ?? 5000,
    now = options.now ?? Date.now;
  if (
    !client ||
    typeof client.rpc !== "function" ||
    !Number.isInteger(timeoutMs) ||
    timeoutMs < 1 ||
    timeoutMs > 5000 ||
    typeof now !== "function"
  )
    throw new InitialFeedbackOperationError("invalid");
  async function run<T>(
    method: Options,
    work: (
      call: (
        name: string,
        args: Record<string, unknown>,
        mutation?: boolean,
      ) => Promise<unknown>,
      active: () => void,
      signal: AbortSignal,
    ) => Promise<T>,
  ): Promise<T> {
    const { signal } = parse(methodSchema, method),
      controller = new AbortController(),
      started = now();
    let sent = false;
    if (!Number.isFinite(started))
      throw new InitialFeedbackOperationError("invalid");
    const failure = () =>
      new InitialFeedbackOperationError(
        signal?.aborted ? "cancelled" : "unavailable",
        sent ? "unknown" : "not_sent",
      );
    let stop!: (e: Error) => void;
    const stopped = new Promise<never>((_, reject) => {
      stop = reject;
    });
    const abort = () => {
      stop(failure());
      controller.abort();
    };
    const active = () => {
      const at = now();
      if (
        signal?.aborted ||
        controller.signal.aborted ||
        !Number.isFinite(at) ||
        at < started ||
        at >= started + timeoutMs
      )
        throw failure();
    };
    signal?.addEventListener("abort", abort, { once: true });
    let timer: ReturnType<typeof setTimeout> | undefined;
    try {
      active();
      timer = setTimeout(abort, timeoutMs);
      const call = async (
        name: string,
        args: Record<string, unknown>,
        mutation = false,
      ) => {
        active();
        if (mutation) sent = true;
        let reply;
        try {
          const request = client.rpc(name, args);
          reply = await (request.abortSignal
            ? request.abortSignal(controller.signal)
            : request);
        } catch {
          throw failure();
        }
        active();
        if (!reply || typeof reply !== "object") throw failure();
        if (reply.error) {
          const d =
            typeof reply.error === "object"
              ? Object.getOwnPropertyDescriptor(reply.error, "code")
              : null;
          const code = d && "value" in d ? d.value : null;
          const fixed =
            code === "22023"
              ? "invalid"
              : code === "42501"
                ? "not_found"
                : code === "P0C01"
                  ? "stale"
                  : code === "P0C03"
                    ? "unavailable"
                    : null;
          throw new InitialFeedbackOperationError(
            fixed ?? "unavailable",
            sent ? (fixed ? "refused" : "unknown") : "not_sent",
          );
        }
        return reply.data;
      };
      const result = await Promise.race([
        Promise.resolve().then(() => {
          active();
          return work(call, active, controller.signal);
        }),
        stopped,
      ]);
      active();
      return result;
    } catch (e) {
      if (e instanceof InitialFeedbackOperationError) throw e;
      throw failure();
    } finally {
      if (timer) clearTimeout(timer);
      signal?.removeEventListener("abort", abort);
      controller.abort();
    }
  }
  async function ready(
    call: (name: string, args: Record<string, unknown>) => Promise<unknown>,
    capabilities: readonly Capability[],
  ) {
    const raw = await call("read_initial_feedback_readiness", {});
    try {
      return validateInitialFeedbackReadiness(raw, profile, capabilities);
    } catch {
      throw new InitialFeedbackOperationError("unavailable");
    }
  }
  function ownedOperation(op: InitialFeedbackOperation) {
    if (!("analysisId" in scope))
      throw new InitialFeedbackOperationError("invalid");
    if (
      op.ownerId !== scope.ownerId ||
      op.organizationId !== scope.organizationId ||
      op.analysisId !== scope.analysisId
    )
      throw new InitialFeedbackOperationError("unavailable", "unknown");
  }
  return {
    assertReady(capabilities: readonly Capability[], method: Options = {}) {
      return run(method, async (call) => ready(call, capabilities));
    },
    exportOwned(
      method: Options = {},
    ): Promise<InitialFeedbackOwnedExport | InitialFeedbackPairedOwnedExport> {
      return run(method, async (call, active, signal) => {
        await ready(call, ["exportExecutable"]);
        const value = parse(
          InitialFeedbackExportSchema,
          await call("export_initial_feedback_records", {
            p_user_id: scope.ownerId,
          }),
          true,
        );
        const operations = new Map<string, InitialFeedbackOperation>(),
          attemptIds = new Set<string>(),
          publicationIds = new Set<string>();
        for (const row of value.operations) {
          if (
            row.owner_id !== scope.ownerId ||
            operations.has(row.operation_id)
          )
            throw new InitialFeedbackOperationError("unavailable");
          const rows = value.attempts
            .filter((a) => a.operation_id === row.operation_id)
            .sort((a, b) => a.ordinal - b.ordinal);
          const attempts = rows.map((a) => {
            if (
              a.owner_id !== row.owner_id ||
              a.organization_id !== row.organization_id ||
              a.analysis_id !== row.analysis_id ||
              attemptIds.has(a.attempt_id)
            )
              throw new InitialFeedbackOperationError("unavailable");
            attemptIds.add(a.attempt_id);
            if ((a.claimed_at === null) !== (a.claim_lease === null))
              throw new InitialFeedbackOperationError("unavailable");
            return parse(
              InitialFeedbackAttemptSchema,
              {
                attemptId: a.attempt_id,
                ordinal: a.ordinal,
                state: a.state,
                claimedAt: a.claimed_at,
                usage: a.usage,
                errorCode: a.error_code,
                budgetAttemptId: a.budget_attempt_id,
              },
              true,
            );
          });
          const op = parse(
            InitialFeedbackOperationSchema,
            {
              schemaVersion: "initial-feedback-operation/1.0.0",
              operationId: row.operation_id,
              analysisId: row.analysis_id,
              ownerId: row.owner_id,
              organizationId: row.organization_id,
              binding: row.binding,
              bindingSha256: row.binding_sha256,
              state: row.state,
              attemptLimit: row.attempt_limit,
              attempts,
              publicationId: row.publication_id,
            },
            true,
          );
          if (
            row.resume_id !== op.binding.resumeId ||
            row.resume_version_id !== op.binding.resumeVersionId ||
            row.policy_version !== op.binding.policy.version ||
            row.policy_sha256 !== op.binding.policy.sha256
          )
            throw new InitialFeedbackOperationError("unavailable");
          operations.set(row.operation_id, op);
        }
        if (attemptIds.size !== value.attempts.length)
          throw new InitialFeedbackOperationError("unavailable");
        for (const row of value.publications) {
          const op = operations.get(row.operation_id);
          if (
            !op ||
            row.owner_id !== scope.ownerId ||
            row.organization_id !== op.organizationId ||
            row.analysis_id !== op.analysisId ||
            row.resume_id !== op.binding.resumeId ||
            row.resume_version_id !== op.binding.resumeVersionId ||
            row.id !== op.publicationId ||
            row.binding_sha256 !== op.bindingSha256 ||
            publicationIds.has(row.id) ||
            Object.hasOwn(row.publication_core, "receipt")
          )
            throw new InitialFeedbackOperationError("unavailable");
          const p = parse(
            InitialFeedbackPublicationSchema,
            { ...row.publication_core, receipt: row.receipt },
            true,
          );
          publicationIds.add(row.id);
          if (
            !matchesInitialFeedbackPublication(
              p,
              op,
              row.attempt_id,
              row.completion,
            ) ||
            p.publicationId !== row.id ||
            p.state !== row.state ||
            p.reason !== row.reason ||
            row.receipt.runId !== row.run_id ||
            row.receipt.publicationSha256 !== row.publication_sha256 ||
            row.receipt.resultSha256 !== row.result_content_sha256 ||
            (row.receipt.writerSha256 !== null &&
              row.receipt.writerSha256 !== row.writer_content_sha256)
          )
            throw new InitialFeedbackOperationError("unavailable");
          if (
            row.render &&
            (!isDeepStrictEqual(row.render.prose, p.prose) ||
              !isDeepStrictEqual(row.render.parts, p.ownedSupport?.parts) ||
              !isDeepStrictEqual(row.render.mapping, p.ownedSupport?.mapping))
          )
            throw new InitialFeedbackOperationError("unavailable");
        }
        const retiredIds = new Set<string>();
        let retirementRows: readonly OwnedAiRetirementRow[] | undefined;
        if (retirementProfile) {
          // Only the real fresh reader can issue this capability; caller JSON,
          // a callback or a supplied array cannot mint retirement authority.
          const capability = await readOwnedAiRetirementExport(
            client,
            { ownerId: scope.ownerId },
            retirementProfile,
            { timeoutMs, now, signal },
          );
          retirementRows = snapshotOwnedAiRetirementExport(
            capability,
            scope.ownerId,
          ).retirements;
          for (const row of retirementRows) {
            const op = operations.get(row.operation_id);
            if (
              !op ||
              op.state !== "published" ||
              op.publicationId !== row.publication_id ||
              row.owner_id !== scope.ownerId ||
              row.organization_id !== op.organizationId ||
              row.analysis_id !== op.analysisId ||
              row.resume_id !== op.binding.resumeId ||
              row.resume_version_id !== op.binding.resumeVersionId ||
              row.binding_sha256 !== op.bindingSha256 ||
              publicationIds.has(row.publication_id) ||
              value.publications.some(
                (p) => p.operation_id === row.operation_id,
              )
            )
              throw new InitialFeedbackOperationError("unavailable");
            const att =
              row.attempt_id === null
                ? null
                : op.attempts.find((a) => a.attemptId === row.attempt_id);
            if (
              row.publication_state === "assessed"
                ? !att ||
                  att.state !== "published" ||
                  att.usage === null ||
                  att.errorCode !== null ||
                  att.claimedAt === null ||
                  row.run_id === null
                : row.run_id !== null ||
                  (row.attempt_id === null
                    ? op.attempts.some((a) => a.claimedAt !== null)
                    : !att ||
                      !["failed", "held_unknown"].includes(att.state) ||
                      att.errorCode === null ||
                      att.claimedAt === null)
            )
              throw new InitialFeedbackOperationError("unavailable");
            retiredIds.add(row.publication_id);
          }
        }
        if (
          [...operations.values()].some(
            (op) =>
              op.publicationId !== null &&
              !publicationIds.has(op.publicationId) &&
              !retiredIds.has(op.publicationId),
          )
        )
          throw new InitialFeedbackOperationError("unavailable");
        active();
        return retirementRows === undefined
          ? value
          : parse(InitialFeedbackPairedOwnedExportSchema, {
              ...value,
              retirements: retirementRows,
            });
      });
    },
    async read(method: Options = {}): Promise<InitialFeedbackOwnedRead> {
      if (!("analysisId" in scope))
        throw new InitialFeedbackOperationError("invalid");
      return run(method, async (call, active) => {
        await ready(call, ["readerExecutable"]);
        const value = parse(
          InitialFeedbackReadSchema,
          await call("read_initial_feedback", {
            p_user_id: scope.ownerId,
            p_analysis_id: scope.analysisId,
          }),
          true,
        );
        if (value.status === "pending") ownedOperation(value.operation);
        else if (value.status === "legacy_unbound") {
          if (value.analysisId !== scope.analysisId)
            throw new InitialFeedbackOperationError("unavailable");
        } else if (
          value.publication.binding.ownerId !== scope.ownerId ||
          value.publication.binding.organizationId !== scope.organizationId ||
          value.publication.binding.analysisId !== scope.analysisId
        )
          throw new InitialFeedbackOperationError("unavailable");
        active();
        return value;
      });
    },
    async finish(
      rawOperation: InitialFeedbackOperation,
      attemptId: string | null,
      rawCompletion: InitialFeedbackCompletion,
      rawAudit: z.infer<typeof auditSchema>,
      method: Options = {},
    ) {
      if (!("lease" in scope))
        throw new InitialFeedbackOperationError("invalid");
      const operation = parse(InitialFeedbackOperationSchema, rawOperation),
        completion = parse(InitialFeedbackCompletionSchema, rawCompletion),
        audit = parse(auditSchema, rawAudit);
      parse(id.nullable(), attemptId);
      ownedOperation(operation);
      if (
        attemptId !== null &&
        !operation.attempts.some((a) => a.attemptId === attemptId)
      )
        throw new InitialFeedbackOperationError("invalid");
      return run(method, async (call, active) => {
        await ready(call, ["finishExecutable", "exportExecutable"]);
        const value = parse(
          InitialFeedbackFinishSchema,
          await call(
            "finish_initial_feedback",
            {
              p_analysis_id: scope.analysisId,
              p_lease: structuredClone(scope.lease),
              p_operation_id: operation.operationId,
              p_attempt_id: attemptId,
              p_binding_sha256: operation.bindingSha256,
              p_completion: completion,
              p_process_id: audit.processId,
              p_request_id: audit.requestId,
            },
            true,
          ),
          true,
        );
        if (
          !matchesInitialFeedbackPublication(
            value.publication,
            operation,
            attemptId,
            completion,
          )
        )
          throw new InitialFeedbackOperationError("unavailable", "unknown");
        active();
        return value.publication;
      });
    },
  };
}

/** Owned consumer read: exact analysis scope, no worker lease or mutation API. */
export function createInitialFeedbackPublicationReader(
  client: InitialFeedbackOperationClient,
  rawScope: InitialFeedbackPublicationReadScope,
  expectedProfile: InitialFeedbackProfile,
  options: RepositoryOptions = {},
) {
  const scope = parse(readScopeSchema, rawScope);
  const repository = createInitialFeedbackPublicationRepository(
    client,
    scope,
    expectedProfile,
    options,
  );
  return Object.freeze({ read: repository.read });
}
/** Complete owner collection: never invent or filter by analysis/org/lease. */
export function createInitialFeedbackOwnerExporter(
  client: InitialFeedbackOperationClient,
  rawScope: InitialFeedbackPublicationOwnerScope,
  expectedProfile: InitialFeedbackProfile,
  options: RepositoryOptions = {},
) {
  const scope = parse(ownerScopeSchema, rawScope);
  const repository = createInitialFeedbackPublicationRepository(
    client,
    scope,
    expectedProfile,
    options,
  );
  return Object.freeze({ exportOwned: repository.exportOwned });
}
