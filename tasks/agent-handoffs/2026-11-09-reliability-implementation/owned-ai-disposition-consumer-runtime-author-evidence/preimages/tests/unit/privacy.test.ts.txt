const filters: unknown[][] = [];
const ownerId = "00000000-0000-4000-8000-000000000001";
const requestId = "00000000-0000-4000-8000-000000000002";
beforeEach(() => {
  vi.resetAllMocks();
  filters.length = 0;
  mocks.error = { code: "55P03", message: "private barrier detail" };
  mocks.from.mockImplementation((table: string) => {
    const query = {
      select: () => query,
      limit: () => query,
      eq: (field: string, value: string) => {
        filters.push([table, field, value]);
        return query;
      },
      update: (value: unknown) => {
        filters.push([table, "update", value]);
        return query;
      },
      insert: async (value: unknown) => {
        filters.push([table, "insert", value]);
        return { error: mocks.error };
      },
      maybeSingle: async () =>
        table === "memberships"
          ? { data: { organization_id: requestId }, error: null }
          : { data: null, error: mocks.error },
    };
    return query;
  });
});

describe("privacy policy", () => {
  it("uses an explicit versioned AI consent policy", () => {
    expect(AI_CONSENT_POLICY_VERSION).toBe("candidate-ai-v1");
  });

  it("denies AI transfer before any database call when consent is absent", async () => {
    await expect(
      requireAndRecordAiConsent({
        userId: "user-1",
        accepted: false,
        requestId: "request-1",
      }),
    ).rejects.toMatchObject({ code: "FORBIDDEN", status: 403 });
  });
});

describe("privacy mutation contention", () => {
  it("maps actual request creation contention to a safe retryable 503", async () => {
    const error: unknown = await createDataRequest({
      userId: ownerId,
      kind: "deletion",
      requestId: "synthetic-request",
    }).catch((caught: unknown) => caught);
    expect(error).toBeInstanceOf(PublicApiError);
    expect(error).toMatchObject({
      code: "SERVICE_UNAVAILABLE",
      status: 503,
      retryable: true,
      message: "Privacy changes are busy. Try again shortly.",
    });
    expect(error).not.toHaveProperty("cause");
    expect(String(error) + JSON.stringify(error)).not.toContain(
      "private barrier detail",
    );
    expect(mocks.from).not.toHaveBeenCalledWith("audit_events");
    expect(filters).toContainEqual(["memberships", "user_id", ownerId]);
    expect(filters).toContainEqual(["memberships", "role", "owner"]);
  });

  it("maps actual cancellation contention while retaining exact owner and queued predicates", async () => {
    const error: unknown = await cancelDataRequest({
      userId: ownerId,
      requestId,
      auditRequestId: "synthetic-request",
    }).catch((caught: unknown) => caught);
    expect(error).toBeInstanceOf(PublicApiError);
    expect(error).toMatchObject({
      code: "SERVICE_UNAVAILABLE",
      status: 503,
      retryable: true,
    });
    expect(error).not.toHaveProperty("cause");
    expect(String(error) + JSON.stringify(error)).not.toContain(
      "private barrier detail",
    );
    expect(filters).toContainEqual([
      "data_requests",
      "update",
      { status: "cancelled" },
    ]);
    expect(filters).toContainEqual(["data_requests", "id", requestId]);
    expect(filters).toContainEqual(["data_requests", "user_id", ownerId]);
    expect(filters).toContainEqual(["data_requests", "status", "queued"]);
    expect(mocks.from).not.toHaveBeenCalledWith("audit_events");
  });

  it("keeps unknown SQL failures as fixed generic errors without attaching private details", async () => {
    mocks.error = { code: "XX000", message: "private barrier detail" };
    for (const [mutation, message] of [
      [
        () =>
          createDataRequest({
            userId: ownerId,
            kind: "deletion",
            requestId: "synthetic-request",
          }),
        "Privacy request could not be created.",
      ],
      [
        () =>
          cancelDataRequest({
            userId: ownerId,
            requestId,
            auditRequestId: "synthetic-request",
          }),
        "Privacy request could not be cancelled.",
      ],
    ] as const) {
      const error: unknown = await mutation().catch(
        (caught: unknown) => caught,
      );
      expect(error).toBeInstanceOf(Error);
      expect(error).not.toBeInstanceOf(PublicApiError);
      expect(error).toHaveProperty("message", message);
      expect(error).not.toHaveProperty("cause");
      expect(String(error) + JSON.stringify(error)).not.toContain(
        "private barrier detail",
      );
    }
  });
});

const exportUuid = (n: number) =>
  `90000000-0000-4000-8000-${String(n).padStart(12, "0")}`;
const hashText = (value: string) =>
  createHash("sha256").update(value).digest("hex");
function scoringInputRow(
  organizationId = exportUuid(2),
  analysisId = exportUuid(3),
) {
  const text = "Experience\nBuilt services using Python";
  const description =
    "Python experience required.\nFictional original job requirement.";
  const textChecksum = hashText(text);
  const context = {
    schemaVersion: "legacy-scoring-input/1.0.0",
    analysisId,
    ownerId,
    organizationId,
    requestId: null,
    source: {
      resumeId: exportUuid(4),
      resumeVersionId: exportUuid(5),
      fileChecksum: "1".repeat(64),
      bytes: 1234,
      mediaType: "application/pdf",
    },
    extraction: {
      text,
      pageCount: 1,
      pageTexts: [text],
      textChecksum,
      extractorVersion: "fixture-extractor-v1",
      contentSha256: "2".repeat(64),
    },
    job: {
      kind: "owned_job",
      id: exportUuid(6),
      title: "Fictional engineer",
      description,
      companyName: "Fictional company",
      contentSha256: "3".repeat(64),
    },
    binding: {
      schemaVersion: "legacy-scoring-binding/1.0.0",
      analysisId,
      ownerId,
      organizationId,
      resumeId: exportUuid(4),
      resumeVersionId: exportUuid(5),
      fileChecksum: "1".repeat(64),
      textChecksum,
      extractionSha256: "2".repeat(64),
      jobId: exportUuid(6),
      jobSha256: "3".repeat(64),
      inputBindingSha256: "4".repeat(64),
    },
    producer: {
      parserVersion: "parse-sim-v1",
      rulesetVersion: "ats-rules-v4",
      normalizerVersion: "normalizer-v1",
      taxonomyVersion: "skills-taxonomy-v1",
    },
  };
  const result = scoreExtractedResume({
    text,
    pageCount: 1,
    pageTexts: [text],
    jobDescription: description,
    textChecksum,
  });
  return {
    analysis_id: analysisId,
    organization_id: organizationId,
    owner_id: ownerId,
    resume_id: context.source.resumeId,
    resume_version_id: context.source.resumeVersionId,
    extraction_analysis_id: analysisId,
    job_id: context.job.id,
    worker_id: "fixture-scorer",
    attempt: 1,
    lease_fence: exportUuid(7),
    process_id: "fixture-process",
    request_id: "fixture-request",
    input_context: context,
    input_binding_sha256: context.binding.inputBindingSha256,
    result_checksum: result.resultChecksum,
    deterministic_content_sha256: "5".repeat(64),
    created_at: "2026-11-15T00:00:00.123456Z",
  };
}

function privacyExportStore(data: unknown, error: unknown = null) {
  mocks.from.mockImplementation((table: string) => {
    if (table === "analysis_scoring_inputs")
      throw Error("PRIVATE_TABLE_FALLBACK_FORBIDDEN");
    const query = {
      select: vi.fn(() => query),
      eq: vi.fn(async () => ({ data: [], error: null })),
    };
    return query;
  });
  mocks.rpc.mockImplementation(async (name: string) => {
    if (name === "export_legacy_scoring_inputs") return { data, error };
    if (name === "export_resume_workspace_records")
      return {
        data: Object.fromEntries(
          oldWorkspaceCollections.map((key) => [key, []]),
        ),
        error: null,
      };
    if (name === "export_canonical_rewrite_records")
      return {
        data: {
          schemaVersion: "canonical-rewrite-export/1.0.0",
          canonicalRewriteOperations: [],
          canonicalRewriteVersions: [],
          canonicalRewriteHeads: [],
        },
        error: null,
      };
    const paired =
      name === "read_owned_ai_budget_export_readiness"
        ? {
            ...OWNED_AI_BUDGET_EXPORT_PROFILE,
            schemaReady: true,
            exportExecutable: true,
          }
        : name === "export_owned_ai_budget_records"
          ? {
              schemaVersion: "owned-ai-budget-export/1.0.0",
              ownerId: privacyExportRequest.userId,
              operations: [],
              attempts: [],
              events: [],
              reconciliationOperators: [],
              operatorActions: [],
              omissions: {
                sharedPools: "not_subject_attributable",
                periodOpenings: "not_subject_attributable",
                foreignOperatorIdentity: "redacted",
                foreignTargetHistory: "redacted",
              },
            }
          : name === "read_initial_feedback_readiness"
            ? {
                ...INITIAL_FEEDBACK_PROFILE,
                schemaReady: true,
                readerExecutable: true,
                prepareExecutable: true,
                claimExecutable: true,
                settleExecutable: true,
                finishExecutable: true,
                usageExecutable: true,
                exportExecutable: true,
              }
            : name === "export_initial_feedback_records"
              ? {
                  schemaVersion: "initial-feedback-export/1.0.0",
                  operations: [],
                  attempts: [],
                  publications: [],
                }
              : name === "read_owned_ai_retirement_readiness"
                ? {
                    ...OWNED_AI_RETIREMENT_PROFILE,
                    schemaReady: true,
                    exportExecutable: true,
                  }
                : name === "export_owned_ai_retirement_records"
                  ? {
                      schemaVersion: "owned-ai-retirement-export/1.0.0",
                      retirements: [],
                    }
                  : undefined;
    if (paired !== undefined) return { data: paired, error: null };
    throw Error("UNEXPECTED_EXPORT_RPC");
  });
}

describe("original legacy scoring-input privacy export", () => {
  it("includes complete original scoring snapshots across the same owner's historical organizations", async () => {
    const base = scoringInputRow(exportUuid(2), exportUuid(22));
    const noJob = {
      ...base,
      job_id: null,
      input_context: {
        ...base.input_context,
        job: { kind: "none" },
        extraction: { ...base.input_context.extraction, pageTexts: [] },
        binding: {
          ...base.input_context.binding,
          jobId: null,
          jobSha256: null,
        },
      },
      result_checksum: scoreExtractedResume({
        text: base.input_context.extraction.text,
        pageCount: 1,
        pageTexts: [],
        jobDescription: "",
        textChecksum: base.input_context.extraction.textChecksum,
      }).resultChecksum,
    };
    const originalSourceRow = scoringInputRow();
    const rows = [
      originalSourceRow,
      scoringInputRow(exportUuid(20), exportUuid(21)),
      noJob,
    ];
    privacyExportStore({
      schemaVersion: "legacy-scoring-export/1.0.0",
      scoringInputs: rows,
    });
    const manifest =
      await new SupabasePrivacyRequestRepository().buildExportManifest(
        privacyExportRequest,
      );
    expect(manifest.records.legacyScoringInputs).toEqual(rows);
    expect(mocks.rpc).toHaveBeenCalledWith("export_legacy_scoring_inputs", {
      p_user_id: ownerId,
    });
    expect(
      mocks.rpc.mock.calls.filter(
        ([name]) => name === "export_legacy_scoring_inputs",
      ),
    ).toHaveLength(1);
    expect(mocks.from).not.toHaveBeenCalledWith("analysis_scoring_inputs");
    expect(manifest.records.legacyScoringInputs[0]).toHaveProperty(
      "input_context.job.description",
      originalSourceRow.input_context.job.description,
    );
    expect(manifest.records.legacyScoringInputs[0]).toHaveProperty(
      "input_context.requestId",
      null,
    );

      "job_id",

      "input_context.extraction.pageTexts",
